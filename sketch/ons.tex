In this note, we aim to improve the dimension dependence of bandit convex optimization (\bco).
The main theoretical tool we use is the analysis of a continuous version of mirror descent (\mdalg).
Then we move on to \bco.
The current state of affairs can be summarized as follows:
\begin{enumerate}
    \item \oco\footnote{Online Convex Optimization} reduces to \olo\footnote{Online Linear Optimization}: $f(x) - f(x_\star) \leq \langle f'(x), x - x_\star \rangle$.
    \item \textbf{Random} algs are \textit{suboptimal} for \texttt{OCO} and \textit{pointless} for \texttt{OLO}.
    \item First order information, $f'(x)$, makes \texttt{BCO} as hard as \texttt{OLO}.
    \item \textbf{Random} algs are \textit{necessary} for \texttt{BCO} and \texttt{BLO}.
    \item \textbf{Random} algs are \textit{free} for \texttt{BLO}\footnote{Bandit Linear Optimization}
    \item \textbf{Random} algs appear to be \textit{expensive} for \texttt{BCO}: upper bound is $d^{2.5}\sqrt{n}$ while lower bound is $d\sqrt{n}$.
\end{enumerate}
Our main goal here is to prove a $d^{1.5} \sqrt{n}$ regret upper bound for \bco,
which partially closes the gap between the upper and lower bounds for \bco.

\section{Extension and quadratic approximation}
The regret is defined as usual as
\begin{align}
    \Reg_n(x_\star) = \sum_{t=1}^n e_t(X_t) - e_t(x_\star)\,.
\end{align}
The major goal we pursue here is to fully get rid of the optimistic kernel analysis.

\subsection{Quadratic projection}
Fix a generic round $t$.
The algorithm plays $X \sim p = \cN(\mu, \Sigma)$.
For simplicity, suppose that $e \in \cC^2(\Rd)$.
Define the following quantities:
\begin{align}
    A \triangleq \E[e(X)]\,, \qquad
    g \triangleq \E\left[\nabla e(X)\right]\,, \qquad\text{and}\qquad
    H \triangleq \E\left[\nabla^2 e(X)\right]\,.
\end{align}
Then, the degree-two Gaussian Hermite projection of $e$ under $\cN(\mu, \Sigma)$ is given by
\begin{align}
    q(x) \triangleq \left(A - \frac{\tr(H\Sigma)}{2}\right) + \langle g, x-\mu\rangle + \frac{1}{2}\|x-\mu\|^2_{H}\,.
\end{align}

Note that $q$ matches the first and second moments of $e$ under $\cN(\mu, \Sigma)$:
\begin{align}
    \E[q(X)]           = A\,, \qquad
    \E[\nabla q(X)]    = g\,, \qquad\text{and}\qquad
    \E[\nabla^2 q(X)]  = H\,.
\end{align}
Moreover,
\begin{align}
    \E[q(X)] - q(\mu) = \frac{\E[\|X-\mu\|^2_H]}{2} = \frac{\tr(H\Sigma)}{2}\,.
\end{align}
Convexity of $e$ gives
\begin{align}
    \E[e(X)] - e(x_\star)
     & =
    \E\left[\langle \nabla e(X), X - \mu\rangle\right] + \langle g, \mu - x_\star\rangle
    - \E[D_e(x_\star \| X)]
    \label{eq:regret_switch_convex}
    \\
     & =
    \tr(H\Sigma) + \langle g, \mu - x_\star\rangle - \E[D_e(x_\star \| X)]
    \\
     & =
    q(\mu) - q(x_\star) + \tr(H\Sigma) + \frac12 \|x_\star - \mu\|^2_H - \E[D_e(x_\star \| X)]
    \,,
    \label{eq:quadratic_regret_switch}
\end{align}
where the first equality follows from Gaussian integration by parts,
and the second equality follows from the definition of $q$.

\subsubsection{The projected ONS algorithm}
The setting studied is the following: at each round $t$, the algorithm plays $X_t \sim \cN(\mu_t, \Sigma_t)$, and observes the full loss function $e_t$.
The algorithm then computes the Hermite projection $q_t$ of $e_t$ under $\cN(\mu_t, \Sigma_t)$, and updates $\mu_{t+1}$ and $\Sigma_{t+1}$ using the Online Newton Step algorithm on the sequence of quadratics $q_t$.
It will be convenient to allow for a damping factor $\lambda > 0$ in the update of $\Sigma_t$, which leads us to define the quadratics
\begin{align}
    q^\lambda(x) \triangleq \left(A - \frac{\lambda \tr(H\Sigma)}{2}\right) + \langle g, x-\mu\rangle + \frac{\lambda}{2}\|x-\mu\|^2_{H}\,.
\end{align}
\begin{algorithm}[h!]
    \begin{minipage}{13cm}
        \begin{mdframed}
            \begin{lstlisting}
args: $\eta, \lambda, \Sigma_1^{-1} \succ 0, \mu_1 \in \cK$
for $t = 1$ to $n$:
    Play $X_t \sim \cN(\mu_t, \Sigma_t)$
    Observe the full loss function $e_t$ and incur loss $e_t(X_t)$
    Compute $g_t = \E[\nabla e_t(X_t)]$ and $H_t = \E[\nabla^2 e_t(X_t)]$
    Update $\Sigma_{t+1}^{-1} = \Sigma_t^{-1} + \lambda \eta H_t$
    Update $\mu_{t+1} = \argmin_{\mu \in \cK} \| \mu - (\mu_t - \eta \Sigma_{t+1} g_t) \|_{\Sigma_{t+1}^{-1}}$
\end{lstlisting}
            \caption{Hermite-ONS algorithm.}\label{alg:HermiteOnlineNewtonStep}
        \end{mdframed}
    \end{minipage}
\end{algorithm}

\begin{proposition}
    \label{prop:hermite-ons:master}
    Suppose that each $e_t:\R^d\to\R$ is convex and twice differentiable.
    Then for every $x\in\cK$,
    \begin{align}
        \E\left[
            \sum_{t=1}^{n}
            e_t(X_t) - e_t(x)
            \right]
        \leq
         & \frac{1}{2\eta}
        \left[
        \norm{\mu_1-x}_{\Sigma_1^{-1}}^2
        -
        \norm{\mu_{n+1}-x}_{\Sigma_{n+1}^{-1}}^2
        \right]
        +\frac{\eta}{2}\sum_{t=1}^n\norm{g_t}_{\Sigma_{t+1}}^2
        \\
         &
        +\sum_{t=1}^n\tr(H_t\Sigma_t)
        +\sum_{t=1}^n
        \left[
            \frac{\lambda}{2}\norm{x-\mu_t}_{H_t}^2
            - \E[D_{e_t}(x\|X_t)]
            \right]
        \label{eq:hermite-ons:master}
    \end{align}
\end{proposition}

\begin{proof}
    From \cref{eq:quadratic_regret_switch}, we have
    \begin{align}
        \E_{t-1}[e_t(X_t)]-e_t(x)
         & =
        q_t^\lambda(\mu_t)-q_t^\lambda(x)
        +\tr(H_t\Sigma_t)
        +\frac{\lambda}{2}\norm{x-\mu_t}_{H_t}^2
        - \E_{t-1}[D_{e_t}(x\|X_t)].
    \end{align}
    Applying the standard ONS calculation to the quadratics $q_t^\lambda$ gives
    \begin{align}
        \frac12 \|\mu_{t+1}-x\|_{\Sigma_{t+1}^{-1}}^2
         & \leq
        \frac12 \|x - (\mu_t - \eta \Sigma_{t+1} g_t)\|_{\Sigma_{t+1}^{-1}}^2
        \\
         & =
        \frac12 \|x - \mu_t\|_{\Sigma_{t+1}^{-1}}^2
        + \frac{\eta^2}{2}\|g_t\|_{\Sigma_{t+1}}^2
        + \eta \langle g_t, x - \mu_t\rangle
        \\
         & =
        \frac12 \|x - \mu_t\|_{\Sigma_{t}^{-1}}^2
        + \frac{\eta^2}{2}\|g_t\|_{\Sigma_{t+1}}^2
        + \eta \langle g_t, x - \mu_t\rangle
        + \frac{\lambda \eta}{2}\|x - \mu_t\|_{H_t}^2
        \\
         & =
        \frac12 \|x - \mu_t\|_{\Sigma_{t}^{-1}}^2
        + \frac{\eta^2}{2}\|g_t\|_{\Sigma_{t+1}}^2
        - \eta \left(q_t^\lambda(\mu_t)-q_t^\lambda(x)\right) .\label{eq:alternative-form}
    \end{align}
    Summing over $t$ and rearranging gives
    \begin{align}
        \sum_{t=1}^n\left(q_t^\lambda(\mu_t)-q_t^\lambda(x)\right)
        \leq
        \frac{1}{2\eta}
        \left[
        \norm{\mu_1-x}_{\Sigma_1^{-1}}^2
        - \norm{\mu_{n+1}-x}_{\Sigma_{n+1}^{-1}}^2
        \right]
        +\frac{\eta}{2}\sum_{t=1}^n\norm{g_t}_{\Sigma_{t+1}}^2 .
        \label{eq:hermite-ons:qreg}
    \end{align}
\end{proof}
\begin{proposition}[Direct control of the Bregman bracket]
    \label{prop:hermite-ons:bracket}
    Fix $x\in\R^d$, $\delta\in(0,1]$, and $0<\lambda\leq1/8$.
    Suppose that
    \begin{align}
        \lambda
        \left(
        2d\log 2+4\log\frac{2}{\delta}
        \right)
         & \leq \frac{\log 2}{12},
        \label{eq:hermite-ons:bracket-dimension}
        \\
        \lambda
        \|\mu-x\|_{\Sigma^{-1}}
        \sqrt{\log\frac{2\sqrt{2}}{\delta}}
         & \leq \frac{\log 2}{24}.
        \label{eq:hermite-ons:bracket-focus}
    \end{align}
    Then
    \begin{align}
        \frac{\lambda}{2}\|\mu-x\|_H^2
        -\E[D_e(x\|X)]
        \leq &
        \frac{\log 2}{12}
        \tr(H\Sigma)
        +
        \frac{\lambda L\delta}{2\sqrt{\pi}}
        \|\mu-x\|
        \|\mu-x\|_{\Sigma^{-1}}.
        \label{eq:hermite-ons:bracket}
    \end{align}
\end{proposition}
\begin{proof}
    We first transform the problem so that the Gaussian is standard.
    Define
    \begin{align}
        a=\Sigma^{-1/2}(x - \mu),
        \qquad
        f(z)=e(\mu+\Sigma^{1/2}z),
    \end{align}
    and let $W = \Sigma^{-1/2}(X-\mu)$, so that $W\sim\cN(0,I_d)$.
    So we have $f(W)=e(X)$, $\nabla f(W)=\Sigma^{1/2}\nabla e(X)$, and $\nabla^2 f(W)=\Sigma^{1/2}\nabla^2 e(X)\Sigma^{1/2}$.

    Writing the left-hand side of \eqref{eq:hermite-ons:bracket} using the above definitions gives
    \begin{align}
        \frac{\lambda}{2}\|\mu-x\|_{\E[\nabla^2e(X)]}^2
        -\E[D_e(x\|X)]
        =
        \frac{\lambda}{2}\|a\|_{\E[\nabla^2f(W)]}^2
        -\E[D_f(a\|W)]\,,
    \end{align}
    where we upper bound the first term using the triangle inequality as
    \begin{align}
        \frac{\lambda}{2}
        \| a \|_{\E[\nabla^2 f(W)]}^2
        =
        \frac{\lambda}{2}
        \int_{\R^d} \|a\|_{\nabla^2 f(z)}^2 \varphi(z)\,\d z
        \leq
        \lambda
        \int_{\R^d} \|z\|_{\nabla^2 f(z)}^2 \varphi(z)
        +
        \|z - a\|_{\nabla^2 f(z)}^2 \varphi(z)\,\d z\,.
        \label{eq:bracket-triangle}
    \end{align}
    On the other hand the Bregman divergence can be written as the curvature accumulated
    along the segment from $a$ to $W$ as
    \begin{align}
        \E[D_f(a\|W)]
         & =
        \E\left[
        \int_0^1
        s\,
        \|W - a\|_{\nabla^2f(sW + (1-s)a)}^2
        \,ds
        \right]
        \\
         & =
        \int_{\R^d}
        \int_0^1
        s\,
        \|z - a\|_{\nabla^2f(sz + (1-s)a)}^2
        \varphi(z)
        \,\d z
        \,\d s
        \label{eq:bracket-bregman-integral}
    \end{align}
    For $s\in(0,1]$, let $p_s$ denote the density of
    $sW + (1-s) a$, so that
    \begin{align}
        \E[D_f(a\|W)]
        =
        \int_0^1\frac{1}{s}
        \int_{\R^d}
        \|z - a\|_{\nabla^2f(z)}^2
        p_s(z)\,dz\,ds.
        \label{eq:bracket-density-integral}
    \end{align}
    We will show that on a suitable region of the $(s,z)$-space, the density $p_s$ is comparable with the standard Gaussian density $\varphi$,
    and then the bregman term can be used to pay for the $\|z - a\|_{\nabla^2f(z)}^2$ term in \eqref{eq:bracket-triangle},
    and the remaining $\|z\|_{\nabla^2f(z)}^2$ term can be paid for by the trace term.
    The region of interest consists of $s\in[1-2\lambda,1]$ and $z$ in a suitable set $\mathcal A$ defined as
    \begin{align}
        \mathcal A
        =
        \left\{
        z\in\R^d:
        \begin{array}{l}
            \displaystyle
            \|z\|^2
            \leq
            2d\log 2+4\log\frac{2}{\delta},
            \\[0.2cm]
            \displaystyle
            \left|
            \left\langle\frac{a}{\|a\|},z\right\rangle
            \right|^2
            \leq
            4\log\frac{2\sqrt{2}}{\delta}
        \end{array}
        \right\}.
        \label{eq:bracket-set}
    \end{align}

    \paragraph{Density comparison.}
    Since $sW + (1-s) a\sim\cN((1-s)a, s^2I_d)$, a direct calculation gives
    \begin{align}
        \log\frac{p_s(z)}{\varphi(z)}
        ={} &
        -d\log s
        -\frac{1-s^2}{2s^2}\|z\|^2
        +\frac{1-s}{s^2}\langle a,z\rangle
        -\frac{(1-s)^2}{2s^2}\|a\|^2.
        \label{eq:bracket-density-ratio}
    \end{align}
    If $s\in[1-2\lambda,1]$, then $\lambda\leq1/8$ implies
    \begin{align}
        \frac{1-s^2}{2s^2}\leq4\lambda,
        \qquad
        \frac{1-s}{s^2}\leq4\lambda,
        \qquad
        \frac{(1-s)^2}{2s^2}\leq4\lambda^2.
    \end{align}
    Since $-d\log s\geq0$, we therefore obtain
    \begin{align}
        \log\frac{p_s(z)}{\varphi(z)}
        \geq
        -4\lambda\|z\|^2
        -4\lambda|\langle a,z\rangle|
        -4\lambda^2\|a\|^2.
        \label{eq:bracket-density-ratio-lower}
    \end{align}
    Now, \eqref{eq:hermite-ons:bracket-dimension} and \eqref{eq:hermite-ons:bracket-focus}
    for any $z\in\mathcal A$ give
    \begin{align}
        4\lambda\|z\|^2
        \leq
        \frac{\log2}{3}\,,
        \qquad
        4\lambda|\langle a,z\rangle|
         & \leq
        8\lambda\|a\|
        \sqrt{\log\frac{2\sqrt2}{\delta}}
        \leq
        \frac{\log2}{3}.
    \end{align}
    Finally,
    \begin{align}
        4\lambda^2\|a\|^2
         & \leq
        \frac{(\log2)^2}
        {144\log(2\sqrt2/\delta)}
        \leq
        \frac{\log2}{144}.
    \end{align}
    The sum of these three upper bounds is smaller than $\log2$.
    Hence \eqref{eq:bracket-density-ratio-lower} implies
    \begin{align}
        p_s(z)\geq\frac12\varphi(z)
        \qquad
        \text{for every }
        s\in[1-2\lambda,1],\ z\in\mathcal A.
        \label{eq:bracket-density-comparison}
    \end{align}
    We may now return to the Bregman divergence restricted to
    $s\in[1-2\lambda,1]$ and $z\in\mathcal A$ using
    \eqref{eq:bracket-density-comparison}, gives
    \begin{align}
        \E[D_f(a\|W)]
         & \geq
        \frac12
        \int_{1-2\lambda}^1\frac{1}{s}\,ds
        \int_{\mathcal A}
        \|z - a\|_{\nabla^2f(z)}^2
        \varphi(z)\,dz
        \geq
        \lambda
        \int_{\mathcal A}
        \|z - a\|_{\nabla^2f(z)}^2
        \varphi(z)\,dz,
        \label{eq:bracket-bregman-lower}
    \end{align}
    where the last inequality follows from
    $-\log(1-2\lambda)\geq2\lambda$.

    \paragraph{The trace term.}
    The $\|z\|_{\nabla^2f(z)}^2$ term in \eqref{eq:bracket-triangle} is paid by the trace.
    Indeed, on $\mathcal A$,
    \begin{align}
        \lambda\|z\|_{\nabla^2f(z)}^2
         & \leq
        \lambda\|z\|^2\tr(\nabla^2f(z))
        \\
         & \leq
        \lambda\left(
        2d \log 2 + 4 \log \frac{2}{\delta}
        \right)\tr(\nabla^2f(z))
        \\
         & \leq
        \frac{\log2}{12}\tr(\nabla^2f(z))\,,
    \end{align}
    so we have
    \begin{align}
        \lambda \int_{\mathcal{A}} \|z\|_{\nabla^2 f(z)}^2 \varphi(z)\,dz
        \leq
        \frac{\log(2)}{12}
        \tr(\E[\nabla^2 f(W)])
        =
        \frac{\log(2)}{12}
        \tr(\Sigma H)\,,
    \end{align}
    which combined with \eqref{eq:bracket-bregman-lower} gives
    \begin{align}
        \frac{\lambda}{2}
        \|a\|_{\E[\nabla^2f(W) \mathbbm{1}_{\mathcal A}(W)]}^2
         & =
        \frac{\lambda}{2}
        \int_{\mathcal A}
        \|a\|_{\nabla^2f(z)}^2
        \varphi(z)\,\d z \\
         & \leq
        \int_{\mathcal A}
        \lambda
        \|z - a\|_{\nabla^2f(z)}^2
        +
        \lambda
        \|z\|_{\nabla^2f(z)}^2
        \varphi(z)\,\d z \\
         & \leq
        \frac{\log(2)}{12}
        \tr(\Sigma H)
        +
        \E[D_f(a\|W)].
    \end{align}

    \paragraph{Curvature from $\mathcal{A}^c$}.
    It remains to control $\|a\|_{\E[\nabla^2f(W) \mathbbm{1}_{\mathcal A^c}(W)]}^2$.
    Write $u=a/\|a\|$, and let $\varphi_C$ denote the density of
    $\cN(0,C)$. The thresholds defining $\mathcal A$ were chosen so
    that the standard Gaussian density on each part of
    $\mathcal A^c$ is dominated by a wider Gaussian density. More
    precisely,
    \begin{align}
        \varphi(z)
        \mathbbm{1}
        \left\{
        \|z\|^2>
        2d\log 2+4\log\frac{2}{\delta}
        \right\}
         & \leq
        \frac{\delta}{2}\varphi_{2I_d}(z),
        \label{eq:bracket-radial-tail}
        \\
        \varphi(z)
        \mathbbm{1}
        \left\{
        |\langle u,z\rangle|^2>
        4\log\frac{2\sqrt{2}}{\delta}
        \right\}
         & \leq
        \frac{\delta}{2}\varphi_{I_d+uu^\top}(z).
        \label{eq:bracket-directional-tail}
    \end{align}
    Indeed,
    \begin{align}
        \frac{\varphi(z)}{\varphi_{2I_d}(z)}
         & =
        2^{d/2}\exp\left(-\frac{\|z\|^2}{4}\right),
        \\
        \frac{\varphi(z)}{\varphi_{I_d+uu^\top}(z)}
         & =
        \sqrt2
        \exp\left(-\frac{\langle u,z\rangle^2}{4}\right),
    \end{align}
    and the bounds in
    \eqref{eq:bracket-radial-tail}--%
    \eqref{eq:bracket-directional-tail} follow by substituting the
    corresponding thresholds.

    We now bound the average curvature under either of these wider
    Gaussians. Lipschitzness of $e$ implies
    \begin{align}
        |\langle a,\nabla f(z)\rangle|
         & =
        |\langle\mu-x,
        \nabla e(\mu+\Sigma^{1/2}z)\rangle|
        \leq
        L\|\mu-x\|.
        \label{eq:bracket-directional-lipschitz}
    \end{align}
    For $C=2I_d$ and $C=I_d+uu^\top$, Gaussian integration by parts gives
    \begin{align}
        \int_{\R^d}
        \|a\|_{\nabla^2f(z)}^2
        \varphi_C(z)\,dz
         & =
        \int_{\R^d}
        \langle a,\nabla f(z)\rangle
        \langle C^{-1}a,z\rangle
        \varphi_C(z)\,dz
        \nonumber \\
         & \leq
        L\|\mu-x\|
        \sqrt{\frac{2}{\pi}}
        \sqrt{a^\top C^{-1}a}
        \nonumber \\
         & =
        \frac{L}{\sqrt\pi}
        \|\mu-x\|\|a\|.
        \label{eq:bracket-tail-curvature}
    \end{align}
    In the last equality we used $a^\top C^{-1}a=\frac12\|a\|^2$ for both choices of $C$.
    Since $\mathcal A^c$ is contained in the union of the two events
    in \eqref{eq:bracket-radial-tail} and
    \eqref{eq:bracket-directional-tail}, the preceding bounds imply
    \begin{align}
        \frac{\lambda}{2}\ \|a\|_{\E[\nabla^2f(W) \mathbbm{1}_{\mathcal A^c}(W)]}^2
         & \leq
        \frac{\lambda \delta}{4} \E_{Z\sim\cN(0,2I_d)}[\|a\|_{\nabla^2f(Z)}^2]
        +
        \frac{\lambda \delta}{4} \E_{Z\sim\cN(0,I_d+uu^\top)}[\|a\|_{\nabla^2f(Z)}^2]
        \nonumber \\
         & \leq
        \frac{\lambda L\delta}{2\sqrt{\pi}}
        \|\mu-x\|
        \|\mu-x\|_{\Sigma^{-1}}.
        \label{eq:bracket-outside-set}
    \end{align}
    Rearranging proves \eqref{eq:hermite-ons:bracket}.
\end{proof}

\paragraph{Controlling the trace term.}
The covariance update continually shrinks $\Sigma_t$. The following
result shows that the total curvature encountered during this shrinkage
is controlled by the increase of $\det(\Sigma_t^{-1})$.

\begin{proposition}[Trace budget]
    \label{prop:hermite-ons:trace-budget}
    If
    \begin{align}
        \lambda\eta L\sigma\sqrt{\frac{2}{\pi}}\leq1,
        \label{eq:hermite-ons:trace-step-size}
    \end{align}
    then
    \begin{align}
        \eta\sum_{t=1}^n\tr(H_t\Sigma_t)
        \leq
        \frac{4d}{\lambda}
        \log\left(
        1+\frac{n\lambda\eta L\sigma}{\sqrt{2\pi}}
        \right).
        \label{eq:hermite-ons:trace-budget}
    \end{align}
\end{proposition}
\begin{proof}
    Let $Z\sim\cN(0,I_d)$, so that
    $X_t=\mu_t+\Sigma_t^{1/2}Z$. For every unit vector $v$, Gaussian
    integration by parts gives
    \begin{align}
        v^\top\Sigma_t^{1/2}H_t\Sigma_t^{1/2}v
         & =
        \E\left[
            \left\langle
            \nabla e_t(X_t),\Sigma_t^{1/2}v
            \right\rangle
            \langle v,Z\rangle
            \right]
        \leq
        L\|\Sigma_t^{1/2}v\|
        \E|\langle v,Z\rangle|
        \leq
        L\sigma\sqrt{\frac{2}{\pi}}.
        \label{eq:hermite-ons:normalized-hessian}
    \end{align}
    Thus, by \eqref{eq:hermite-ons:trace-step-size}, every eigenvalue
    of $\lambda\eta \Sigma_t^{1/2}H_t\Sigma_t^{1/2}$ belongs to $[0,1]$.
    Now, using $\log(1+u) \geq u/2$ for $u\in[0,1]$, we have
    \begin{align}
        \log\frac{\det(\Sigma_{t+1}^{-1})}
        {\det(\Sigma_t^{-1})}
         & =
        \log\det\left(
        I_d+\lambda\eta
        \Sigma_t^{1/2}H_t\Sigma_t^{1/2}
        \right) \geq
        \frac{\lambda\eta}{2}\tr(H_t\Sigma_t).
        \label{eq:hermite-ons:trace-logdet-lower}
    \end{align}

    It remains to upper-bound how much the determinant can grow.
    A second application of Gaussian integration by parts gives, for
    every unit vector $v$,
    \begin{align}
        v^\top H_tv
        =
        \E\left[
            \langle v,\nabla e_t(X_t)\rangle
            \left\langle
            v,\Sigma_t^{-1}(X_t-\mu_t)
            \right\rangle
            \right]
        \leq
        L\sqrt{\frac{2}{\pi}}
        \sqrt{v^\top\Sigma_t^{-1}v},
    \end{align}
    and hence
    $\lambda_{\max}(H_t)
        \leq
        L\sqrt{\frac{2}{\pi}}
        \sqrt{\lambda_{\max}(\Sigma_t^{-1})}.$
    The covariance update now implies
    \begin{align}
        \lambda_{\max}(\Sigma_{t+1}^{-1})
        \leq{} &
        \lambda_{\max}(\Sigma_t^{-1})
        +
        \lambda\eta L\sqrt{\frac{2}{\pi}}
        \sqrt{\lambda_{\max}(\Sigma_t^{-1})}.
    \end{align}
    Since $\sqrt{r^2+br}\leq r+b/2$ for $r,b\geq0$,
    \begin{align}
        \sqrt{\lambda_{\max}(\Sigma_{t+1}^{-1})}
        \leq
        \sqrt{\lambda_{\max}(\Sigma_t^{-1})}
        +
        \frac{\lambda\eta L}{\sqrt{2\pi}}.
    \end{align}
    Iterating and using $\Sigma_1^{-1}=\sigma^{-2}I_d$ gives
    \begin{align}
        \lambda_{\max}(\Sigma_{n+1}^{-1})
        \leq
        \frac{1}{\sigma^2}
        \left(
        1+\frac{n\lambda\eta L\sigma}{\sqrt{2\pi}}
        \right)^2.
    \end{align}
    Consequently,
    \begin{align}
        \log\frac{\det(\Sigma_{n+1}^{-1})}
        {\det(\Sigma_1^{-1})}
        \leq
        2d\log\left(
        1+\frac{n\lambda\eta L\sigma}{\sqrt{2\pi}}
        \right).
        \label{eq:hermite-ons:trace-logdet-upper}
    \end{align}
    Summing \eqref{eq:hermite-ons:trace-logdet-lower} and applying
    \eqref{eq:hermite-ons:trace-logdet-upper} proves the result.
\end{proof}

\paragraph{Preserving the focus condition.}
We now specialize to a fixed loss $e$. The important consequence of
stationarity is that
\[
    \E[e(X_t)]\geq e(\mu_t)\geq e(x_\star).
\]
As we will see, this expected regret enters the potential recursion with
a negative sign and therefore helps keep the iterates inside the focus
region.

\begin{proposition}[Preservation of the focus condition]
    \label{prop:hermite-ons:focus}
    Let $\cK\subseteq\R^d$ be closed and convex with
    $\operatorname{diam}(\cK)\leq D$.
    Run
    \cref{alg:HermiteOnlineNewtonStep} with the fixed loss $e_t=e$,
    starting from $\mu_1\in\cK$ and $\Sigma_1=\sigma^2I_d$.
    Suppose that
    \begin{align}
        0<\lambda\leq\frac18,
        \qquad
        \lambda
        \left(
        2d\log2+4\log(2n)
        \right)
        \leq
        \frac{\log2}{12},
        \qquad
        \text{and}\qquad
        \lambda\eta L\sigma\sqrt{\frac{2}{\pi}}
        \leq1.
        \label{eq:hermite-ons:focus-dimension}
    \end{align}
    Suppose, moreover, that
    \begin{align}
         & \frac12
        \|\mu_1-x_\star\|_{\Sigma_1^{-1}}^2
        +
        \frac{\eta^2nL^2\sigma^2}{2}
        \nonumber  \\
         & \quad+
        \frac{
            4\left(1+\frac{\log2}{12}\right)d
        }{\lambda}
        \log\left(
        1+\frac{n\lambda\eta L\sigma}{\sqrt{2\pi}}
        \right)
        \nonumber  \\
         & \quad+
        \frac{\eta LD\log2}
        {48\sqrt{\pi\log(2\sqrt2\,n)}}
        \leq
        \frac{(\log2)^2}
        {1152\lambda^2\log(2\sqrt2\,n)}.
        \label{eq:hermite-ons:focus-sufficient}
    \end{align}
    Then, for every $t\in\{1,\ldots,n+1\}$,
    \begin{align}
        \|\mu_t-x_\star\|_{\Sigma_t^{-1}}^2
        \leq
        \frac{\log2}
        {24\lambda\sqrt{\log(2\sqrt2\,n)}}.
        \label{eq:hermite-ons:focus-preserved}
    \end{align}
    Consequently, \cref{prop:hermite-ons:bracket} is applicable with
    $x=x_\star$ and $\delta=1/n$ at every round.
\end{proposition}

\begin{proof}
    We first derive the one-step recursion. Since $\mu_{t+1}$ is the
    $\Sigma_{t+1}^{-1}$-metric projection of $\mu_t-\eta\Sigma_{t+1}g_t$
    onto $\cK$, and $x_\star\in\cK$, the Pythagorean inequality gives
    \begin{align}
        \frac12
        \|\mu_{t+1}-x_\star\|_{\Sigma_{t+1}^{-1}}^2
         & \leq
        \frac12
        \|\mu_t-\eta\Sigma_{t+1}g_t-x_\star
        \|_{\Sigma_{t+1}^{-1}}^2
        \nonumber \\
         & =
        \frac12
        \|\mu_t-x_\star\|_{\Sigma_t^{-1}}^2
        -\eta\langle g_t,\mu_t-x_\star\rangle
        +
        \frac{\lambda\eta}{2}
        \|\mu_t-x_\star\|_{H_t}^2
        +
        \frac{\eta^2}{2}
        \|g_t\|_{\Sigma_{t+1}}^2.
        \label{eq:hermite-ons:focus-projection}
    \end{align}
    Further, \eqref{eq:regret_switch_convex} together with $x_\star$ being a minimizer of $e$ gives
    \begin{align}
        0
        \leq
        \E[e(X_t)]-e(x_\star)
        =
        \tr(H_t\Sigma_t)
        +
        \langle g_t,\mu_t-x_\star\rangle
        -
        \E[D_e(x_\star\|X_t)].
        \label{eq:hermite-ons:focus-regret-switch}
    \end{align}
    Rearranging this identity gives
    \begin{align}
        -\langle g_t,\mu_t-x_\star\rangle
        \leq
        \tr(H_t\Sigma_t)
        -
        \E[D_e(x_\star\|X_t)]\,,
        \label{eq:hermite-ons:focus-rearrangement}
    \end{align}
    which we use to control the inner product term in \eqref{eq:hermite-ons:focus-projection}.
    % which together with \eqref{eq:hermite-ons:focus-projection} gives
    % \begin{align}
    %     \frac12
    %     \|\mu_{t+1}-x_\star\|_{\Sigma_{t+1}^{-1}}^2
    %     -
    %     \frac12
    %     \|\mu_t-x_\star\|_{\Sigma_t^{-1}}^2
    %     \leq
    %     \frac{\eta^2}{2}
    %     \|g_t\|_{\Sigma_{t+1}}^2
    %     +
    %     \eta\tr(H_t\Sigma_t)
    %     +
    %     \eta\left(
    %     \frac{\lambda}{2}
    %     \|\mu_t-x_\star\|_{H_t}^2
    %     -
    %     \E[D_e(x_\star\|X_t)]\right).
    %     \label{eq:hermite-ons:focus-recursion-first}
    % \end{align}
    On the other hand, suppose for the moment that the condition required by \cref{prop:hermite-ons:bracket} holds at round $t$ with
    $\delta=1/n$, i.e.,
    \begin{align}
        \lambda
        \|\mu_t-x_\star\|_{\Sigma_t^{-1}}
        \sqrt{\log(2\sqrt2\,n)}
        \leq
        \frac{\log2}{24}
        \label{eq:hermite-ons:focus-condition-round}
    \end{align}
    so from \cref{prop:hermite-ons:bracket} we have
    \begin{align}
        \frac{\lambda}{2}
        \|\mu_t-x_\star\|_{H_t}^2
        -
        \E[D_e(x_\star\|X_t)]
         & \leq
        \frac{\log2}{12}\tr(H_t\Sigma_t)
        +
        \frac{\lambda L}{2n\sqrt\pi}
        \|\mu_t-x_\star\|
        \|\mu_t-x_\star\|_{\Sigma_t^{-1}}
        \\
         & \leq
        \frac{\log2}{12}\tr(H_t\Sigma_t)
        +
        \frac{LD\log2}
        {48n\sqrt{\pi\log(2\sqrt2\,n)}},
        \label{eq:hermite-ons:focus-bracket-round}
    \end{align}
    where the last inequality uses $\|\mu_t-x_\star\|\leq D$ and
    \eqref{eq:hermite-ons:focus-condition-round}.

    Substituting
    \eqref{eq:hermite-ons:focus-rearrangement} and
    \eqref{eq:hermite-ons:focus-bracket-round} into
    \eqref{eq:hermite-ons:focus-projection} now gives
    \begin{align}
        \frac12
        \|\mu_{t+1}-x_\star\|_{\Sigma_{t+1}^{-1}}^2
        -
        \frac12
        \|\mu_t-x_\star\|_{\Sigma_t^{-1}}^2
        \leq
        \frac{\eta^2}{2}
        \|g_t\|_{\Sigma_{t+1}}^2
        +
        \eta\left(1+\frac{\log2}{12}\right)
        \tr(H_t\Sigma_t)
        +
        \frac{\eta LD\log2}
        {48n\sqrt{\pi\log(2\sqrt2\,n)}}.
        \label{eq:hermite-ons:focus-recursion}
    \end{align}

    We now prove that the focus condition cannot fail. First,
    \eqref{eq:hermite-ons:focus-sufficient} implies
    \eqref{eq:hermite-ons:focus-preserved} at $t=1$, because all the
    other terms on the left-hand side of
    \eqref{eq:hermite-ons:focus-sufficient} are nonnegative.
    Suppose, toward a contradiction, that
    \eqref{eq:hermite-ons:focus-preserved} fails at some later time.
    Let $\tau\in\{2,\ldots,n+1\}$ be the first time at which it fails.
    For every $t<\tau$, we then have
    \begin{align}
        \frac12
        \|\mu_t-x_\star\|_{\Sigma_t^{-1}}^2
        \leq
        \frac{(\log2)^2}
        {1152\lambda^2\log(2\sqrt2\,n)}.
    \end{align}
    Multiplying by $2$ and taking square roots shows that
    \begin{align}
        \lambda
        \|\mu_t-x_\star\|_{\Sigma_t^{-1}}
        \sqrt{\log(2\sqrt2\,n)}
        \leq
        \frac{\log2}{24}.
    \end{align}
    Therefore \eqref{eq:hermite-ons:focus-recursion} is valid for every
    $t<\tau$.
    Summing the recursion in \eqref{eq:hermite-ons:focus-recursion} from $t=1$ to $t=\tau-1$
    gives
    \begin{align}
        \frac12
        \|\mu_\tau-x_\star\|_{\Sigma_\tau^{-1}}^2
         & \leq
        \frac12
        \|\mu_1-x_\star\|_{\Sigma_1^{-1}}^2
        +
        \frac{\eta^2}{2}
        \sum_{t=1}^{\tau-1}
        \|g_t\|_{\Sigma_{t+1}}^2
        \nonumber \\
         & \quad+
        \eta\left(1+\frac{\log2}{12}\right)
        \sum_{t=1}^{\tau-1}
        \tr(H_t\Sigma_t)
        \nonumber \\
         & \quad+
        \frac{\eta LD\log2}
        {48\sqrt{\pi\log(2\sqrt2\,n)}}.
        \label{eq:hermite-ons:focus-summed}
    \end{align}

    It remains to bound the two sums. Since $H_t\succeq0$, the
    covariance update implies $\Sigma_{t+1}\preceq\Sigma_t\preceq\Sigma_1 = \sigma^2I_d$.
    Lipschitzness also gives $\|g_t\|\leq L$. Consequently,
    \begin{align}
        \sum_{t=1}^{\tau-1}
        \|g_t\|_{\Sigma_{t+1}}^2
        \leq
        nL^2\sigma^2.
        \label{eq:hermite-ons:focus-gradient-budget}
    \end{align}
    Applying \cref{prop:hermite-ons:trace-budget} to the first
    $\tau-1$ rounds gives
    \begin{align}
        \eta\sum_{t=1}^{\tau-1}
        \tr(H_t\Sigma_t)
         & \leq
        \frac{4d}{\lambda}
        \log\left(
        1+
        \frac{n\lambda\eta L\sigma}{\sqrt{2\pi}}
        \right).
        \label{eq:hermite-ons:focus-trace-budget}
    \end{align}
    Substituting these bounds into
    \eqref{eq:hermite-ons:focus-summed} yields
    \begin{align}
        \frac12
        \|\mu_\tau-x_\star\|_{\Sigma_\tau^{-1}}^2
         & \leq
        \frac12
        \|\mu_1-x_\star\|_{\Sigma_1^{-1}}^2
        +
        \frac{\eta^2nL^2\sigma^2}{2}
        \nonumber \\
         & \quad+
        \frac{
            4\left(1+\frac{\log2}{12}\right)d
        }{\lambda}
        \log\left(
        1+
        \frac{n\lambda\eta L\sigma}{\sqrt{2\pi}}
        \right)
        \nonumber \\
         & \quad+
        \frac{\eta LD\log2}
        {48\sqrt{\pi\log(2\sqrt2\,n)}}.
    \end{align}
    By \eqref{eq:hermite-ons:focus-sufficient}, the right-hand side is
    at most
    \[
        \frac{(\log2)^2}
        {1152\lambda^2\log(2\sqrt2\,n)}.
    \]
    This contradicts the definition of $\tau$. Therefore
    \eqref{eq:hermite-ons:focus-preserved} holds for every
    $t\leq n+1$.
    The bound \eqref{eq:hermite-ons:focus-preserved} also implies the
    focus condition of \cref{prop:hermite-ons:bracket} with
    $\delta=1/n$, while
    \eqref{eq:hermite-ons:focus-dimension} gives its dimension
    condition. Thus the bracket proposition is applicable at every
    round.
\end{proof}


\paragraph{Stationary regret.}

\begin{theorem}[Stationary population regret]
    \label{thm:hermite-ons:stationary}
    Let $d,n\geq1$. Let $\cK\subseteq\R^d$ be closed and convex with
    $\operatorname{diam}(\cK)\leq D$, where $D>0$. Let
    $e:\R^d\to\R$ be convex, twice continuously differentiable, and
    $L$-Lipschitz, where $L>0$, and fix
    \begin{align}
        x_\star\in\argmin_{x\in\cK}e(x).
    \end{align}
    Write
    \begin{align}
        \ell_n=\log(2\sqrt2\,n).
    \end{align}
    Run \cref{alg:HermiteOnlineNewtonStep} on the fixed loss $e$,
    starting from any $\mu_1\in\cK$, with
    \begin{align}
        \Sigma_1
         & =
        \frac{D^2}{d^2}I_d,
        \label{eq:hermite-ons:stationary-sigma}
        \\
        \lambda
         & =
        \frac{(\log2)^2}
        {288(12+\log2)d\ell_n^2},
        \label{eq:hermite-ons:stationary-lambda}
        \\
        \eta
         & =
        \frac{
            6(12+\log2)d^2\ell_n^{3/2}
        }{
            (\log2)LD\sqrt n
        }.
        \label{eq:hermite-ons:stationary-eta}
    \end{align}
    Then, for every $t\in\{1,\ldots,n+1\}$,
    \begin{align}
        \frac12
        \|\mu_t-x_\star\|_{\Sigma_t^{-1}}^2
        \leq
        72
        \left(
        \frac{12+\log2}{\log2}
        \right)^2
        d^2\ell_n^3.
        \label{eq:hermite-ons:stationary-focus}
    \end{align}
    Moreover,
    \begin{align}
        \E\left[
            \sum_{t=1}^n
            \bigl(e(X_t)-e(x_\star)\bigr)
            \right]
        \leq
        \left(
        \frac{132}{\log2}+12
        \right)
        LD\sqrt n\,\ell_n^{3/2}.
        \label{eq:hermite-ons:stationary-regret}
    \end{align}

    In particular, under the usual normalization
    \begin{align}
        B_1\subseteq\cK\subseteq B_{2d},
    \end{align}
    one may take $D=4d$, and hence
    \begin{align}
        \E\left[
            \sum_{t=1}^n
            \bigl(e(X_t)-e(x_\star)\bigr)
            \right]
        \leq
        \left(
        \frac{528}{\log2}+48
        \right)
        Ld\sqrt n\,\ell_n^{3/2}
        \label{eq:hermite-ons:stationary-regret-normalized}
    \end{align}
\end{theorem}

\begin{proof}
    We first verify the assumptions of
    \cref{prop:hermite-ons:focus}. Since $d\geq1$ and
    $\ell_n=\log(2\sqrt2\,n)>1$,
    \begin{align}
        2d\log2+4\log(2n)
        \leq
        6d\ell_n.
    \end{align}
    Therefore,
    \begin{align}
        \lambda
        \left(
        2d\log2+4\log(2n)
        \right)
        \leq
        6d\ell_n\lambda
         & =
        \frac{(\log2)^2}
        {48(12+\log2)\ell_n}
        \leq
        \frac{\log2}{12}.
    \end{align}
    The chosen value of $\lambda$ also clearly satisfies
    $0<\lambda\leq1/8$.

    With $\sigma=D/d$, the chosen parameters satisfy
    \begin{align}
        \lambda\eta L\sigma
        =
        \frac{\log2}{48\sqrt{n\ell_n}}.
        \label{eq:stationary-product}
    \end{align}
    Consequently,
    \begin{align}
        \lambda\eta L\sigma\sqrt{\frac{2}{\pi}}
        \leq1,
    \end{align}
    so the trace step-size condition holds.

    We will use the following bound on the logarithm appearing in the
    trace budget. By \eqref{eq:stationary-product},
    \begin{align}
        \frac{n\lambda\eta L\sigma}{\sqrt{2\pi}}
        =
        \frac{(\log2)\sqrt n}
        {48\sqrt{2\pi\ell_n}}
        \leq
        \frac{\sqrt n}{2}.
    \end{align}
    Since $n\geq1$,
    \begin{align}
        1+\frac{\sqrt n}{2}
        \leq
        \frac32\sqrt n
        \leq
        \sqrt{2\sqrt2\,n}.
    \end{align}
    Hence
    \begin{align}
        \log\left(
        1+
        \frac{n\lambda\eta L\sigma}{\sqrt{2\pi}}
        \right)
        \leq
        \frac{\ell_n}{2}.
        \label{eq:stationary-log-bound}
    \end{align}

    We next verify the sufficient condition
    \eqref{eq:hermite-ons:focus-sufficient}.
    Since
    $\|\mu_1-x_\star\|\leq D$,
    \begin{align}
        \frac12
        \|\mu_1-x_\star\|_{\Sigma_1^{-1}}^2
        \leq
        \frac{d^2}{2}.
        \label{eq:stationary-initial-budget}
    \end{align}
    The gradient contribution is
    \begin{align}
        \frac{\eta^2nL^2\sigma^2}{2}
        =
        18
        \left(
        \frac{12+\log2}{\log2}
        \right)^2
        d^2\ell_n^3.
        \label{eq:stationary-gradient-focus-budget}
    \end{align}
    Using \eqref{eq:stationary-log-bound}, the trace contribution is at
    most
    \begin{align}
         & \frac{
            4\left(1+\frac{\log2}{12}\right)d
        }{\lambda}
        \log\left(
        1+
        \frac{n\lambda\eta L\sigma}{\sqrt{2\pi}}
        \right)
        \leq
        48
        \left(
        \frac{12+\log2}{\log2}
        \right)^2
        d^2\ell_n^3.
        \label{eq:stationary-trace-focus-budget}
    \end{align}
    Finally,
    \begin{align}
        \frac{\eta LD\log2}
        {48\sqrt{\pi\ell_n}}
        =
        \frac{12+\log2}{8\sqrt\pi}
        \frac{d^2\ell_n}{\sqrt n}
        \leq
        d^2\ell_n^3.
        \label{eq:stationary-tail-focus-budget}
    \end{align}
    Together with \eqref{eq:stationary-initial-budget}, this gives
    \begin{align}
        \frac12
        \|\mu_1-x_\star\|_{\Sigma_1^{-1}}^2
        +
        \frac{\eta LD\log2}
        {48\sqrt{\pi\ell_n}}
        \leq
        \frac32d^2\ell_n^3.
    \end{align}

    On the other hand,
    \begin{align}
        \frac{(\log2)^2}
        {1152\lambda^2\ell_n}
        =
        72
        \left(
        \frac{12+\log2}{\log2}
        \right)^2
        d^2\ell_n^3.
        \label{eq:stationary-focus-radius}
    \end{align}
    The gradient and trace contributions sum to
    \begin{align}
        66
        \left(
        \frac{12+\log2}{\log2}
        \right)^2
        d^2\ell_n^3,
    \end{align}
    while the remaining margin satisfies
    \begin{align}
        6
        \left(
        \frac{12+\log2}{\log2}
        \right)^2
        d^2\ell_n^3
        \geq
        \frac32d^2\ell_n^3.
    \end{align}
    Thus \eqref{eq:hermite-ons:focus-sufficient} holds, and
    \cref{prop:hermite-ons:focus} proves
    \eqref{eq:hermite-ons:stationary-focus}.

    We now prove the regret bound. At every round, the preserved focus
    condition allows us to apply
    \cref{prop:hermite-ons:bracket} with
    $x=x_\star$ and $\delta=1/n$. Since
    $\|\mu_t-x_\star\|\leq D$, it gives
    \begin{align}
        \frac{\lambda}{2}
        \|\mu_t-x_\star\|_{H_t}^2
        -
        \E_{t-1}[D_e(x_\star\|X_t)]
        \leq
        \frac{\log2}{12}\tr(H_t\Sigma_t)
        +
        \frac{LD\log2}
        {48n\sqrt{\pi\ell_n}}.
        \label{eq:stationary-bracket-round}
    \end{align}
    Applying \cref{prop:hermite-ons:master}, substituting
    \eqref{eq:stationary-bracket-round}, and dropping the nonnegative
    final potential gives
    \begin{align}
        \E\left[
            \sum_{t=1}^n
            \bigl(e(X_t)-e(x_\star)\bigr)
            \right]
        \leq{} &
        \frac{1}{2\eta}
        \|\mu_1-x_\star\|_{\Sigma_1^{-1}}^2
        +
        \frac{\eta}{2}
        \sum_{t=1}^n
        \|g_t\|_{\Sigma_{t+1}}^2
        \nonumber  \\
               & +
        \left(
        1+\frac{\log2}{12}
        \right)
        \sum_{t=1}^n
        \tr(H_t\Sigma_t)
        +
        \frac{LD\log2}
        {48\sqrt{\pi\ell_n}}.
        \label{eq:stationary-regret-decomposition}
    \end{align}

    The initial term satisfies
    \begin{align}
        \frac{1}{2\eta}
        \|\mu_1-x_\star\|_{\Sigma_1^{-1}}^2
        \leq
        \frac{\log2}{12(12+\log2)}
        \frac{LD\sqrt n}{\ell_n^{3/2}}.
        \label{eq:stationary-initial-regret}
    \end{align}
    Since
    $\Sigma_{t+1}\preceq\Sigma_1=(D^2/d^2)I_d$
    and $\|g_t\|\leq L$,
    \begin{align}
        \frac{\eta}{2}
        \sum_{t=1}^n
        \|g_t\|_{\Sigma_{t+1}}^2
        \leq
        3
        \left(
        \frac{12+\log2}{\log2}
        \right)
        LD\sqrt n\,\ell_n^{3/2}.
        \label{eq:stationary-gradient-regret}
    \end{align}
    Finally, \cref{prop:hermite-ons:trace-budget},
    \eqref{eq:stationary-log-bound}, and
    \begin{align}
        \lambda\eta
        =
        \frac{(\log2)d}
        {48LD\sqrt{n\ell_n}}
    \end{align}
    give
    \begin{align}
        \left(
        1+\frac{\log2}{12}
        \right)
        \sum_{t=1}^n
        \tr(H_t\Sigma_t)
         & \leq
        \frac{
            4\left(1+\frac{\log2}{12}\right)d
        }{\lambda\eta}
        \frac{\ell_n}{2}
        =
        8
        \left(
        \frac{12+\log2}{\log2}
        \right)
        LD\sqrt n\,\ell_n^{3/2}.
        \label{eq:stationary-trace-regret}
    \end{align}
    The gradient and trace terms therefore sum to
    \begin{align}
        11
        \left(
        \frac{12+\log2}{\log2}
        \right)
        LD\sqrt n\,\ell_n^{3/2}.
    \end{align}
    Since $n\geq1$ and $\ell_n>1$,
    \begin{align}
         & \frac{\log2}{12(12+\log2)}
        \frac{LD\sqrt n}{\ell_n^{3/2}}
        +
        \frac{LD\log2}{48\sqrt{\pi\ell_n}}
        \leq
        LD\sqrt n\,\ell_n^{3/2}.
    \end{align}
    Substituting these estimates into
    \eqref{eq:stationary-regret-decomposition} gives
    \begin{align}
        \E\left[
            \sum_{t=1}^n
            \bigl(e(X_t)-e(x_\star)\bigr)
            \right]
         & \leq
        \left[
            11
            \left(
            \frac{12+\log2}{\log2}
            \right)
            +1
            \right]
        LD\sqrt n\,\ell_n^{3/2}
        \\
         & =
        \left(
        \frac{132}{\log2}+12
        \right)
        LD\sqrt n\,\ell_n^{3/2},
    \end{align}
    which proves \eqref{eq:hermite-ons:stationary-regret}.

    If $\cK\subseteq B_{2d}$, then
    $\operatorname{diam}(\cK)\leq4d$. Substituting $D=4d$ into
    \eqref{eq:hermite-ons:stationary-regret} proves
    \eqref{eq:hermite-ons:stationary-regret-normalized}.
\end{proof}

\section{Stochastic bandit convex optimization}
The loss $f$ is fixed,
but in round $t$ the learner plays $A_t\in\cK$ and observes a noisy loss
\[
    f(A_t)+\xi_t,
\]
where, conditionally on the past and on the Gaussian query $X_t$,
\begin{align}
    \E_{t-1}[\xi_t\mid X_t]=0,
    \qquad
    \E_{t-1}[\exp(\xi_t^2)\mid X_t]\leq2.
    \label{eq:direct-stoch-noise}
\end{align}
Throughout this section, $\mathcal F_t$ is the sigma-field generated by
the algorithm up to the end of round $t$, and $\E_t$ denotes conditional
expectation given $\mathcal F_t$.
We assume that the body has first been put in a position satisfying
\begin{align}
    B_1\subseteq\cK\subseteq B_{2d}.
    \label{eq:direct-stoch-rounded}
\end{align}
Let $\pi$ be the Minkowski functional of $\cK$ and set
\begin{align}
    M=\max\left\{
    \frac1{\sqrt d},
    \E_{U\sim\operatorname{Unif}(\mathbb S^{d-1})}[\pi(U)]
    \right\}.
    \label{eq:direct-stoch-M}
\end{align}
Fix a confidence level $\delta\in(0,1/2)$ and, throughout this section,
write
\begin{align}
    \ell=C_0\log\frac{C_0nd}{\delta},
    \qquad
    \beta=e^{-\ell},
    \label{eq:direct-stoch-log-factor}
\end{align}
where $C_0$ is a sufficiently large universal constant.  The number
$\delta$ is the confidence parameter.  The much smaller number $\beta$
will only be used as the deterministic tail parameter in
\cref{prop:hermite-ons:bracket}; keeping the two parameters separate
avoids paying a stochastic union bound in that proposition.

\subsection{The convex extension}
\begin{theorem}[Proposition 3.23 of Tor]
    Suppose that $f:\cK\to[0,1]$ is convex and $\mathbb{B}_r^d \subset \cK$.
    Let $\pi$ be the Minkowski functional of $\cK$,
    and $\epsilon \in (0, 1)$, and let $\pi_\epsilon(x) = \max\{1, \pi(x)/(1-\epsilon)\}$,
    and
    \begin{align}
        e(x)
        =\pi_\varepsilon(x)
        f\left(\frac{x}{\pi_\varepsilon(x)}\right)
        +\frac{1 - \varepsilon}{\varepsilon}
        \left(\pi_\varepsilon(x)-1\right).
        \label{eq:direct-stoch-extension}
    \end{align}
    Then the following properties hold:
    \begin{enumerate}
        \item $e$ is convex.
        \item $e(x) = f(x)$ for all $x \in (1-\varepsilon)\cK$.
        \item $\text{lip}(e) \leq \frac{2}{\varepsilon(1-\varepsilon)r}$.
        \item For all $x \notin (1-\varepsilon)\cK$, we have $e\left(\frac{x}{\pi_\varepsilon(x)}\right) \leq e(x)$.
    \end{enumerate}
\end{theorem}


The extension is generally not twice differentiable.  This causes no
problem, but it is worth recording precisely how second derivatives are
understood below.

\begin{lemma}[Gaussian derivatives of a nonsmooth convex function]
    \label{lem:direct-stoch-nonsmooth}
    Let $e:\R^d\to\R$ be convex and Lipschitz, let
    $X\sim\cN(\mu,\Sigma)$, and $Z=\Sigma^{-1/2}(X-\mu)$.  Define
    \begin{align}
        g
        =\E\left[e(X)\Sigma^{-1/2}Z\right],
        \qquad
        H
        =\E\left[e(X)\Sigma^{-1/2}
            (ZZ^\top-I_d)\Sigma^{-1/2}\right].
        \label{eq:direct-stoch-nonsmooth-H}
    \end{align}
    Then $g$ and $H$ are respectively the gradient and Hessian, with
    respect to $\mu$, of $\E[e(\mu+\Sigma^{1/2}Z)]$.  In particular,
    $H\succeq0$.
\end{lemma}

\begin{algorithm}[h!]
    \begin{minipage}{13cm}
        \begin{mdframed}
            \begin{lstlisting}
args: $\eta,\lambda,\sigma^2,\varepsilon$
$\mu_1=0$, $\Sigma_1=\sigma^2I_d$
for $t=1$ to $n$:
    Sample $X_t\sim\cN(0,I_d)$ and set $X_t=\mu_t+\Sigma_t^{1/2}Z_t$
    Play $A_t=X_t/\pi_\varepsilon(X_t)$
    Observe $f(A_t)+\xi_t$ and form
        $Y_t=\pi_\varepsilon(X_t)(f(A_t)+\xi_t)
             +2(\pi_\varepsilon(X_t)-1)/\varepsilon$
    Set $\widehat g_t=Y_t\Sigma_t^{-1/2}Z_t$
    Set $\widehat H_t
       =Y_t\Sigma_t^{-1/2}(Z_tZ_t^\top-I_d)\Sigma_t^{-1/2}$
    Set $\Sigma_{t+1}^{-1}=\Sigma_t^{-1}+\lambda\eta\widehat H_t$
    Set $\mu_{t+1}$ equal to the $\Sigma_{t+1}^{-1}$-metric
        projection of $\mu_t-\eta\Sigma_{t+1}\widehat g_t$
        onto $\cK_\varepsilon$
\end{lstlisting}
            \caption{Stochastic direct Hermite--ONS.}
            \label{alg:direct-stochastic-hermite-ons}
        \end{mdframed}
    \end{minipage}
\end{algorithm}

The matrix $\widehat H_t$ need not be positive semidefinite.  The
algorithm therefore checks that the proposed $\Sigma_{t+1}^{-1}$ is
positive definite before carrying out the projection.  If this check
fails, it stops updating and plays $0$ thereafter.  The proof below
shows that this safety rule is never invoked on the high-probability
event used in the analysis.

Conditionally on $\mathcal F_{t-1}$, define
\begin{align}
    g_t & =\E_{t-1}[\nabla e(X_t)],
        &
    H_t & =\E_{t-1}[\nabla^2e(X_t)].
    \label{eq:direct-stoch-population-gH}
\end{align}
For a nonsmooth $e$, these quantities are understood by Gaussian
mollification.

\begin{lemma}[Unbiased Hermite estimates]
    \label{lem:direct-stoch-unbiased}
    For every $t$,
    \begin{align}
        \E_{t-1}[\widehat g_t] & =g_t,
                               &
        \E_{t-1}[\widehat H_t] & =H_t.
        \label{eq:direct-stoch-unbiased}
    \end{align}
\end{lemma}

\begin{proof}
    By construction and \eqref{eq:direct-stoch-noise}, $\E_{t-1}[Y_t\mid X_t]=e(X_t)$.
    Gaussian integration by parts, first at order one and then at order
    two, gives
    \begin{align}
        \E_{t-1}\left[
            e(X_t)\Sigma_t^{-1/2}Z_t
            \right]
         & =\E_{t-1}[\nabla e(X_t)],
        \\
        \E_{t-1}\left[
            e(X_t)\Sigma_t^{-1/2}
            (Z_tZ_t^\top-I_d)\Sigma_t^{-1/2}
            \right]
         & =\E_{t-1}[\nabla^2e(X_t)].
    \end{align}
\end{proof}

It is useful to introduce the population and estimated damped
quadratics, both normalized to vanish at $\mu_t$:
\begin{align}
    q_t(x)
     & =\langle g_t,x-\mu_t\rangle
    +\frac\lambda2\|x-\mu_t\|_{H_t}^2,
    \\
    \widehat q_t(x)
     & =\langle\widehat g_t,x-\mu_t\rangle
    +\frac\lambda2\|x-\mu_t\|_{\widehat H_t}^2.
    \label{eq:direct-stoch-quadratics}
\end{align}
The preceding lemma says that $\E_{t-1}[\widehat q_t(x)]=q_t(x)$.

\subsection{The predictable covariance and the trace budget.}
The one-sample matrix $\widehat H_t$ is indefinite, so the random
covariance is not monotone.  We compare it with the predictable
covariance
\begin{align}
    \overline\Sigma_1^{-1} & =\Sigma_1^{-1},
                           &
    \overline\Sigma_{t+1}^{-1}
                           & =\overline\Sigma_t^{-1}+\lambda\eta H_t.
    \label{eq:direct-stoch-reference-covariance}
\end{align}
Because $H_t\succeq0$, the matrices $\overline\Sigma_t$ decrease
monotonically.

\begin{lemma}[Predictable trace budget]
    \label{lem:direct-stoch-trace}
    Let $m\leq n$.  Suppose that, for every $t\leq m$,
    \begin{align}
        \frac12\overline\Sigma_t^{-1}
        \preceq\Sigma_t^{-1}
        \preceq\frac32\overline\Sigma_t^{-1},
        \label{eq:direct-stoch-covariance-comparison}
    \end{align}
    and that $\Sigma_1=\sigma^2I_d$ and
    $\lambda\eta\operatorname{Lip}(e)\sigma\leq c$ for a sufficiently
    small universal constant $c$.  Then
    \begin{align}
        \eta\sum_{t=1}^m\tr(H_t\Sigma_t)
        \leq
        \frac{Cd}{\lambda}
        \log\left(
        1+Cm\lambda\eta\operatorname{Lip}(e)\sigma
        \right).
        \label{eq:direct-stoch-reference-trace}
    \end{align}
\end{lemma}

\begin{proof}
    The comparison assumption implies
    $\Sigma_t\preceq2\overline\Sigma_t$.  We first show that one
    predictable update is small in the normalized metric.  For a unit
    vector $u$, Gaussian integration by parts gives
    \begin{align}
         & u^\top\overline\Sigma_t^{1/2}H_t
        \overline\Sigma_t^{1/2}u
        \\
         & \quad=
        \E_{t-1}\left[
            \left\langle
            \nabla e(X_t),\overline\Sigma_t^{1/2}u
            \right\rangle
            \left\langle
            \Sigma_t^{-1}\overline\Sigma_t^{1/2}u,
            X_t-\mu_t
            \right\rangle
            \right]
        \\
         & \quad\leq
        \operatorname{Lip}(e)
        \|\overline\Sigma_t^{1/2}u\|
        \sqrt{\frac2\pi}
        \sqrt{
            u^\top\overline\Sigma_t^{1/2}
            \Sigma_t^{-1}\overline\Sigma_t^{1/2}u
        }
        \leq C\operatorname{Lip}(e)\sigma.
    \end{align}
    Hence every eigenvalue of
    $\lambda\eta\overline\Sigma_t^{1/2}H_t
        \overline\Sigma_t^{1/2}$ lies in $[0,1]$.  The inequality
    $\log(1+s)\geq s/2$ on this interval and
    $\Sigma_t\preceq2\overline\Sigma_t$ yield
    \begin{align}
        \eta\tr(H_t\Sigma_t)
        \leq
        \frac{4}{\lambda}
        \log\frac{\det(\overline\Sigma_{t+1}^{-1})}
        {\det(\overline\Sigma_t^{-1})}.
        \label{eq:direct-stoch-trace-one-step}
    \end{align}

    It remains to bound the terminal determinant.  For a unit vector
    $u$, a second integration by parts gives
    \begin{align}
        u^\top H_tu
        \leq
        \operatorname{Lip}(e)\sqrt{\frac2\pi}
        \sqrt{u^\top\Sigma_t^{-1}u}
        \leq
        C\operatorname{Lip}(e)
        \sqrt{\lambda_{\max}(\overline\Sigma_t^{-1})}.
    \end{align}
    Therefore, if
    $r_t=\sqrt{\lambda_{\max}(\overline\Sigma_t^{-1})}$, then
    \begin{align}
        r_{t+1}\leq r_t+C\lambda\eta\operatorname{Lip}(e).
    \end{align}
    Iteration from $r_1=1/\sigma$ gives
    \begin{align}
        \lambda_{\max}(\overline\Sigma_{m+1}^{-1})
        \leq
        \frac1{\sigma^2}
        \left(
        1+Cm\lambda\eta\operatorname{Lip}(e)\sigma
        \right)^2.
    \end{align}
    Summing \eqref{eq:direct-stoch-trace-one-step} and bounding the
    determinant by the largest eigenvalue proves
    \eqref{eq:direct-stoch-reference-trace}.
\end{proof}

\subparagraph{The extension pays for the feedback variance.}
This is the only estimate in which the particular form of the
extension matters.

\begin{lemma}[Second moment of the extended observation]
    \label{lem:direct-stoch-feedback}
    Let $\tau\leq n$ be a stopping time such that
    \eqref{eq:direct-stoch-covariance-comparison} holds for
    $t\leq\tau$, and suppose that
    \begin{align}
        \sigma M\sqrt{2d}\leq1,
        \qquad
        \lambda\eta\operatorname{Lip}(e)\sigma\leq c,
        \qquad
        \log\left(
        1+Cn\lambda\eta\operatorname{Lip}(e)\sigma
        \right)\leq\ell,
        \label{eq:direct-stoch-feedback-trace-conditions}
        \\
        \frac{d\ell^2}{\lambda\eta\varepsilon}\leq n,
        \qquad
        \frac{\ell^3}{\varepsilon^2}\leq n,
        \qquad
        \frac{\ell e^{-\ell}}{\varepsilon^2}\leq1.
        \label{eq:direct-stoch-feedback-conditions}
    \end{align}
    Then
    \begin{align}
        \sum_{t=1}^{\tau}\E_{t-1}[Y_t^2]\leq Cn.
        \label{eq:direct-stoch-predictable-Y2}
    \end{align}
    Moreover, with probability at least $1-\delta/8$,
    \begin{align}
        \max_{t\leq\tau}
        \bigl(|Y_t|+\E_{t-1}|Y_t|\bigr)
                               & \leq\frac{C\ell}{\varepsilon},
        \label{eq:direct-stoch-Ymax}
        \\
        \sum_{t=1}^{\tau}Y_t^2 & \leq Cn.
        \label{eq:direct-stoch-Y2}
    \end{align}
\end{lemma}

\begin{proof}
    Recall that $v=\pi_\varepsilon-1$.  We first prove the Gaussian
    convex-moment estimate that we need.  If
    $X\sim\cN(\mu,\Sigma)$, $\mu\in\cK_\varepsilon$, and
    $\|\Sigma\|_{\rm op}\leq2\sigma^2$, then
    \begin{align}
        \E[v(X)^2]
        \leq
        C\ell\left[
        \tr\left(
        \Sigma\E[\nabla^2v(X)]
        \right)+e^{-\ell}
        \right].
        \label{eq:direct-stoch-v-second-moment}
    \end{align}
    For completeness, convexity, $v(\mu)=0$, and Gaussian integration
    by parts first give
    \begin{align}
        \E[v(X)]
        \leq
        \tr\left(\Sigma\E[\nabla^2v(X)]\right).
    \end{align}
    The function $v$ is $2$-Lipschitz.  Gaussian concentration, the
    Minkowski mean bound
    $\E[\pi(X)]\leq1+M\sqrt{d\|\Sigma\|_{\rm op}}$, and
    $\sigma M\sqrt{2d}\leq1$ show that
    $\|v(X)\|_{\psi_1}\leq C$.  With $T=C\ell$,
    \begin{align}
        \E[v(X)^2]
         & \leq
        T\E[v(X)]
        +\E\left[v(X)^2\mathbbm1\{v(X)>T\}\right]
        \\
         & \leq
        C\ell\tr\left(\Sigma\E[\nabla^2v(X)]\right)
        +e^{-\ell}.
    \end{align}
    This proves \eqref{eq:direct-stoch-v-second-moment}.  Mollification
    justifies the distributional Hessian.

    On the stopped interval,
    $\Sigma_t\preceq2\overline\Sigma_t\preceq2\sigma^2I_d$, so the
    covariance hypothesis used above is satisfied.  Furthermore,
    \eqref{eq:direct-stoch-extension-curvature} gives
    \begin{align}
        \tr\left(
        \Sigma_t\E_{t-1}[\nabla^2v(X_t)]
        \right)
        \leq\varepsilon\tr(H_t\Sigma_t).
    \end{align}
    Since $f\in[0,1]$ and
    $\E_{t-1}[\xi_t^2\mid X_t]\leq1$, expansion of $Y_t$ now yields
    \begin{align}
        \E_{t-1}[Y_t^2]
        \leq
        C+\frac{C\ell}{\varepsilon}\tr(H_t\Sigma_t)
        +\frac{C\ell e^{-\ell}}{\varepsilon^2}.
        \label{eq:direct-stoch-Y-by-trace}
    \end{align}
    Summing, using \cref{lem:direct-stoch-trace}, and absorbing its
    logarithm into $\ell$ gives
    \begin{align}
        \sum_{t=1}^{\tau}\E_{t-1}[Y_t^2]
        \leq
        Cn+\frac{Cd\ell^2}{\lambda\eta\varepsilon}
        +\frac{Cn\ell e^{-\ell}}{\varepsilon^2}
        \leq Cn,
    \end{align}
    which proves \eqref{eq:direct-stoch-predictable-Y2}.

    Gaussian concentration for $\pi(X_t)$, the noise assumption, and
    a union bound show that, with probability at least $1-\delta/16$,
    \begin{align}
        \max_{t\leq\tau}\pi_\varepsilon(X_t)\leq C\sqrt\ell,
        \qquad
        \max_{t\leq\tau}|\xi_t|\leq C\sqrt\ell.
    \end{align}
    Moreover, the Minkowski mean bound and
    $\E_{t-1}[|\xi_t|\mid X_t]\leq1$ give
    \begin{align}
        \E_{t-1}|Y_t|\leq C/\varepsilon.
    \end{align}
    Hence these bounds and the definition of $Y_t$ imply
    \eqref{eq:direct-stoch-Ymax}.  Finally, apply the scalar Bernstein
    inequality to the centered, clipped variables obtained from
    $Y_t^2\mathbbm 1\{|Y_t|\leq C\ell/\varepsilon\}$.  On the preceding
    pointwise event the clipping changes none of the realized
    variables.  With probability
    at least $1-\delta/16$,
    \begin{align}
        \sum_{t=1}^{\tau}Y_t^2
        \leq
        2\sum_{t=1}^{\tau}\E_{t-1}[Y_t^2]
        +\frac{C\ell^3}{\varepsilon^2}
        \leq Cn.
    \end{align}
    This proves \eqref{eq:direct-stoch-Y2}.
\end{proof}

\subparagraph{Concentration of the two Hermite coefficients.}
The next result is a direct Freedman calculation.  It is stated for a
stopping time because later $\tau$ will be the first possible failure
of focus or covariance comparison.

\begin{proposition}[Stopped Hermite concentration]
    \label{prop:direct-stoch-concentration}
    Let $x\in\cK_\varepsilon$ be deterministic and let $\tau\leq n$ be
    a stopping time such that, for every $t\leq\tau$,
    \begin{align}
        \frac12\overline\Sigma_t^{-1}
        \preceq\Sigma_t^{-1}
        \preceq\frac32\overline\Sigma_t^{-1},
        \qquad
        \lambda\|\mu_t-x\|_{\Sigma_t^{-1}}\sqrt\ell\leq c.
        \label{eq:direct-stoch-concentration-assumptions}
    \end{align}
    Suppose that \cref{lem:direct-stoch-feedback} applies and that
    \begin{align}
        \varepsilon\geq\frac{\ell^{3/2}}{\sqrt n}.
        \label{eq:direct-stoch-q-increment-condition}
    \end{align}
    Finally, suppose that the following crude tail bound holds on the
    stopped path:
    \begin{align}
        \lambda^{-1}
        +\max_{t\leq\tau}
        \left(\E_{t-1}|Y_t|^8\right)^{1/8}
        +\max_{t\leq\tau}
        \left(
        \|\Sigma_t\|_{\rm op}
        +\|\Sigma_t^{-1}\|_{\rm op}
        \right)
        \leq e^{\ell/C}.
        \label{eq:direct-stoch-crude-tail}
    \end{align}
    This only requires the displayed quantities to be polynomial in
    $n,d,1/\delta$; it is verified by the final parameter choice.
    Then, with probability at least $1-\delta/4$, simultaneously,
    \begin{align}
        \left|
        \sum_{t=1}^{\tau}
        \bigl(\widehat q_t(x)-q_t(x)\bigr)
        \right|
         & \leq\frac{C\sqrt{n\ell}}{\lambda},
        \label{eq:direct-stoch-q-concentration}
        \\
        \sum_{t=1}^{\tau}
        \|\widehat g_t\|_{\Sigma_t}^2
         & \leq Cdn\ell,
        \label{eq:direct-stoch-gradient-energy}
    \end{align}
    and
    \begin{align}
        -C\ell^2\left(
        \sqrt{dn}+\frac{d^2\ell}{\varepsilon}
        \right)\overline\Sigma_{\tau+1}^{-1}
        \preceq
        \sum_{t=1}^{\tau}(\widehat H_t-H_t)
        \preceq
        C\ell^2\left(
        \sqrt{dn}+\frac{d^2\ell}{\varepsilon}
        \right)\overline\Sigma_{\tau+1}^{-1}.
        \label{eq:direct-stoch-H-path}
    \end{align}
\end{proposition}

\begin{proof}
    We first isolate the small point caused by the dependence between
    $Y_t$ and $Z_t$.  If $G_t$ is any conditionally standard Gaussian
    linear functional of $Z_t$, then, for $k=1,2$,
    \begin{align}
        \E_{t-1}\left[Y_t^2|G_t|^{2k}\right]
        \leq
        C\ell^k\E_{t-1}[Y_t^2]+e^{-\ell}.
        \label{eq:direct-stoch-weighted-gaussian-moment}
    \end{align}
    Indeed, split at $|G_t|=C\sqrt\ell$.  On the first event the
    bound is immediate.  On its complement it follows from
    Cauchy--Schwarz, \eqref{eq:direct-stoch-crude-tail}, and the
    Gaussian tail estimate, after increasing the universal constant
    in the definition of $\ell$.  Thus no independence between the
    response and the Gaussian score is used below.

    Put
    \begin{align}
        U_t
            & =\langle x-\mu_t,\Sigma_t^{-1/2}Z_t\rangle,
            &
        r_t & =\|x-\mu_t\|_{\Sigma_t^{-1}}.
    \end{align}
    Direct substitution gives
    \begin{align}
        \widehat q_t(x)
        =Y_t\left[
            U_t+\frac\lambda2(U_t^2-r_t^2)
            \right].
        \label{eq:direct-stoch-q-formula}
    \end{align}
    Conditionally on $\mathcal F_{t-1}$, $U_t/r_t$ is standard
    Gaussian.  The focus condition therefore gives, after truncating
    at $|U_t|\leq r_t\sqrt{C\ell}$,
    \begin{align}
        |\widehat q_t(x)| & \leq\frac{C|Y_t|}{\lambda},
        \\
        \E_{t-1}\left[
            (\widehat q_t(x)-q_t(x))^2
            \right]
                          & \leq
        \frac{C}{\lambda^2}
        \left(\E_{t-1}[Y_t^2]+e^{-\ell}\right).
    \end{align}
    The contribution of the truncated Gaussian tail is at most
    $e^{-\ell}$ by
    \eqref{eq:direct-stoch-weighted-gaussian-moment}.  Freedman's inequality,
    \cref{lem:direct-stoch-feedback}, and a union bound now give
    \begin{align}
        \left|
        \sum_{t=1}^{\tau}
        (\widehat q_t(x)-q_t(x))
        \right|
        \leq
        \frac{C}{\lambda}\left(
        \sqrt{n\ell}+\frac{\ell^2}{\varepsilon}
        \right).
    \end{align}
    Condition \eqref{eq:direct-stoch-q-increment-condition} absorbs
    the second term and proves
    \eqref{eq:direct-stoch-q-concentration}.

    The gradient estimate satisfies the exact identity
    \begin{align}
        \|\widehat g_t\|_{\Sigma_t}^2
        =Y_t^2\|Z_t\|^2.
    \end{align}
    Since $\max_{t\leq\tau}\|Z_t\|^2\leq Cd\ell$ with probability
    at least $1-\delta/16$, \eqref{eq:direct-stoch-gradient-energy}
    follows from \eqref{eq:direct-stoch-Y2}.

    It remains to control the Hessian path.  For any vector $u$,
    \begin{align}
        u^\top\widehat H_tu
        =Y_t\left(
        \langle\Sigma_t^{-1/2}u,Z_t\rangle^2
        -\|u\|_{\Sigma_t^{-1}}^2
        \right).
        \label{eq:direct-stoch-H-direction}
    \end{align}
    The parenthesized variable has conditional $\psi_1$ norm at most
    $C\|u\|_{\Sigma_t^{-1}}^2$.  In particular, if
    $\Delta_{t,u}=u^\top(\widehat H_t-H_t)u$, then
    \eqref{eq:direct-stoch-weighted-gaussian-moment} gives
    \begin{align}
        \E_{t-1}[\Delta_{t,u}^2]
        \leq
        C\ell^2\E_{t-1}[Y_t^2]
        \|u\|_{\Sigma_t^{-1}}^4
        +e^{-\ell}\|u\|^4.
        \label{eq:direct-stoch-H-direction-variance}
    \end{align}
    Applying scalar Freedman to
    \eqref{eq:direct-stoch-H-direction}, first on a sufficiently fine
    Euclidean net of the sphere and then extending by the usual
    quadratic-form argument.  Writing
    \begin{align}
        \alpha_\tau
        =C\ell^2\left(
        \sqrt{d\sum_{t=1}^{\tau}\E_{t-1}[Y_t^2]}
        +d^2\max_{t\leq\tau}
        (|Y_t|+\E_{t-1}|Y_t|)
        \right),
    \end{align}
    this gives, simultaneously for every positive definite matrix $A$ that
    dominates all $\Sigma_t^{-1}$ on the stopped path and satisfies
    $\kappa(A)\leq e^{C\ell}$,
    \begin{align}
        -\alpha_\tau A
        \preceq
        \sum_{t=1}^{\tau}(\widehat H_t-H_t)
        \preceq
        \alpha_\tau A.
        \label{eq:direct-stoch-H-Freedman}
    \end{align}
    Here is the uniformity detail.  By
    \eqref{eq:direct-stoch-crude-tail}, all relevant condition numbers
    are at most $e^{C\ell}$.  Take a Euclidean net with mesh
    $e^{-C\ell}$; its logarithmic cardinality is at most $Cd\ell$.
    For a fixed net point,
    \eqref{eq:direct-stoch-weighted-gaussian-moment} controls the
    conditional variance.  Truncating the standardized Gaussian
    multiplier at $C\sqrt{d\ell}$ controls the increments.  The two
    terms in Freedman's inequality are consequently bounded by
    \begin{align}
        C\ell^2
        \sqrt{d\sum_{t\leq\tau}\E_{t-1}[Y_t^2]}
        \quad\text{and}\quad
        Cd^2\ell^2
        \max_{t\leq\tau}
        (|Y_t|+\E_{t-1}|Y_t|),
    \end{align}
    respectively, in the metric $A$.  One first proves these bounds
    on dyadic spectral shells for deterministic $A$ and then unions
    over the $O(\ell)$ shells.  The quadratic-form extension from the
    fine net proves \eqref{eq:direct-stoch-H-Freedman} simultaneously
    for every admissible terminal $A$, including random ones.  On the stopped path,
    $\Sigma_t^{-1}\preceq\frac32
        \overline\Sigma_{\tau+1}^{-1}$.  Taking
    $A=\frac32\overline\Sigma_{\tau+1}^{-1}$ in
    \eqref{eq:direct-stoch-H-Freedman} and using
    \cref{lem:direct-stoch-feedback} proves
    \eqref{eq:direct-stoch-H-path}.
\end{proof}

\subparagraph{Noisy preservation of the focus region.}
We now combine the population bracket with the preceding concentration
result.  Let $x_\star$ minimize $f$ on $\cK_\varepsilon$.  By
\eqref{eq:direct-stoch-extension-properties-2}, $x_\star$ is also a
global minimizer of $e$.

\begin{proposition}[Noisy focus preservation]
    \label{prop:direct-stoch-focus}
    Let $b_0>0$ be a sufficiently small universal constant and set
    \begin{align}
        B=\frac{b_0}{\lambda^2\ell}.
        \label{eq:direct-stoch-focus-B}
    \end{align}
    Suppose that
    \begin{align}
         & 0<\lambda\leq\frac18,
        \qquad
        \lambda\left(
        2d\log2+4\log\frac2\beta
        \right)\leq\frac{\log2}{12},
        \nonumber                                        \\
         & \lambda\eta\operatorname{Lip}(e)\sigma\leq c,
        \label{eq:direct-stoch-focus-basic-conditions}
        \\
         & \sigma M\sqrt{2d}\leq1,
        \qquad
        \log\left(
        1+Cn\lambda\eta\operatorname{Lip}(e)\sigma
        \right)\leq\ell,
        \nonumber                                        \\
         & \frac{d\ell^2}{\lambda\eta\varepsilon}\leq n,
        \qquad
        \frac{\ell^3}{\varepsilon^2}\leq n,
        \nonumber                                        \\
         & \varepsilon\geq\frac{\ell^{3/2}}{\sqrt n},
        \qquad
        \frac{\ell e^{-\ell}}{\varepsilon^2}\leq1,
        \label{eq:direct-stoch-focus-moment-conditions}
        \\
         & C\lambda\eta\ell^2\left(
        \sqrt{dn}+\frac{d^2\ell}{\varepsilon}
        \right)\leq\frac14,
        \qquad
        n\operatorname{Lip}(e)d\beta\leq1,
        \label{eq:direct-stoch-focus-matrix-condition}
        \\
         & \max\left\{
        \sigma^{-1},\lambda^{-1},\varepsilon^{-1},
        n\lambda\eta\operatorname{Lip}(e)\sigma
        \right\}
        \leq e^{\ell/C}.
        \label{eq:direct-stoch-focus-crude-size}
    \end{align}
    Suppose, moreover, that
    \begin{align}
         & \frac12\|\mu_1-x_\star\|_{\Sigma_1^{-1}}^2
        +C\eta^2dn\ell
        +\frac{Cd\ell}{\lambda}
        +\frac{C\eta\sqrt{n\ell}}{\lambda}
        +C\eta
        \leq B.
        \label{eq:direct-stoch-focus-budget}
    \end{align}
    Then, with probability at least $1-\delta/2$,
    \begin{align}
        \frac12\|\mu_t-x_\star\|_{\Sigma_t^{-1}}^2\leq B
        \quad\text{and}\quad
        \frac12\overline\Sigma_t^{-1}
        \preceq\Sigma_t^{-1}
        \preceq\frac32\overline\Sigma_t^{-1}
        \label{eq:direct-stoch-focus-conclusion}
    \end{align}
    for every $t\leq n+1$.
    On the same event,
    \eqref{eq:direct-stoch-predictable-Y2},
    \eqref{eq:direct-stoch-Y2},
    \eqref{eq:direct-stoch-q-concentration}, and
    \eqref{eq:direct-stoch-gradient-energy} hold with $\tau=n$ and
    $x=x_\star$.
\end{proposition}

\begin{proof}
    As long as $\Sigma_{t+1}^{-1}$ is positive definite, the projection
    calculation gives
    \begin{align}
         & \frac12\|\mu_{t+1}-x_\star\|_{\Sigma_{t+1}^{-1}}^2
        -\frac12\|\mu_t-x_\star\|_{\Sigma_t^{-1}}^2
        \nonumber                                             \\
         & \quad\leq
        \eta\widehat q_t(x_\star)
        +\frac{\eta^2}{2}
        \|\widehat g_t\|_{\Sigma_{t+1}}^2.
        \label{eq:direct-stoch-noisy-projection}
    \end{align}
    If the potential is at most $B$, then, by choosing $b_0$ small
    enough,
    \begin{align}
        \lambda\|\mu_t-x_\star\|_{\Sigma_t^{-1}}
        \sqrt{\log(2\sqrt2/\beta)}
        \leq\frac{\log2}{24}.
    \end{align}
    The same choice of $b_0$ also gives
    $\lambda\|\mu_t-x_\star\|_{\Sigma_t^{-1}}\sqrt\ell\leq c$, which
    is the focus condition in
    \cref{prop:direct-stoch-concentration}.
    Thus the population regret-switching identity and
    \cref{prop:hermite-ons:bracket}, used with its deterministic
    parameter $\beta$, imply
    \begin{align}
        q_t(x_\star)
        \leq
        -\bigl(\E_{t-1}[e(X_t)]-e(x_\star)\bigr)
        +\left(1+\frac{\log2}{12}\right)
        \tr(H_t\Sigma_t)
        +b_t,
        \label{eq:direct-stoch-population-one-step}
    \end{align}
    where
    \begin{align}
        b_t=
        \frac{\lambda\operatorname{Lip}(e)\beta}{2\sqrt\pi}
        \|\mu_t-x_\star\|
        \|\mu_t-x_\star\|_{\Sigma_t^{-1}}.
        \label{eq:direct-stoch-bracket-error}
    \end{align}
    Since $\mu_t,x_\star\in\cK_\varepsilon$ and
    $\operatorname{diam}(\cK_\varepsilon)\leq4d$, focus gives
    \begin{align}
        \sum_{t=1}^{\tau}b_t
        \leq
        \frac{C n\operatorname{Lip}(e)d\beta}{\sqrt\ell}
        \leq C
        \label{eq:direct-stoch-total-bracket-error}
    \end{align}
    for every $\tau\leq n$.
    Therefore
    \begin{align}
         & \frac12\|\mu_{t+1}-x_\star\|_{\Sigma_{t+1}^{-1}}^2
        -\frac12\|\mu_t-x_\star\|_{\Sigma_t^{-1}}^2
        \nonumber                                             \\
         & \quad\leq
        -\eta\bigl(\E_{t-1}[e(X_t)]-e(x_\star)\bigr)
        +C\eta\tr(H_t\Sigma_t)
        +\eta b_t
        \nonumber                                             \\
         & \qquad+
        \eta(\widehat q_t(x_\star)-q_t(x_\star))
        +\frac{\eta^2}{2}
        \|\widehat g_t\|_{\Sigma_{t+1}}^2.
        \label{eq:direct-stoch-noisy-recursion}
    \end{align}
    Convexity and optimality of $x_\star$ show that the first term on
    the right is nonpositive.

    Define $\tau$ to be the first $t\leq n$ for which the proposed
    matrix $\Sigma_{t+1}^{-1}$ is not positive definite or, when it is
    positive definite, the proposed state violates either inequality
    in \eqref{eq:direct-stoch-focus-conclusion}; if there is no such
    round, set $\tau=n$.  Since the proposed state at time $t+1$ is
    $\mathcal F_t$-measurable, $\tau$ is a stopping time.  By its
    definition, both inequalities hold at the beginning of every round
    $t\leq\tau$.
    The largest-eigenvalue calculation in
    \cref{lem:direct-stoch-trace}, covariance comparison, and
    \eqref{eq:direct-stoch-focus-crude-size} imply the spectral part of
    \eqref{eq:direct-stoch-crude-tail}.  Gaussian Minkowski moments and
    the noise assumption also give
    $(\E_{t-1}|Y_t|^8)^{1/8}\leq C/\varepsilon$, so the remaining part
    follows from the same size condition.  Consequently
    \cref{prop:direct-stoch-concentration} is applicable to this
    stopped path.
    On the intersection of the events in
    \cref{lem:direct-stoch-feedback,prop:direct-stoch-concentration},
    the matrix condition
    \eqref{eq:direct-stoch-focus-matrix-condition}, the Hessian-path
    bound \eqref{eq:direct-stoch-H-path}, and
    \eqref{eq:direct-stoch-reference-covariance} show that
    \begin{align}
        \frac34\overline\Sigma_{\tau+1}^{-1}
        \preceq\Sigma_{\tau+1}^{-1}
        \preceq\frac54\overline\Sigma_{\tau+1}^{-1}.
    \end{align}
    Thus covariance comparison cannot be the first failure, and in
    particular the proposed inverse covariance is positive definite.
    Moreover,
    \begin{align}
        \Sigma_{t+1}
        \preceq2\overline\Sigma_{t+1}
        \preceq2\overline\Sigma_t
        \preceq3\Sigma_t
        \label{eq:direct-stoch-Sigma-next}
    \end{align}
    for $t\leq\tau$.  Hence
    \begin{align}
        \sum_{t=1}^\tau
        \|\widehat g_t\|_{\Sigma_{t+1}}^2
        \leq
        3\sum_{t=1}^\tau
        \|\widehat g_t\|_{\Sigma_t}^2
        \leq Cdn\ell.
    \end{align}

    Summing \eqref{eq:direct-stoch-noisy-recursion}, using
    \cref{lem:direct-stoch-trace},
    \eqref{eq:direct-stoch-total-bracket-error}, and
    \eqref{eq:direct-stoch-q-concentration} now gives
    \begin{align}
        \frac12\|\mu_{\tau+1}-x_\star\|_{
        \Sigma_{\tau+1}^{-1}}^2
         & \leq
        \frac12\|\mu_1-x_\star\|_{\Sigma_1^{-1}}^2
        +C\eta^2dn\ell
        +\frac{Cd\ell}{\lambda}
        \nonumber \\
         & \quad+
        \frac{C\eta\sqrt{n\ell}}{\lambda}
        +C\eta.
    \end{align}
    By \eqref{eq:direct-stoch-focus-budget}, this is at most $B$.
    Thus focus cannot be the first failure either.  The two conclusions
    follow by the first-failure argument, so $\tau=n$; the accompanying
    stopped concentration bounds therefore hold with $\tau=n$ as well.
\end{proof}

\subparagraph{Regret theorem.}

\begin{theorem}[Stochastic bandit regret without optimistic smoothing]
    \label{thm:direct-stoch-main}
    Suppose that \eqref{eq:direct-stoch-rounded} holds and that
    $f:\cK\to[0,1]$ is convex.  Suppose that the noise satisfies
    \eqref{eq:direct-stoch-noise}, and let $\ell$ be defined by
    \eqref{eq:direct-stoch-log-factor}.  There are universal
    constants $c_1,c_2,C_1>0$ such that
    \cref{alg:direct-stochastic-hermite-ons}, with
    \begin{align}
        \sigma
         & =\frac{1}{M\sqrt{2d}},
         &
        \lambda
         & =\frac{c_1}{d^{3/2}M\ell^2},
        \label{eq:direct-stoch-parameters-1}
        \\
        \eta
         & =\frac{c_2dM}{\sqrt n},
         &
        \varepsilon
         & =\frac{C_1d^2M\ell^4}{\sqrt n},
        \label{eq:direct-stoch-parameters-2}
    \end{align}
    satisfies, with probability at least $1-\delta$,
    \begin{align}
        \Reg_n
        =\sum_{t=1}^n
        \left(f(A_t)-\min_{x\in\cK}f(x)\right)
        \leq
        C d^2M\ell^4\sqrt n.
        \label{eq:direct-stoch-main-regret}
    \end{align}
    If the value of $\varepsilon$ in
    \eqref{eq:direct-stoch-parameters-2} exceeds $1/2$, the same bound
    follows by playing any fixed action, so the algorithm only needs to
    be run in the nontrivial regime $\varepsilon\leq1/2$.
\end{theorem}

\begin{proof}
    We first check the scale of the focus budget.  Since
    $\|x_\star\|\leq2d$ and
    $\Sigma_1=\sigma^2I_d$,
    \begin{align}
        \frac12\|x_\star\|_{\Sigma_1^{-1}}^2
        \leq Cd^3M^2.
        \label{eq:direct-stoch-main-initial}
    \end{align}
    The right-hand side of \eqref{eq:direct-stoch-focus-B} is, up to a
    universal constant,
    \begin{align}
        \frac1{\lambda^2\ell}
        =d^3M^2\ell^3.
        \label{eq:direct-stoch-main-focus-size}
    \end{align}
    The remaining terms in
    \eqref{eq:direct-stoch-focus-budget}, after substituting
    \eqref{eq:direct-stoch-parameters-1}--%
    \eqref{eq:direct-stoch-parameters-2}, are bounded respectively by
    \begin{align}
        \eta^2dn\ell
             & \leq C d^3M^2\ell,
        \\
        \frac{d\ell}{\lambda}
             & \leq C d^{5/2}M\ell^3,
        \\
        \frac{\eta\sqrt{n\ell}}{\lambda}
             & \leq C d^{5/2}M^2\ell^{5/2},
        \\
        \eta & \leq C dM/\sqrt n.
    \end{align}
    Since $M\geq d^{-1/2}$, all these terms are at most a universal
    constant times $d^3M^2\ell^3$.  Choosing $c_1,c_2$ sufficiently
    small and $C_1$ sufficiently large verifies
    \eqref{eq:direct-stoch-focus-budget}.

    We next check the remaining numerical conditions rather than hide
    them in the concentration argument.  The parameter choices give
    \begin{align}
        \lambda\eta
         & =\frac{c_1c_2}{\sqrt{dn}\,\ell^2},
        \label{eq:direct-stoch-lambda-eta}
        \\
        \frac{d\ell^2}{\lambda\eta\varepsilon}
         & \leq\frac{n}{\sqrt d\,M}\leq n,
        \\
        \frac{\ell^3}{\varepsilon^2}
         & \leq n,
        \qquad
        \varepsilon\geq\frac{\ell^{3/2}}{\sqrt n}.
        \label{eq:direct-stoch-moment-checks}
    \end{align}
    Here we used $\sqrt d\,M\geq1$ and enlarged $C_1$ where needed.
    Also,
    \begin{align}
        \lambda\eta\ell^2\sqrt{dn}
         & =c_1c_2,
        \\
        \frac{\lambda\eta d^2\ell^3}{\varepsilon}
         & \leq\frac{C c_1c_2}{\sqrt d\,M\ell^3}.
        \label{eq:direct-stoch-matrix-checks}
    \end{align}
    Thus the matrix condition follows by taking $c_1c_2$ small.
    The bounds $\operatorname{Lip}(e)\leq C/\varepsilon$ and
    $\sigma=1/(M\sqrt{2d})$ verify the normalized trace step condition.
    They also make
    $1+Cn\lambda\eta\operatorname{Lip}(e)\sigma$ polynomial in
    $n,d,1/\delta$, so its logarithm is at most $\ell$; similarly,
    $\ell e^{-\ell}/\varepsilon^2\leq1$ after increasing $C_0$.
    On the stopped path, the largest-eigenvalue calculation in
    \cref{lem:direct-stoch-trace} and covariance comparison show that
    \begin{align}
        \|\Sigma_t\|_{\rm op} & \leq2\sigma^2,
        \\
        \|\Sigma_t^{-1}\|_{\rm op}
                              & \leq
        \frac{3}{2\sigma^2}
        \left(
        1+Cn\lambda\eta\operatorname{Lip}(e)\sigma
        \right)^2.
    \end{align}
    Furthermore, Gaussian Minkowski moments, the condition
    $\sigma M\sqrt{2d}\leq1$, and the noise assumption give
    \begin{align}
        \left(\E_{t-1}|Y_t|^8\right)^{1/8}
        \leq C/\varepsilon.
    \end{align}
    These quantities, $\lambda^{-1}$, $\varepsilon^{-1}$, and
    $\sigma^{-1}$ are polynomial in $n,d,1/\delta$.  Increasing $C_0$
    therefore verifies \eqref{eq:direct-stoch-crude-tail}.
    Finally, the bracket dimension condition follows from
    $d^{3/2}M\geq d$ (and also implies $\lambda\leq1/8$), while
    $n\operatorname{Lip}(e)d\beta\leq1$ follows from
    $\beta=e^{-\ell}$ after increasing $C_0$.  Hence all assumptions
    of \cref{prop:direct-stoch-focus} apply.

    We may consequently sum
    \eqref{eq:direct-stoch-noisy-recursion} through round $n$, retain
    the expected-regret term, and drop the nonnegative final potential.
    This gives
    \begin{align}
         & \sum_{t=1}^n
        \left(\E_{t-1}[e(X_t)]-e(x_\star)\right)
        \nonumber       \\
         & \quad\leq
        \frac{\|x_\star\|_{\Sigma_1^{-1}}^2}{2\eta}
        +C\sum_{t=1}^n\tr(H_t\Sigma_t)
        +\frac\eta2\sum_{t=1}^n
        \|\widehat g_t\|_{\Sigma_{t+1}}^2
        \nonumber       \\
         & \qquad+
        \left|\sum_{t=1}^n
        (\widehat q_t(x_\star)-q_t(x_\star))
        \right|
        +C.
        \label{eq:direct-stoch-main-decomposition}
    \end{align}
    The bounds proved above yield
    \begin{align}
        \frac{\|x_\star\|_{\Sigma_1^{-1}}^2}{2\eta}
         & \leq C d^2M\sqrt n,
        \\
        \sum_{t=1}^n\tr(H_t\Sigma_t)
         & \leq C d^{3/2}\ell^3\sqrt n,
        \\
        \eta\sum_{t=1}^n
        \|\widehat g_t\|_{\Sigma_{t+1}}^2
         & \leq C d^2M\ell\sqrt n,
        \\
        \left|\sum_{t=1}^n
        (\widehat q_t(x_\star)-q_t(x_\star))
        \right|
         & \leq C d^{3/2}M\ell^{5/2}\sqrt n.
    \end{align}
    Since $M\geq d^{-1/2}$, the right-hand side of
    \eqref{eq:direct-stoch-main-decomposition} is at most
    $Cd^2M\ell^3\sqrt n$.

    It remains only to return from the extension to the original loss.
    Let $x_0$ minimize $f$ over $\cK$.  Convexity, $f\in[0,1]$, and
    $0\in\cK$ imply
    \begin{align}
        f(x_\star)
        \leq f((1-\varepsilon)x_0)
        \leq f(x_0)+\varepsilon.
        \label{eq:direct-stoch-shrink}
    \end{align}
    Moreover,
    \begin{align}
        f(A_t)=e(A_t)\leq e(X_t).
    \end{align}
    Since $f(A_t)\in[0,1]$, Azuma's inequality gives
    \begin{align}
        \sum_{t=1}^n
        \left(
        f(A_t)-\E_{t-1}[f(A_t)]
        \right)
        \leq C\sqrt{n\ell}
        \label{eq:direct-stoch-action-concentration}
    \end{align}
    with probability at least $1-\delta/2$.  Intersecting this event
    with the event of \cref{prop:direct-stoch-focus} and combining the
    last three displays,
    \begin{align}
        \Reg_n
         & \leq
        n\varepsilon+C\sqrt{n\ell}
        +\sum_{t=1}^n
        \left(\E_{t-1}[e(X_t)]-e(x_\star)\right)
        \\
         & \leq C d^2M\ell^4\sqrt n,
    \end{align}
    which proves \eqref{eq:direct-stoch-main-regret}.
\end{proof}

\begin{corollary}[L\"owner position]
    If $\cK/d$ is in L\"owner position, then
    $M=\widetilde O(d^{-1/2})$, and hence
    \begin{align}
        \Reg_n=\widetilde O(d^{3/2}\sqrt n).
    \end{align}
\end{corollary}

The algorithm and proof above contain neither the optimistic Gaussian
surrogate nor its density ratio.  The only surrogate used by the
algorithm is the ordinary degree-two Hermite projection of the
queryable extension, estimated directly from the same noisy function
value that determines the incurred loss.
