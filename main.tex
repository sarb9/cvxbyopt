\section{Overview}
\begin{enumerate}
    \item \oco reduces to \olo: $f(x) - f(x_\star) \leq \langle \nabla f(x), x - x_\star \rangle$.
    \item \textbf{Random} algs are \textit{suboptimal} for \texttt{OCO} and \textit{pointless} for \texttt{OLO}.
    \item First order information $\nabla f(x)$ makes \texttt{BCO} as hard as \texttt{OLO}.
    \item \textbf{Random} algs are \textit{necessary} for \texttt{BCO} and \texttt{BLO}.
    \item \textbf{Random} algs are \textit{free} for \texttt{BLO} but appear to be \textit{expensive} ($d^{1.5}$) for \texttt{BCO}.
\end{enumerate}
We study algs for \texttt{BCO} and \texttt{BLO} which are based on Stochastic Mirror Decent(\textsf{SMD}) which means choosing\\
(1) randomization scheme, (2) estimation scheme, (3) potential + $\eta$.

$\star$ Duality with the Bayesian world is curious and helps with the analysis.
\subsection{Notation}
Let $[n] = \{1,2, \cdots, n\}$.
Let $D_F(a, b) = F(a) - F(b) - \nabla F(b)^\top (a - b)$ be the Bregman divergence.
Also let $\cP(A)$ be the set of all probability measures over the set $A$,
equipped with appropriate $\sigma$-algebra.

\section{Online linear optimization(\texttt{OLO})}
The adversary chooses a sequence of losses $\ell_1, \cdots, \ell_n \in \cL \subseteq \Rd$.
In each round $t \in [n]$, the player chooses an action $a_t \in \cA \subseteq \Rd$, and observes $y_t = \langle a_t, \ell_t \rangle$.
The regret w.r.t to an action $a_\star \in \cA$ is
\begin{align*}
    \Reg_n(a_\star) = \sum_{t=1}^{n} \langle \ell_t, a_t - a_\star \rangle\,.
\end{align*}

\subsection{Mirror decent(\mdalg)}
For a sequence of losses $\ell_{1:t-1}$, we define the functions $\fmd(\ell_{1:t-1})$
and $\tfmd(\ell_{1:t-1})$ to be the actions $a_t$ and $\ta_t$
in \cref{alg:MD}, when the loss sequence $\ell_{1:t-1}$ is fed to the algorithm.

We also overload this notation as $\fmd(\ell; a) = \argmin_{b \in \cA} \eta\langle b, \ell \rangle + D_F(b, a)$,
and $\tfmd(\ell; a) = \argmin_{b \in \Rd} \eta\langle b, \ell \rangle + D_F(b, a)$.
\begin{algorithm}[h!]
    \begin{minipage}{12cm}
        \begin{mdframed}
            \begin{lstlisting}
args: $\eta \in \R$, $F$
$a_1 = \argmin_{a \in \cA} F(a)$
$\ta_1 = \argmin_{a \in \Rd} F(a)$
for $t = 1$ to $n$:
    Play $a_t$ and observe $\ell_t$
    $a_{t+1} = \argmin_{a \in \cA} \eta \langle a, \ell_t\rangle + D_F(a, a_{t+1})$
    $\ta_{t+1} = \argmin_{a \in \Rd} \eta \langle a, \ell_t\rangle + D_F(a, a_{t+1})$
\end{lstlisting}
            \caption{\mdalg}\label{alg:MD}
        \end{mdframed}
    \end{minipage}
\end{algorithm}
\begin{theorem}\label{thm:MD}
    Let $F$ be Legendre with domain $\cD$ such that $\cA \cap \interior(\cD)\neq \emptyset$.
    Then
    \begin{align*}
        \Reg_n(a_\star)
         & \leq
        \sum_{t=1}^{n} \langle \ell_t, a_t - a_{t+1} \rangle
        + \frac{1}{\eta}
        \left(
        D_F(a_\star, a_1) -
        \sum_{t=1}^{n} D_F(a_{t+1}, a_t)
        \right) \\
         & \leq
        \frac{1}{\eta}
        \left(
        D_F(a_\star, a_1) + \sum_{t=1}^{n} D_F(a_{t}, \tilde{a}_{t+1})
        \right) \\
         & =
        \frac{1}{\eta}
        \left(
        D_{F}(a_\star, a_1) + \sum_{t=1}^{n} D_{F^\star}\left(\nabla F(a_t) - \eta \ell_t, \nabla F(a_t)\right)
        \right)\,.
    \end{align*}
\end{theorem}
\begin{proof}
    We have
    \begin{align*}
        \Reg_n(a_\star)
         & = \sum_{t=1}^{n} \langle \ell_t, a_t - a_\star \rangle                                                                                                                                                                         \\
         & = \sum_{t=1}^{n} \underbrace{\langle \ell_t, a_t - a_{t+1} \rangle}_{\triangleq \texttt{st}_t \text{ or stability}} + \underbrace{\langle \ell_t, a_{t+1} - a_\star \rangle}_{\triangleq \texttt{tr}_t \text{ or tracking}}\,.
    \end{align*}
    What do we know about $a_{t+1}$?
    \begin{align*}
         & a_{t+1} = \argmin_{a \in \cA} \eta \langle \ell_t, a \rangle + D_F(a, a_t)                                                                                                          \\
        \Rightarrow
         & \langle \eta \ell_t + \nabla F(a_{t+1}) - \nabla F(a_t), \tilde{a} - a_{t+1} \rangle \geq 0                                                           & (\forall \tilde{a} \in \cA) \\
        \Rightarrow
         & \texttt{tr}_t = \langle \ell_t, a_{t+1} - a_\star \rangle \leq \frac{1}{\eta} \langle \nabla F(a_{t+1}) - \nabla F(a_t), a_\star - a_{t+1} \rangle\,.
    \end{align*}
    Then there's the nice equality
    \begin{align*}
        \langle \nabla F(a_{t+1}) - \nabla F(a_t), a_\star - a_{t+1} \rangle
         & =
        D_F(a_\star, a_t) - D_F(a_{t+1}, a_t) - D_F(a_\star, a_{t+1})\,,
    \end{align*}
    which gives us a bound on the \textit{tracking} terms
    \begin{align*}
        \eta \sum_{t=1}^{n} \langle \ell_t, a_{t+1} - a_\star \rangle
         & \leq \sum_{t=1}^{n} D_F(a_\star, a_t)  - D_F(a_\star, a_{t+1}) - D_F(a_{t+1}, a_t)                   \\
         & = D_F(a_\star, a_1) - \underbrace{D_F(a_\star, a_{n+1})}_{\geq 0} - \sum_{t=1}^{n} D_F(a_{t+1}, a_t) \\
         & = D_F(a_\star, a_1) - \sum_{t=1}^{n} D_F(a_{t+1}, a_t)\,.
    \end{align*}
    Note that the tracking term scales inversely with $\eta$ as expected.
    Without touching the stability term, we have proved the first inequality,
    \begin{align*}
        \Reg_n(a_\star)
         & \leq
        \sum_{t=1}^{n} \langle \ell_t, a_t - a_{t+1} \rangle
        + \frac{1}{\eta}
        \left(
        D_F(a_\star, a_1) -
        \sum_{t=1}^{n} D_F(a_{t+1}, a_t)
        \right)\,.
    \end{align*}
    Next, we move onto the stability term.
    $\textsf{MD}$ stores the latest loss in the gradient of $\tilde{a}_{t+1}$,
    \begin{align*}
         & \tilde{a}_{t+1}  = \argmin_{a \in \R^d} \eta \langle \ell_t, a \rangle + D_F(a, a_t) \\
        \Rightarrow
         & \ell_t = \frac{1}{\eta}
        \left(
        \nabla F(a_t) - \nabla F(\tilde{a}_{t+1})
        \right)
    \end{align*}
    which results in another nice equality:
    \begin{align*}
        \texttt{st}_t
         & = \langle \ell_t, a_{t} - a_{t+1} \rangle                                                 \\
         & = \frac{1}{\eta} \langle \nabla F(a_t) - \nabla F(\tilde{a}_{t+1}), a_t - a_{t+1} \rangle \\
         & = \frac{1}{\eta} \left(D_F(a_{t+1}, a_t) + D_F(a_t, \tilde{a}_{t+1})\right)
    \end{align*}
    and proves the second inequality,
    \begin{align*}
        \Reg_n(a_\star)
         & \leq
        \frac{1}{\eta}
        \left(
        D_F(a_\star, a_1) + \sum_{t=1}^{n} D_F(a_{t}, \tilde{a}_{t+1})
        \right)\,.
    \end{align*}
    To see the third part, note that
    \begin{align*}
        D_F(a_{t}, \tilde{a}_{t+1})
         & =D_{F^\star}\left(\nabla F(\tilde{a}_{t+1}), \nabla F(a_t)\right)       \\
         & = D_{F^\star}\left(\nabla F(a_t) - \eta \ell_t, \nabla F(a_t)\right)\,.
    \end{align*}
\end{proof}
\begin{remark}
    \label{rem:ellt-adapted}
    The analysis is remarkably amenable to different modifications.
    For instance, the result continues to work even if $\ell_t$ is adapted to $a_1, \cdots, a_{t-1}$---equivalently, $\ell_t$ is chosen adaptively by an adversary.
    Much of what comes after hinges on the generality of the proof of this theorem.
\end{remark}

\subsubsection{When $\cA=\Delta(d)$ and $F$ is the negentropy}\label{sec:md-negentropy}
The case that $\cA = \Delta(d)$ is the simplex and the potential
is negentropy,
\begin{align*}
    R(p) = \begin{cases}
               \langle p, \ln(p) - 1 \rangle & p \in \Delta(d)         \\
               \infty                        & p \in \text{otherwise,} \\
           \end{cases}
\end{align*}
is of special interest as it results in a sort of universal algorithm for all
action sets as long as they have a covering---the learner
can come up with the cover $\cA$ and randomize over that.
That being said, while being theoretically interesting, such an algorithm
would be computationally expensive as the size of the cover grows.

Since $\cA = \Delta(d)$ in this section, instead of denoting actions by `$a$' we use `$p$' to denote a probability measures over $d$ atoms.
Note that $D_R = D_{KL}$, $R^\star(q) = \langle \exp(q), 1\rangle$,
$R'(p) = \ln(p)$, and ${R^\star}'(q) = \exp(q)$.
\begin{theorem}[\mdalg on simplex]\label{thm:md-simplex}
    Suppose that $\eta \|\ell_t\|_\infty \leq 1$ for all $t \in [n]$,
    then the regret of \mdalg with $R$ as the potential,
    when $\cA = \Delta(d)$ is bounded as
    \begin{align*}
        \Reg_n(a_\star) \leq
        \frac{\ln(d)}{\eta}
        +
        \eta\sum_{t=1}^{n} \langle a_t, \ell_t^2\rangle\,.
    \end{align*}
\end{theorem}
\begin{proof}
    From theorem~\ref{thm:MD}, we have
    \begin{align*}
        \Reg_n(a_\star)
         & \leq
        \frac{1}{\eta}
        \left(
        D_R(a_\star, a_1) + \sum_{t=1}^{n} D_{R^\star}\left(R'(a_t) - \eta \ell_t, R'(a_t)\right)
        \right)
        \\
         & \leq
        \frac{1}{\eta}
        \left(
        \ln(d) + \sum_{t=1}^{n} R^\star(R'(a_t) - \eta \ell_t) - R^\star(R'(a_t)) - a_t^\top(-\eta \ell_t)
        \right)
        \\
         & =
        \frac{1}{\eta}
        \left(
        \ln(d) + \sum_{t=1}^{n} \langle \exp(\ln(a_t) - \eta \ell_t) - \exp(\ln(a_t)), 1 \rangle + \langle a_t, \eta \ell_t\rangle
        \right)
        \\
         & =
        \frac{1}{\eta}
        \left(
        \ln(d) + \sum_{t=1}^{n} \langle a_t, \exp(- \eta \ell_t) - 1 \rangle + \langle a_t, \eta \ell_t\rangle
        \right)
        \\
         &
        \abovelabel{$\eta \|\ell_t\|_\infty \leq 1$}
        \leq
        \frac{1}{\eta}
        \left(
        \ln(d) + \sum_{t=1}^{n} \langle a_t, 1 - \eta \ell_t + \eta^2 \ell_t^2 - 1 \rangle + \langle a_t, \eta \ell_t\rangle
        \right)
        \\
         &
        =
        \frac{1}{\eta}
        \left(
        \ln(d) + \eta^2\sum_{t=1}^{n} \langle a_t, \ell_t^2\rangle
        \right)\,.
    \end{align*}
\end{proof}
\begin{remark}[An explicit bound]\label{rem:md-simplex-explicit}
    In order to get a bound in terms of $d$ and $n$,
    we can further suppose that $\ell_t \in [0,1]^d$
    and then from theorem~\ref{thm:md-simplex} we have
    \begin{align*}
        \Reg_n(a_\star) & \leq
        \frac{\ln(d)}{\eta}
        +
        \eta\sum_{t=1}^{n} \langle a_t, \ell_t^2\rangle \\
                        & \leq
        \frac{\ln(d)}{\eta}
        +
        \eta \sum_{t=1}^{n} \|a_t\| \|\ell_t^2\|_\infty
        \\
                        & \leq
        \frac{\ln(d)}{\eta}
        +
        \eta n
        \leq 2 \sqrt{n\ln(d)}\,.
    \end{align*}
    The better bound of $\sqrt{2n\ln(d)}$ is also possible by using
    $\exp(-x) \leq 1 - x + \frac{x^2}{2}$.
\end{remark}

\section{Bandit linear optimization(\blo)}
In the bandit setting, the learner only observes $y_t = \langle \ell_t, a_t \rangle$, instead of $\ell_t$.

\paragraph{The need for randomization:}
In \blo, if the learner plays a deterministic action $a_t$ in round $t$,
then an adversary can choose $\ell_t \perp a_t$ such that
\begin{align*}
    \langle \ell_t, a_t \rangle = 0\,,
\end{align*}
while $\langle \ell_t, a_\star \rangle$ can be large for some $a_\star \in \cA$.

\subsection{Stochastic mirror decent(\smdalg)}
\begin{algorithm}[h!]
    \begin{minipage}{12cm}
        \begin{mdframed}
            \begin{lstlisting}
args: $\eta \in \R$, $F$
$a_1 = \argmin_{a \in \cA} F(a)$
$\ta_1 = \argmin_{a \in \Rd} F(a)$
for $t = 1$ to $n$:
    Choose $P_t \in \Delta(\cA)$, s.t. $\E_{A \sim P_t}[A] = a_t$
    Play $A_t \sim P_t$ and observe $Y_t=\langle A_t, \ell_t \rangle$
    Compute $\hl_t = E_t(A_t, Y_t)$
    $a_{t+1} = \argmin_{a \in \cA} \eta \langle a, \hl_t\rangle + F(a)$
    $\ta_{t+1} = \argmin_{a \in \Rd} \eta \langle a, \hl_t\rangle + F(a)$
\end{lstlisting}
            \caption{\smdalg}\label{alg:SMD}
        \end{mdframed}
    \end{minipage}
\end{algorithm}

\marginnote{\textit{Exploration is free in \blo}}
Instead of choosing an action $a_t \in \cA$, \smdalg chooses an
\textit{exploration} distribution $P_t \in \cP(\cA)$
\textbf{with mean} $a_t$ in round $t \in [n]$.
Since losses are linear in {\blo}, we have that
\begin{align*}
    \E_t[\langle A_t, \ell \rangle] = \langle a_t, \ell\rangle\,,
\end{align*}
which means \smdalg and \mdalg have the \textbf{same expected loss} given
the \mdalg action $a_t$.

\marginnote{\textit{$\hl$ vs $\ell$}}
The issue is that \smdalg does not observe $\ell_t$.
Instead, \smdalg pretends that $\hl_t$ which is an \textit{estimate} of $\ell_t$ is the actual loss.
\textbf{In the most general form}, this estimation can be made through a function
$E_t: \cA \times \R \to \Rd$ in each round $t \in [n]$ such that $\hl_t = E_t(A_t, Y_t)$.
\begin{theorem}[Unbiased-\smdalg]\label{thm:SMD-unbiased}
    Suppose $\E[\hl_t|\fmd(\hl_{1:t-1})] = \ell_t$ (unbiased estimator).
    The regret of $\smdalg$ is bounded as
    \begin{align*}
        \Reg_n(a_\star)
         & \leq
        \frac{D_F(a_\star, a_1)}{\eta}
        +
        \E\left[
            \sum_{t=1}^{n}
            \left(
            \langle \hl_t, \fmd(\hl_{1:t-1}) - \fmd(\hl_{1:t}) \rangle
            - \frac{1}{\eta}
            D_F(\fmd(\hl_{1:t-1}), \fmd(\hl_{1:t}))
            \right)
        \right] \\
         & \leq
        \frac{1}{\eta}\left(
        D_F(a_\star, a_1)
        +
        \E\left[
            \sum_{t=1}^{n} D_F(\fmd(\hl_{1:t-1}), \tfmd(\hl_{1:t}))
            \right]
        \right)\,.
    \end{align*}
\end{theorem}
\begin{proof}
    The proof follows simply from
    \begin{align*}
        \Reg_n(a_\star)
        =
        \E\left[
            \sum_{t=1}^{n} \langle A_t - a_\star, \ell_t\rangle
            \right]
        =
        \E\left[
            \sum_{t=1}^{n} \langle \fmd(\hl_{1:t-1}) - a_\star, \hl_t\rangle
            \right]\,,
    \end{align*}
    and applying theorem~\ref{thm:MD} to the random game inside the expectation.

    Note that $\hl_t$ is adapted to $a_1, \cdots, a_{t-1}$, but as stated in
    remark~\ref{rem:ellt-adapted}, the analysis of theorem~\ref{thm:MD}
    continues to hold in this case.
\end{proof}


\subsubsection{Unbiased estimation with one sample}
In \cref{thm:SMD-unbiased}, we assumed that $\hl_t$ is an unbiased estimator of $\ell_t$.
However, it is not clear if it is possible to construct such an estimator
with one sample $Y_t = \langle A_t, \ell_t \rangle$.
However, the next lemma not only shows that such a task is possible,
but also shows that it is possible for whole classes of distributions
$P_t$ with invertible covariance matrix.
\begin{lemma}[Unbiased 1-sample estimator]
    \label{lem:1-sample-estimator}
    Let $v \in \R^d$ be a vector,
    and $P \in \cP(\Rd)$ be a distribution.
    There exists a function $E: \Rd \times \R \to \Rd$ such that
    \begin{align*}
        \E_{A \sim P}[E(A, \langle A, v \rangle)] = v\,,
    \end{align*}
    if $\E_{A \sim P}[A A^\top]$ is invertible.
\end{lemma}
\begin{proof}
    Let $Y = \langle A, v \rangle$.
    It suffices to use
    \begin{align*}
        E(a, y) = \E_{A \sim P}[A A^\top]^{-1} a  y\,.
    \end{align*}
    This ensures that
    \begin{align*}
        \E_{A \sim P}[E(A, \langle A, v \rangle)]
         & = \E_{A \sim P}[\E_{A \sim P}[A A^\top]^{-1} A  \langle A, v \rangle] \\
         & = \E_{A \sim P}[A A^\top]^{-1} \E_{A \sim P}[A  A^\top v]             \\
         & = \E_{A \sim P}[A A^\top]^{-1} \E_{A \sim P}[A A^\top]  v             \\
         & = v\,.
    \end{align*}
\end{proof}
This shows that as long as the covariance matrix of $P_t$ is invertible,
it is possible to construct a sequence of unbiased estimators $\hl_t$,
such that $\E_t[\hl_t] = \ell_t$.

\subsubsection{Biased \smdalg}
\begin{algorithm}[h!]
    \begin{minipage}{13cm}
        \begin{mdframed}
            \begin{lstlisting}
args: $\eta \in \R$, $F$
$a_1 = \argmin_{a \in \cA} F(a)$
$\ta_1 = \argmin_{a \in \Rd} F(a)$

    Play $A_t \sim P_t$ and observe $Y_t = \langle A_t, \ell_t\rangle$
    Compute $\hl_t = E_t(A_t, Y_t)$

\end{lstlisting}
            \caption{Biased \smdalg}\label{alg:biased-SMD}
        \end{mdframed}
    \end{minipage}
\end{algorithm}

Suppose there are two sampling rules $P_t, P_t' \in \cP(\cA)$ where
$\E_{A \sim P_t}[A] = \E_{A \sim P_t'}[A]$, and \smdalg should choose
one of them for sampling $A_t$ at round $t$.
In \blo, these two \textit{result in the exact same loss} in round $t$,
and the reason for prefering one over the other is \textit{better exploration}.
The situation is different for \bco later.

This observation hints us that $P_t$ can (should?) be
chosen as a function of $\fmd(\ell_{1:t-1})$:
the best action to choose is $\fmd(\ell_{1:t-1})$,
and amongst all distributions $P$ that have the mean $\fmd(\ell_{1:t-1})$
(and so are effectively taking $\fmd(\ell_{1:t-1})$ in \blo), choose the one that
explores the best---results in \textit{better} $\hl_{1:n}$ sequence.

However, $\hl_t$ is a random vector that depends on both $P_t$ and $E_t$;
in this sense the best sampling (exploration) rule depends on the
estimation scheme in use ($E_t$), and vice-versa.

\paragraph{Bias:}
Maintaining {\smdalg}'s two degrees of freedoms $E_t$ and $P_t$ in
a way that $\E[\hl_t|\fmd(\hl_{1:t-1})] = \ell_t$ can be costly in terms of the bound on
\begin{align*}
    \E\left[
        \sum_{t=1}^{n} D_F(\bar{A}_{t}, \tilde{A}_{t+1})
        \right]\,,
\end{align*}
and its look-alikes. We have access to two sources of bias to control this quantity:
\begin{enumerate}
    \item $\E_{A \sim P_t}[A] \neq a_t$, where $a_t = \fmd(\hl_{1:t-1})$,
    \item $\E_{A \sim P_t}[\hl_t] \neq \ell_t$. Recall that $\hl_t = E_t(A_t, \langle A_t, \ell \rangle)$, so randomness is only by $A_t$.
\end{enumerate}
\begin{remark}[Is this an \mdalg-type of algorithm anymore?]
    At this point, it is fair to ask in what sense this is related
    to \mdalg? The key that distinguish Biased-\smdalg from
    \textit{all possible algorithms} is that $a_t = \fmd(\hl_{1:t-1})$
    is a sufficient statistics for choosing $P_t$ and $E_t$
    in Biased-\smdalg.
\end{remark}

\begin{theorem}[Biased-\smdalg, (\cite{lattimore2024bandit}, Theorem 8.14)]\label{thm:SMD-biased}
    The expected regret of Biased-\smdalg is bounded by
    \begin{align*}
        \Reg_n(a_\star) \leq
        \frac{D_F(a_\star, a_1)}{\eta} +
        \sum_{t=1}^{n}
        \E\left[
            \langle A_t - a_\star, \ell_t\rangle
            - \langle \fmd(\hl_{1:t-1}) - a_\star, \hl_t\rangle
            + \frac{1}{\eta} D_F(\fmd(\hl_{1:t-1}), \tfmd(\hl_{1:t}))
            \right]\,.
    \end{align*}
\end{theorem}
\begin{proof}
    Let $\hat{a}_t = \fmd(\hl_{1:t-1})$, we have
    \begin{align*}
        \Reg_n(a_\star)
         & =
        \E\left[
            \sum_{t=1}^{n}
            \langle A_t - a_\star, \ell_t\rangle
        \right] \\
         & =
        \E\left[
            \sum_{t=1}^{n}
            \langle A_t - a_\star, \ell_t\rangle
            - \langle \hat{a}_t - a_\star, \hl_t\rangle
            + \langle \hat{a}_t - a_\star, \hl_t\rangle
            \right]
        \\
         & \leq
        \frac{D_F(a_\star, a_1)}{\eta} +
        \E\left[
            \sum_{t=1}^{n}
            \langle A_t - a_\star, \ell_t\rangle
            - \langle \hat{a}_t - a_\star, \hl_t\rangle
            + \frac{1}{\eta} D_F(\fmd(\hl_{1:t-1}), \tfmd(\hl_{1:t}))
            \right]\,,
    \end{align*}
    where the inequality is by theorem~\ref{thm:MD}.
\end{proof}
\begin{remark}[\fmd in a virtual game]
    Note that $\hl_t$ is adapted to $A_t$ since the algorithm actually plays
    $A_t \sim P_t$.
    So you can think that $P_t$ is the exploration policy that
    is actually played which is not \smdalg as
    $\E[A_t|\fmd(\hl_{1:t-1})] \neq \fmd(\hl_{1:t-1})$.
    However, in the analysis we suppose that $\fmd(\hl_{1:t-1})$ is played
    \textbf{without any randomization} (so technically \mdalg) on
    $\hl_1, \cdots, \hl_n$.
\end{remark}

\begin{remark}[Closer look at theorem~\ref{thm:SMD-biased}---dummy bound]
    \label{rem:dummy-bound}
    If we look closer at theorem~\ref{alg:biased-SMD}, we can see that
    $\sum_{t=1}^{n} \E[\langle A_t - a_\star, \ell_t\rangle]$ term
    that appears on the RHS of the claim is exactly $\Reg_n(a_\star)$,
    so we did bound $\Reg_n(a_\star)$ by $\Reg_n(a_\star) + \square$ and
    for this theorem to be useful this $\square$ better be negative.
    More precisely, we did nothing other than adding and subtracting the value of the game
    that is played on $\hl_{1:n}$: the algorithm is \mdalg (not {\smdalg})
    and the loss sequence is $\hl_{1:n}$.
    So if we show the value of this game by $\hat{G}_n$, all we did is
    \begin{align*}
        \Reg_n(a_\star) = \Reg_n(a_\star) - \hat{G}_n + \hat{G}_n \,,
    \end{align*}
    which is an equality so far.
    The inequality is there for two reasons: (1) the upper-bound on
    $+ \hat{G}_n$, which is \textit{unavoidable},
    (2) $a_\star \neq \hat{a}_\star$---the true minimizer of the
    \textit{estimated game sequence} $\hl_{1:n}$ is not necessarily
    $a_\star$.
    The second reason might actually make the $\square$ term positive
    and the theorem useless.
    Needless to say, the theorem still holds if we exchange $a_\star$
    for $\hat{a}_\star$.
\end{remark}

\subsection{Exploration By Optimization(\expbyopt)}
\begin{algorithm}[h!]
    \begin{minipage}{12cm}
        \begin{mdframed}
            \begin{lstlisting}
args: learning rate $\eta >0$, precision $\epsilon > 0$
for $t = 1$ to $n$:
    compute $a_t = \fmd(\hl_{1:t-1})$
    find distribution $P_t \in \Delta(\cA)$ and $E_t \in (\Rd)^{(\cA \times \R)}$ s.t.
        $\sup_{\ell, a_\star} \Lambda_{a_t, \eta}(P_t, E_t) \leq \inf_{P,E} \sup_{\ell, a_\star} \Lambda_{a_t, \eta}(P, E) + \epsilon$
    sample $A_t \sim P_t$, and observe $Y_t = \langle \ell_t, A_t \rangle$
    compute $\hl_t = E_t(A_t, Y_t)$
\end{lstlisting}
            \caption{\expbyopt}\label{alg:exp-by-opt}
        \end{mdframed}
    \end{minipage}
\end{algorithm}

Suppose it's round $t$; the learner knows what \mdalg suggests to play, i.e., $a_t = \fmd(\hl_{1:t-1})$.
The idea is to choose the best $P_t$ and $E_t$ given $a_t$,
hoping that the term inside the expectation in theorem~\ref{thm:SMD-biased} could
then be bounded uniformly for all $t \in [n]$.
To this end, with $A\sim P$ and
$\hl = E(A, Y) = E(A, \langle A, \ell\rangle)$, define
\begin{align*}
    \Lambda_{a, \eta}(P, E || a_\star, \ell) \triangleq
    \frac{1}{\eta}
    \E\left[
        \langle A - a_\star, \ell\rangle
        - \langle a - a_\star, \hl\rangle
        + \frac{1}{\eta} D_{F^\star}(\nabla F(a) - \eta \hl, \nabla F(a))
        \right]\,,
\end{align*}
A bound on the following quantity would be ideal:
\begin{align*}
    \Lambda_{\cA, \cL, F}^\star \triangleq
    \sup_{\eta \geq 0, a \in \cA}
    \inf_{P \in \Delta(\cA), E \in (\R^d)^{\cA \times \R}}
    \sup_{a_\star \in \cA, \ell \in \cL}
    \Lambda_{a, \eta}(P, E || a_\star, \ell)\,.
\end{align*}
Moreover, if we allow the second sup player to randomize, the resulting
saddle point value can't get smaller,
so let $\nu \in \cP(\cA \times \cL)$ and
\begin{align*}
    \Lambda_{a, \eta}(P, E || \nu) \triangleq
    \frac{1}{\eta}
    \E_{(A_\star, \ell) \sim \nu}\left[
        \langle A - A_\star, \ell\rangle
        - \langle a - A_\star, \hl\rangle
        + \frac{1}{\eta} D_{F^\star}(\nabla F(a) - \eta \hl, \nabla F(a))
        \right]\,,
\end{align*}
and note that
\begin{align*}
    \sup_{a_\star \in \cA, \ell \in \cL}
    \Lambda_{a, \eta}(P, E || a_\star, \ell)
    \leq
    \sup_{\nu \in \cP(\cA \times \cL)}
    \Lambda_{a, \eta}(P, E || \nu)\,.
\end{align*}
This change introduces convexity in the second player's strategy set,
which results in the following minimax theorem.
\begin{lemma}[\cite{lattimore2024bandit}, Lemma 8.21]
    \label{lem:exp-by-opt-minimax}
    For all $\eta > 0, a \in \cA$,  we have
    \begin{align*}
        \inf_{P \in \Delta(\cA), E \in (\R^d)^{\cA \times \R}}
        \sup_{\nu \in \cP(\cA \times \cL)}
        \Lambda_{a,\eta}(P, E || \nu)
        =
        \sup_{\nu \in \cP(\cA \times \cL)}
        \inf_{P \in \Delta(\cA), E \in (\R^d)^{\cA \times \R}}
        \Lambda_{a,\eta}(P, E || \nu)\,.
    \end{align*}
\end{lemma}
\begin{remark}
    The second $A_\star$ in $\Lambda(P, E || \nu)$
    can be switched with any fixed action $a \in \cA$.
    See remark~\ref{rem:dummy-bound}.
\end{remark}

Let $\Delta(\nu, P) = \E_{A \sim P, (A_\star, \ell) \sim \nu}\!\left[\langle A - A_\star, \ell \rangle\right]$.
\begin{lemma}[Alternative $\Lambda(P, E || \nu)$]
    We have
    \begin{align}
        \Lambda(P, E || \nu)
        =
        \frac{1}{\eta}
        \left(
        \Delta(\nu, P) +
        \E\!\left[
                \langle A_\star, \hl \rangle
                + \frac{1}{\eta} F^\star\left(F'(a) - \eta \hl\right)
                - \frac{1}{\eta} F^\star(F'(a))
                \right]
        \right)
        \,.
        \label{eq:Lambda-simplified}
    \end{align}
\end{lemma}
\begin{proof}
    With $\hl = E(A, Y)= E(A, \langle A, \ell \rangle)$, we have
    \begin{align}
        \eta \Lambda(P, E || \nu)
         & =
        \E\!\left[
            \langle A - A_\star, \ell \rangle
            - \langle a - A_\star, \hl \rangle
            + \frac{1}{\eta} D_{F^\star}(F'(a) - \eta \hl, F'(a))
            \right]
        \nonumber \\
         & =
        \Delta(\nu, P) +
        \E\!\left[
        \langle A_\star - a, \hl \rangle
        + \frac{1}{\eta}
        \left(
        F^\star\left(F'(a) - \eta \hl\right)
        - F^\star(F'(a))
        - {F^\star}'(F'(a))^\top (- \eta \hl)
        \right)
        \right]
        \nonumber \\
         & =
        \Delta(\nu, P) +
        \E\!\left[
            \langle A_\star - a, \hl \rangle
            + \frac{1}{\eta}
            \left(
            F^\star\left(F'(a) - \eta \hl\right)
            - F^\star(F'(a))
            + \eta a^\top \hl
            \right)
            \right]
        \nonumber \\
         & =
        \Delta(\nu, P) +
        \E\!\left[
            \langle A_\star, \hl \rangle
            + \frac{1}{\eta} F^\star\left(F'(a) - \eta \hl\right)
            - \frac{1}{\eta} F^\star(F'(a))
            \right]\,.
        \nonumber
    \end{align}
\end{proof}

\subsubsection{Optimizing for $E$}
The inf player can choose the best optimizer given the
$\nu$ choice of the adversary and its own sampling policy $P$.
The following lemma shows that the \textit{optimal unrestricted estimator}
has a closed form.
\begin{lemma}
    \label{lem:optimal-E}
    Given $\nu \in \Delta(\cA \times \cL)$, and $P \in \Delta(d)$
    define $G_{\nu, p} \in (\Rd)^{(\cA \times \R)}$ as
    \begin{align*}
        G_{\nu, P}(b, y) & =
        \frac{1}{\eta}\left(
        F'(a) - F'(\E[A_\star|Y = y, A = b])
        \right)\,,
    \end{align*}
    then we have
    $
        \inf_{E \in (\Rd)^{\cA \times \R}}
        \Lambda(P, E|| \nu)
        =
        \Lambda(P, G_{\nu, p}|| \nu)\,.
    $
\end{lemma}
\begin{proof}
    By taking the derivative of $\Lambda(p, E || \nu)$ with respect to $E(b,y)$
    for any pair $(b,y) \in \cA \times \R$ we have
    \begin{align*}
        \tfrac{\partial}{\partial E(b,y)} \Lambda_\eta(P, E || \nu)
         & =
        \tfrac{\partial}{\partial E(b,y)}
        \left(
        \frac{1}{\eta}
        \left(
            \Delta(\nu, P) +
            \E\!\left[
                    \langle A_\star, \hl \rangle
                    + \frac{1}{\eta} F^\star\left(F'(a) - \eta \hl\right)
                    - \frac{1}{\eta} F^\star(F'(a))
                    \right]
            \right)
        \right)
        \\
         & =
        \tfrac{\partial}{\partial E(b,y)}
        \left(
        \frac{1}{\eta}
        \E\!\left[
                \langle A_\star, \hl \rangle
                + \frac{1}{\eta} F^\star\left(F'(a) - \eta \hl\right)
                \right]
        \right) \\
         & =
        \tfrac{\partial}{\partial E(b,y)}
        \tfrac{\PP(A=b, Y=y)}{\eta}
        \E\!\left[
            \langle A_\star, \hl \rangle
            + \frac{1}{\eta} F^\star\left(F'(a) - \eta \hl\right)
            \bigg| A = b, Y = y
        \right] \\
         & =
        \tfrac{\partial}{\partial E(b,y)}
        \tfrac{\PP(A=b, Y=y)}{\eta}
        \E\!\left[
            \langle A_\star, E(b, y) \rangle
            + \frac{1}{\eta} F^\star\left(F'(a) - \eta E(b, y)\right)
            \bigg|A = b, Y = y
        \right] \\
         & =
        \tfrac{\PP(A=b, Y=y)}{\eta}
        \E\!\left[
        A_\star
        - {F^\star}'\left(F'(a) - \eta E(b, y)\right)
        \bigg| A = b, Y = y
        \right] \\
         & =
        \tfrac{\PP(A=b, Y=y)}{\eta}
        \left(
        \E\!\left[
            A_\star
            |A = b, Y = y\right]
        - {F^\star}'\left(F'(a) - \eta E(b, y)\right)
        \right)\,,
    \end{align*}
    and by setting the gradient to zero, we get
    \begin{align*}
        \E\!\left[
            A_\star
            | A = b, Y = y\right]
         & = {F^\star}'\left(F'(a) - \eta E(b, y)\right) \\
        \Rightarrow
        F'\left(\E\!\left[
            A_\star
            | A = b, Y = y\right]\right)
         & = F'(a) - \eta E(b, y)                        \\
        \Rightarrow
        E(b, y)
         & =
        \frac{1}{\eta}
        \left(
        F'(a)
        -
        F'\left(\E\!\left[
            A_\star
            | A = b, Y = y\right]\right)
        \right)\,. \qedhere
    \end{align*}
\end{proof}

The next theorem shows the connection between this optimal estimator and the information ratio.
For reasons that will be clear later, let
$\texttt{pr} = \E[A_\star]$ (`pr' for prior) and
$\texttt{po} = \E[A_\star|A, Y]$ (`po' for posterior).
\begin{lemma}
    \label{lem:optimal-Lambda-wrong-turn}
    When using the optimal estimator $G_{\nu, P}$, we have
    \begin{align*}
        \Lambda(P, G_{\nu, P}|| \nu)
         & = \frac{1}{\eta} \Delta(\nu, P)
        - \frac{1}{\eta^2}
        \E\!\left[
            D_F\left(\texttt{po}, a\right)
            \right]\,.
    \end{align*}
\end{lemma}
\begin{proof}
    From lemma~\ref{lem:optimal-E}, the optimal estimator is
    $
        G_{\nu, P}(b, y)  =
        \frac{1}{\eta}\left(
        F'(a) - F'(\E[A_\star|\langle \ell, b\rangle = y])
        \right)$.
    By plugging in $G_{\nu, P}$ into eq~\ref{eq:Lambda-simplified}, we have
    \begin{align*}
         & \eta \Lambda(P, G_{\nu, P}|| \nu)  - \Delta(\nu, P)
        \\
         & =
        \E\!\left[
            \langle A_\star, \hl \rangle
            + \frac{1}{\eta} F^\star\left(F'(a) - \eta \hl\right)
            - \frac{1}{\eta} F^\star(F'(a))
            \right]
        \\
         & =
        \E\!\left[
            \langle A_\star, G_{\nu, P}(A, Y) \rangle
            + \frac{1}{\eta} F^\star\left(F'(a) - \eta G_{\nu, P}(A, Y)\right)
            - \frac{1}{\eta} F^\star(F'(a))
            \right]
        \\
         & =
        \E\!\left[
            \eta^{-1}\langle A_\star,
            F'(a) - F'(\E[A_\star|\langle \ell, A\rangle = Y])
            \rangle
            + \frac{1}{\eta} F^\star\left(
            F'(\E[A_\star|\langle \ell, A\rangle = Y])
            \right)
            - \frac{1}{\eta} F^\star(F'(a))
            \right]
        \\
         & =\eta^{-1}
        \E\!\bigg[
            \big\langle
            \E[A_\star|\langle \ell, A\rangle = Y],
            F'(a) - F'(\E[A_\star|\langle \ell, A\rangle = Y])
            \big\rangle
            + F^\star\left(
            F'(\E[A_\star|\langle \ell, A\rangle = Y])
            \right)
            - F^\star(F'(a))
            \bigg]
        \\
         & =
        -\eta^{-1}
        \E\!\bigg[
            D_{F^\star}\left(F'(a), F'(\E[A_\star|\langle \ell, A\rangle = Y])\right)
            \bigg]
        \\
         & =
        -\eta^{-1}
        \E\!\bigg[
            D_F\left(
            \E[A_\star|\langle \ell, A\rangle = Y], a
            \right)
            \bigg]\,. \qedhere
    \end{align*}
\end{proof}
Although the result of lemma~\ref{lem:optimal-Lambda-wrong-turn} is
elegant, it is not useful for proving the connection to the information ratio.
\begin{lemma}[Expected Bregman Identity]
    Let $F:\R^d \to \R$ be a strictly convex and differentiable function.
    For any random vector $X$ and any point $a$ in the domain of $F$, we have
    \begin{align*}
        \E\left[D_F(X, a)\right]
         & =
        \E\left[
            D_F\left( X, \E[X]\right)
            \right]
        +
        D_F\left( \E[X], a\right)\,.
    \end{align*}
\end{lemma}

\begin{lemma}
    \label{lem:the-right-turn}
    When using the optimal estimator $G_{\nu, P}$, we have
    \begin{align*}
        \E\left[D_F\left(\texttt{po}, a\right)\right]
         & =
        \E\left[
            D_F\left( \texttt{po}, \texttt{pr}\right)
            \right]
        +
        D_F\left( \texttt{pr}, a\right)\,.
    \end{align*}
\end{lemma}
\begin{proof}
    Follows directly from the expected Bregman identity.
\end{proof}
\begin{remark}
    Note how the result of lemma~\ref{lem:the-right-turn} is similar to the
    Pythagorean theorem for Bregman divergences
    \begin{align*}
        D_R(z, x) \geq D_R(z, \Pi(x)) + D_R(\Pi(x), x)\,.
    \end{align*}
\end{remark}
The significance of the following theorem is that it connects
the value of the game to the information ratio.
This makes the whole analysis we did so far worthwhile since now we can
leverage existing bounds on the information ratio to bound
the value of the game, and hence the regret of \expbyopt.
\begin{theorem}
    \label{thm:optimal-Lambda-info-ratio}
    When using the optimal estimator $G_{\nu, P}$, we have
    \begin{align*}
        \Lambda(P, G_{\nu, P}|| \nu)
         & \leq \frac{
            \Delta(\nu, P)^2
        }{
            4 \E\left[D_F\left( \E[A^\star|A,Y], \E[A^\star]\right)\right]
        }\,.
    \end{align*}
\end{theorem}
\begin{proof}
    From lemma~\ref{lem:optimal-Lambda-wrong-turn} and lemma~\ref{lem:the-right-turn}, we have
    \begin{align*}
        \Lambda(P, G_{\nu, P}|| \nu)
         & = \frac{1}{\eta} \Delta(\nu, P)
        - \frac{1}{\eta^2}
        \E\!\left[
            D_F\left(\texttt{po}, a\right)
        \right]                            \\
         & = \frac{1}{\eta} \Delta(\nu, P)
        - \frac{1}{\eta^2}
        \left(
        \E\left[
            D_F\left( \texttt{po}, \texttt{pr}\right)
            \right]
        +
        D_F\left( \texttt{pr}, a\right)
        \right)                            \\
         & = \frac{1}{\eta} \Delta(\nu, P)
        - \frac{1}{\eta^2}
        \E\left[
            D_F\left( \E[A^\star|A,Y], \E[A^\star]\right)
        \right]                            \\
         & \leq \frac{
            \Delta(\nu, P)^2
        }{
            4 \E\left[D_F\left( \E[A^\star|A,Y], \E[A^\star]\right)\right]
        }\,,
    \end{align*}
    where the first inequality is by dropping the non-negative term $D_F( \texttt{pr}, a)$,
    and the second inequality is by maximizing over $\eta$.
\end{proof}

\subsubsection{Optimizing for $E$---multiplicative in $Y$}
The player (even in the primal problem), has the knowledge of
$\eta$ and $a_t = \fmd(\hl_{1:t-1})$, and also has the control over $P_t$,
so given these three variables, it can limit its search for the
optimal estimator.
However, the space $(\Rd)^{(\cA \times \R)}$ is huge.
The learner can further restrict its search to be only over
estimators $E_T(x,y)$ of the form
\begin{align*}
    E_T(b,y) & = \frac{y}{\eta P(b)} T(b)
\end{align*}
and then only optimizes for $T \in (\Rd)^\cA \equiv \R^{d \times |\cA|}$.
\begin{theorem}[Multiplicative-$Y$ estimators]
    \label{thm:optimal-mult-y}
    Given $\nu \in \Delta(\cA \times \cL)$,
    and $P \in \Delta(d)$, for a matrix $T \in \R^{|\cA| \times d}$,
    define the estimator $E_T:\cA \times \R \to \Rd$ as
    \begin{align*}
        E_T(a, y) = \frac{y}{\eta P_a} T_a\,.
    \end{align*}
    Then, for $T^\star \in \R^{|\cA| \times d}$ such that
    \begin{align*}
        \inf_{T \in \R^{|\cA| \times d}}
        \Lambda(P, E_T|| \nu)
        =
        \Lambda(P, E_{T^\star}|| \nu)\,,
    \end{align*}
    it holds that
    \begin{align}
        \E\!\left[
            \hl \odot A_\star
            | A
            \right]
         & =
        \E\!\left[
            \hl \odot {F^\star}'\left(F'(a) - \eta \hl\right)
            | A
            \right]\,
        \label{eq:mul-E-opt}
    \end{align}
    almost surely, where $\hl = E_{T^\star}(A, Y)$.
\end{theorem}
\begin{proof}
    Starting from eq~\ref{eq:Lambda-simplified}, we are interested in the $T$ that minimizes
    \begin{align*}
        \eta \Lambda(P, T || \nu)
         & =
        \Delta(\nu, P) +
        \E\!\left[
            \langle A_\star, \hl \rangle
            + \frac{1}{\eta} F^\star\left(F'(a) - \eta \hl\right)
            - \frac{1}{\eta} F^\star(F'(a))
            \right]\,,
    \end{align*}
    but note that the first term $\Delta(\nu, P)$
    and the last term $F^\star(F'(a))$ can be dropped
    from the optimization since they don't depend on the estimator choice
    $T$ (no $\hl$ present).

    For $b \in \cA$, we solve for $T^\star_b$ through differentiation:
    (dropping $\Delta(\nu, P)$ and $F^\star(F'(a))$)
    \begin{align*}
        \tfrac{\partial}{\partial T_b} \Lambda(P, T||\nu)
         & =
        \tfrac{\partial}{\partial T_b}
        \eta^{-1}
        \E\!\left[
            \langle A_\star, \hl \rangle
            + \frac{1}{\eta} F^\star\left(F'(a) - \eta \hl\right)
            \right]
        \\
         & =
        \tfrac{P_b}{\eta} \cdot
        \tfrac{\partial}{\partial T_b}
        \E\!\left[
            \langle A_\star, \tfrac{Y}{\eta P_b}T_b \rangle
            + \frac{1}{\eta} F^\star\left(F'(a) - \eta \tfrac{Y}{\eta P_b}T_b\right)
            | A = b
            \right]
        \\
         & =
        \tfrac{P_b}{\eta^2} \cdot
        \tfrac{\partial}{\partial T_b}
        \E\!\left[
            \langle A_\star, \tfrac{Y}{P_b}T_b \rangle
            + F^\star\left(F'(a) - \tfrac{Y}{P_b}T_b\right)
            | A = b
            \right]
        \\
         & =
        \tfrac{P_b}{\eta^2} \cdot
        \E\!\left[
            \tfrac{Y}{P_b} A_\star
            - \tfrac{Y}{P_b} {F^\star}'\left(F'(a) - \tfrac{Y}{P_b}T_b\right)
            | A = b
            \right]
        \\
         & =
        \tfrac{1}{\eta^2} \cdot
        \E\!\left[
            Y A_\star
            - Y {F^\star}'\left(F'(a) - \tfrac{Y}{P_b}T_b\right)
            | A = b
            \right]\,,
    \end{align*}
    and by setting the derivative to zero, we have
    \begin{align}
        \E\!\left[
            Y A_\star
            | A = b
            \right]
         & =
        \E\!\left[
            Y {F^\star}'\left(F'(a) - \tfrac{Y}{P_b}T_b\right)
            | A = b
        \right] \nonumber \\
        \abovelabel{$\odot \tfrac{T_b^\star}{\eta P_b}$}
        \Rightarrow
        \E\!\left[
            \hl \odot A_\star
            | A = b
            \right]
         & =
        \E\!\left[
            \hl \odot {F^\star}'\left(F'(a) - \eta \hl\right)
            | A = b
            \right]\,,
    \end{align}
    for all $b \in \cA$.
\end{proof}
It can be seen that $T^\star$ doesn't have a closed form,
but we may hope that eq~\ref{eq:mul-E-opt} appear useful.
\begin{lemma}
    \label{lem:E-optimized-Lambda}
    For $T^\star$ defined in Thm~\ref{thm:optimal-mult-y}, we have
    \begin{align}
        \Lambda(P, E_{T^\star} || \nu)
         & = \frac{1}{\eta}
        \Delta(\nu, P) -
        \frac{1}{\eta^2}
        \E\!\left[
            D_{F^\star}\left(F'(a), F'(a) - \eta \hl\right)
            \right]\,.
        \label{eq:E-optimized-Lambda}
    \end{align}
\end{lemma}
\begin{proof}
    Again, starting from eq~\ref{eq:Lambda-simplified}, we have
    \begin{align*}
        \eta \Lambda(P, E_{T^\star} || \nu)
         & =
        \Delta(\nu, P) +
        \E\!\left[
            \langle A_\star, \hl \rangle
            + \frac{1}{\eta} F^\star\left(F'(a) - \eta \hl\right)
            - \frac{1}{\eta} F^\star(F'(a))
            \right]
        \\
         & =
        \Delta(\nu, P) +
        \E\!\left[
            \E\!\left[
                \langle A_\star, \hl \rangle
                + \frac{1}{\eta} F^\star\left(F'(a) - \eta \hl\right)
                - \frac{1}{\eta} F^\star(F'(a))
                \bigg|
                A
                \right]
            \right]
        \\
         &
        \abovelabel{by eq~\ref{eq:mul-E-opt}}
        =
        \Delta(\nu, P) +
        \E\!\left[
            \E\!\left[
                \langle {F^\star}'\left(F'(a) - \eta \hl\right), \hl \rangle
                + \frac{1}{\eta} F^\star\left(F'(a) - \eta \hl\right)
                - \frac{1}{\eta} F^\star(F'(a))
                \bigg|
                A
                \right]
            \right]
        \\
         & =
        \Delta(\nu, P) -
        \frac{1}{\eta}
        \E\!\left[
            D_{F^\star}\left(F'(a), F'(a) - \eta \hl\right)
            \right]\,.
    \end{align*}
\end{proof}

Let
\begin{align*}
    \tilde{A} = {F^\star}'(F'(a) - \eta \hl)\,,
\end{align*}

\begin{remark}
    By combining the original form of $\Lambda(P, E || \nu)$
    and eq~\ref{eq:E-optimized-Lambda}, we have
    \begin{align*}
        \E\!\left[
            \langle A_\star - a, \hl \rangle
            \right]
         & =
        - \frac{1}{\eta}
        \left(
        \E\!\left[
                D_F(\tilde{A}, a)
                \right]
        +
        \E\!\left[
                D_F(\tilde{A}, a)
                \right]
        \right)\,.
    \end{align*}
\end{remark}

\begin{remark}[Side by side]
    \label{rem:posterior-tildeA}
    From Lemmas~\ref{lem:optimal-Lambda-wrong-turn}
    and eq~\ref{eq:E-optimized-Lambda}, we have
    \begin{align}
        \Lambda(P, G_{\nu, P}|| \nu)
         & =
        \frac{1}{\eta} \Delta(\nu, P)
        - \frac{1}{\eta^2}
        \E\!\left[D_F(\texttt{po}, a)\right]\,,
        \tag{unrestricted}
        \label{eq:Lambda-unrestricted}     \\
        \Lambda(P, E_{T^\star} || \nu)
         & = \frac{1}{\eta} \Delta(\nu, P)
        - \frac{1}{\eta^2}
        \E\!\left[
            D_F(\tilde{A}, a)
            \right]\,.
        \tag{mul-$Y$}
        \label{eq:Lambda-mul-Y}
    \end{align}
\end{remark}
Rewrite eq~\ref{eq:mul-E-opt} as
\begin{align}
    \E\!\left[
        \hl \odot \tilde{A}
        | A
        \right]
     & =
    \E\!\left[
        \hl \odot A_\star
        | A
        \right]
    \nonumber
    \\
     & =
    \E\!\left[
        \E\!\left[
            \hl \odot A_\star
            | A, Y
            \right]
        | A
        \right]
    \nonumber
    \\
     & =
    \E\!\left[
        \hl
        \odot
        \E\!\left[
            A_\star
            | A, Y
            \right]
        | A
        \right]
    \nonumber
    \\
     & =
    \E\!\left[
        \hl
        \odot
        \texttt{po}
        | A
        \right]\,.
    \label{eq:tilde-A-po}
\end{align}
Secondly, note that
\begin{align}
    F'(a) - F'( \tilde{A} )
     & =
    \eta \hl\qquad
    \Rightarrow\qquad
    \hl
    =
    \frac{1}{\eta}
    \left(
    F'(a) - F'( \tilde{A} )
    \right)\,, \label{eq:hl-alternate}
\end{align}
which together with eq~\ref{eq:tilde-A-po} gives
\begin{align}
    \E\!\left[
        \left(
        F'(a) - F'( \tilde{A} )
        \right)
        \odot
        \left(
        \tilde{A}
        -
        \texttt{po}
        \right)
        | A
        \right]
     & = 0
    \Rightarrow
    \E\!\left[
        \left\langle
        F'(a) - F'( \tilde{A} )
        ,
        \texttt{po}
        -
        \tilde{A}
        \right\rangle
        | A
        \right]
    = 0\,.
    \label{eq:orthogonality}
\end{align}
Recall the three points identity for Bregman divergences:
\begin{align*}
    \langle
    F'(z) - F'(y),
    x - y
    \rangle
     & =
    D_F(x, y) + D_F(y, z) - D_F(x, z)\,,
\end{align*}
which when applied to eq~\ref{eq:orthogonality} with
$(x, y, z) = (\texttt{po}, \tilde{A}, a)$, gives
\begin{align*}
    \E\!\left[
        D_F(\texttt{po}, \tilde{A})
        + D_F(\tilde{A}, a)
        | A
        \right]
     & =
    \E\!\left[
        D_F(\texttt{po}, a)
        | A
        \right]
    \,,
\end{align*}
which together with lemma~\ref{lem:the-right-turn} and averaging over $A$, implies that
\begin{align}
    \E\!\left[
        D_F(\texttt{po}, \tilde{A})
        + D_F(\tilde{A}, a)
        \right]
     & =
    \E\!\left[
        D_F(\texttt{po}, \texttt{pr})
        \right]
    +
    D_F(\texttt{pr}, a)\,.
    \label{eq:expected-bregman-identity}
\end{align}

Now things become interesting.
To make the connection to the information ratio (similar to Theorem~\ref{thm:optimal-Lambda-info-ratio}),
we need (or at least naively want) that
\begin{align*}
    \E[D_F(\dA, a)]
     & =
    \E[D_F(\po, \pr)]\,,
\end{align*}
which, now, by eq~\ref{eq:expected-bregman-identity} is equivalent to
\begin{align}
    \label{eq:alternative-condition}
    \E\!\left[
        D_F(\texttt{po}, \tilde{A})
        \right]
     & = D_F(\texttt{pr}, a)\,.
\end{align}
This should hold \textbf{for all values of} $a$.
For the choice of $a = \texttt{pr}$, eq~\ref{eq:alternative-condition} becomes
\begin{align*}
    \E\!\left[
        D_F(\texttt{po}, \tilde{A})
        \right]
     & = 0\,,
\end{align*}
which by the non-negativity of Bregman divergences implies that
\begin{align*}
    \texttt{po} = \tilde{A}
    \qquad \text{i.e.} \qquad
    \E\left[A_\star| A, Y\right] = {F^\star}'(F'(\texttt{pr}) - \eta \hl)\,,
\end{align*}
almost surely.

Now it's easy to see that this can't be ensured, since the LHS in
\begin{align*}
    \E\left[A_\star| A, Y\right]
     & = {F^\star}'\left(F'(\texttt{pr}) - \frac{T^\star_A}{P_A} Y \right)
    \\
    \Leftrightarrow
    F'\left(\E\left[A_\star| A, Y\right]\right)
     & = F'(\texttt{pr}) - \frac{T^\star_A}{P_A} Y
    \\
    \Leftrightarrow
    \frac{T^\star_A}{P_A} Y
     & = F'(\texttt{pr}) - F'\left(\po\right)\,,
\end{align*}
depends on the joint distribution of $A_\star$ and $Y$ given $A$,
while the RHS depends only on the marginal distribution of $Y$ given $A$.

\subsubsection{Multiplicative factor?}
It's natural to wonder if there exists a multiplicative factor $\alpha$
such that
\begin{align*}
    \frac{
        \E\left[
            D_F(\dA, a)
            \right]
    }{
        \E\left[
            D_F(\po, \pr)
            \right]
    } \geq \alpha > 0\,,
\end{align*}
which would still allow us to make the connection to the information ratio.
However, this is not the case.

Let $Z \sim \cN(0,1)$ be a standard normal random variable.
Make the following choices:
\begin{itemize}
    \item $F$ is the quadratic potential, i.e. $F(x) = \frac{1}{2} \|x\|^2$.
    \item $A^\star = (-\beta Z, \omega (Z^2 - 1), 0, \ldots, 0)$,
          where $\beta, \omega > 0$ are parameters to be chosen later.
    \item $\ell = (Z, 0, \ldots, 0)$.
\end{itemize}
With these choices, we have
\begin{align*}
    Y                 & = \langle \ell, A \rangle = Z A_1\,,                                                                      \\
    \pr               & = \E\left[A_\star\right] = (0,0,\ldots,0)\,,                                                              \\
    \po               & = \E\left[A_\star | A, Y\right]=A_\star\,,                                                                \\
    \dA               & = {F^\star}'(F'(\pr) - \eta \hl) = -\eta \hl = -\frac{Y}{\eta P_A} T^\star_A\,, \tag{quadratic potential} \\
    \E[D_F(\po, \pr)] & = \frac{1}{2} \E[\|\po\|^2]
    = \frac{1}{2} \E\left[\beta^2 Z^2 + \omega^2 (Z^2 - 1)^2\right]
    = \frac{1}{2} \left(\beta^2 + 2 \omega^2\right)\,,
\end{align*}
and it remains to compute $\E[D_F(\dA, \pr)]$.
From eq~\ref{eq:mul-E-opt}, we have
\begin{align*}
    \E\left[Y \cdot A_\star | A\right] = \E\left[Y \cdot \dA | A\right]\,,
\end{align*}
where the RHS is
\begin{align*}
    \E[Y \cdot \dA]
     & = \E\left[Z A_1 \cdot \left(-\frac{Z A_1}{\eta P_A} T^\star_A\right) | A\right]
    = -\frac{A_1^2}{\eta P_A} \cdot T^\star_A\,,
\end{align*}
and the LHS is
\begin{align*}
    \E[Y \cdot A^\star |A] & = \E\left[Z A_1 \cdot (-\beta Z, \omega (Z^2 - 1), 0, \ldots, 0) | A\right] \\
                           & = (-\beta \E[Z^2] A_1, \omega \E[Z^3 - Z] A_1, 0, \ldots, 0)                \\
                           & = (-\beta A_1, 0, 0, \ldots, 0)\,.
\end{align*}
Thus, we have
\begin{align*}
    T^\star_A = \left(\frac{\eta P_A}{A_1} \beta, 0, \ldots, 0\right)\,.
\end{align*}

Now, note that
\begin{align*}
    \E[D_F(\dA, \pr)]
     & = \E[D_F(\dA, 0)]                                                               \\
     & = \E[F(\dA) - F(0) - \langle F'(0), \dA - 0\rangle] = \frac{1}{2} \E[\|\dA\|^2] \\
     & = \frac{1}{2} \E\left[\left\|-\frac{Y}{\eta P_A} T^\star_A\right\|^2\right]
    = \frac{1}{2} \E\left[\left(\frac{Z A_1}{\eta P_A} \cdot \frac{\eta P_A}{A_1} \beta\right)^2\right]
    = \frac{1}{2} \cdot \beta^2\,.
\end{align*}
And now we can compute the ratio:
\begin{align*}
    \frac{\E[D_F(\dA, \pr)]}{\E[D_F(\po, \pr)]}
     & = \frac{\frac{1}{2} \beta^2}{\frac{1}{2} (\beta^2 + 2 \omega^2)}
    = \frac{\beta^2}{\beta^2 + 2 \omega^2}\,,
\end{align*}
which can be made arbitrarily close to $0$ by choosing $\omega \to \infty$.
Thus, there is no multiplicative factor $\alpha$ that works in general.

\begin{proposition}[Characterization in the multiplicative-$Y$ class]
    \label{prop:multY-iff}
    Consider $E_T$ with $\hl=\frac{Y}{\eta P_A}T_A$. Then $\dA=\po$ a.s.\ if and only if
    \begin{equation}
        \label{eq:affine-dual}
        F'\!\big(\po\big)
        \;=\;
        F'(a) - \frac{Y}{P_A}\,T_A
        \qquad\text{almost surely.}
    \end{equation}
    Equivalently, conditional on $A=b$, the map $y\mapsto F'\!\big(\E[A_\star\mid A=b,Y=y]\big)$ is (a.s.) \emph{affine in $y$} with slope $-T_b/P_b$ and intercept $F'(a)$.
\end{proposition}

\begin{proof}
    Using the multiplicative form $\hl=\frac{Y}{\eta P_A}T_A$ we have
    \[
        \dA
        \;=\;
        {F^\star}'\!\left(F'(a)-\eta\hl\right)
        \;=\;
        {F^\star}'\!\left(F'(a)-\frac{Y}{P_A}T_A\right).
    \]
    Thus $\dA=\po$ a.s.\ $\iff$
    \(
    \po = {F^\star}'\!\left(F'(a)-\frac{Y}{P_A}T_A\right)
    \)
    a.s.\ $\iff$ applying $F'$ (which is injective) yields \eqref{eq:affine-dual}.
\end{proof}

\section{Closed form for $\cA = [d]$?}
Let's look back at the result of Lemma~\ref{lem:E-optimized-Lambda}:
\begin{align*}
    \Lambda(P, E_{T^\star} || \nu)
     & =
    \E\!\left[
        \frac{1}{\eta}
        \langle A - A_\star, \ell\rangle
        -
        \frac{1}{\eta^2}
        D_{F^\star}\left(F'(a), F'(a) - \eta \hl\right)
        \right]\,.
\end{align*}
Two questions arise for $\cA = [d]$, and $F$ being the negentropy:
\begin{itemize}
    \item What does $\Lambda(P, E_{T^\star} || \nu)$ look like?
    \item Can we have a closed form for $T^\star$?
\end{itemize}
Note that an action $A \sim P$ is simply $A = e_a$ with probability $P_a$.
Therefore, there is no policy optimization, since for a given $P \in \Delta(d)$,
there is only one set of weights $P_a$ such that
\begin{align*}
    P = \sum_{a=1}^d P_a e_a\,,
\end{align*}
which means that once the learner chooses $E_{T^\star}$, the value of the game is determined.

For the first question, we have
\begin{align}
    D_{F^\star}\left(F'(a), F'(a) - \eta \hl\right)
     & =
    F^\star(F'(a)) - F^\star(F'(a) - \eta \hl)
    - \eta {F^\star}'(F'(a) - \eta \hl) \hl
    \nonumber
    \\
     & =
    \langle 1, a\rangle -
    \langle 1, a \odot \exp(-\eta \hl)\rangle
    -
    \langle 1, a \odot \exp(-\eta \hl) \odot \eta \hl\rangle
    \nonumber
    \\
     & =
    \langle a,
    1
    - \exp(-\eta \hl)
    - \eta \hl \odot \exp(-\eta \hl)
    \rangle
    \label{eq:linearized-D_F}
    \\
     & =
    \langle a,
    1
    - \exp(-Y \tfrac{T_A}{\eta P_A})
    - Y \tfrac{T_A}{\eta P_A} \odot \exp(-Y \tfrac{T_A}{\eta P_A})
    \rangle\,.\nonumber
\end{align}
For the second question, we should look at \cref{thm:optimal-mult-y} and
see if we can solve for $T^\star$ in closed form.
Starting from the result of \cref{thm:optimal-mult-y} we have
\begin{align*}
    \E\!\left[
        Y A_\star
        | A = b
        \right]
     & =
    \E\!\left[
        Y {F^\star}'\left(F'(a) - \tfrac{Y}{P_b}T_b\right)
        | A = b
    \right] \nonumber \\
    \abovelabel{F negentropy}
    \Rightarrow
    \E\!\left[
        Y A_\star
        | A = b
        \right]
     & =
    \E\!\left[
        Y \exp\left(\ln(a) - \tfrac{Y}{\eta P_b}T_b\right)
        | A = b
    \right] \nonumber \\
     & =
    a \odot
    \E\!\left[
        Y \exp\left(- \tfrac{Y}{\eta P_b}T_b\right)
        | A = b
    \right] \nonumber \\
    \abovelabel{$Y = \langle \ell, b\rangle$, and $A \perp \ell, A_\star$}
    \Rightarrow
    \E\!\left[
        \ip{\ell, b} A_\star
        \right]
     & =
    a \odot
    \E\!\left[
        \ip{\ell, b} \exp\left(- \tfrac{\ip{\ell, b}}{\eta P_b}T_b\right)
        \right]
    \\
    \abovelabel{$b$ is one-hot}
    \Rightarrow
    \E\!\left[
        \ell_b A_\star
        \right]
     & =
    a \odot
    \E\!\left[
        \ell_b \exp\left(- \tfrac{\ell_b}{\eta P_b}T_b\right)
        \right]
    \\
    \abovelabel{$z$-th coord}
    \Rightarrow
    \E\!\left[
        \ell_b A_{\star, z}
        \right]
     & =
    a_z \cdot
    \E\!\left[
        \ell_b \exp\left(- \tfrac{\ell_b}{\eta P_b}T_{b, z}\right)
        \right]
    \\
     & =
    a_z \cdot
    \E\!\left[
        \ell_b
        \exp\left(- \tfrac{T_{b, z}}{\eta P_b}\ell_b\right)
        \right]
    \\
     & =
    a_z \cdot
    M_{\ell_{b}}'\left(- \tfrac{T_{b, z}}{\eta P_b}\right)
\end{align*}
Therefore, since the inverse of $M_{\ell_b}'$ is well-defined,
and is bijective from $(-\infty, \infty)$ to $(0, \infty)$,
we have
\begin{align*}
    T_{b, z}
     & =
    - \eta P_b
    {M_{\ell_b}'}^{-1}\left(
    \frac{\E\!\left[
        \ell_b A_{\star, z}
        \right]}{a_z}
    \right)\,.
\end{align*}

\paragraph{The problem with linear loss class}
Consider the \olo problem over $\cK = [-1, 1]$.
For a discretization dimension $d$
and a real number $m_\ell$, define the loss
\begin{align*}
    \ell = (-m_\ell, -\tfrac{d-1}{d}m_\ell, \cdots,
    -\tfrac{1}{d}m_\ell, 0, \tfrac{1}{d}m_\ell,
    \tfrac{2}{d}m_\ell, \cdots, \tfrac{d-1}{d} m_\ell, m_\ell),
\end{align*}
and the action set is the simplex $\Delta(2d + 1)$.

Then for an action $p \in \Delta(2d + 1)$ we have
\begin{align*}
    \langle \ell, p\rangle = m_\ell
    \langle
    (-1, -\tfrac{d-1}{d}, \cdots, -\tfrac{1}{d},
    0, \tfrac{1}{d}, \cdots, \tfrac{d-1}{d}, 1)
    , p
    \rangle\,,
\end{align*}
which means the \textit{effective action} is
$
    A \triangleq
    \langle
    (-1, -\tfrac{d-1}{d}, \cdots, -\tfrac{1}{d},
    0, \tfrac{1}{d}, \cdots, \tfrac{d-1}{d}, 1)
    , p
    \rangle
    \in [-1, 1]
$ and
\begin{align*}
    Y = \langle p, \ell \rangle = m_\ell A,\,\,
    \text{and}
    \qquad
    \hl = Y \tfrac{T_A}{P_A} = m_\ell A \tfrac{T_A}{P_A}\,.
\end{align*}
Then Thm~\ref{thm:optimal-mult-y} translates to
\begin{align*}
    \E\!\left[
        \hl \odot A_\star
        | A
        \right]
     & =
    \E\!\left[
        \hl \odot {F^\star}'\left(F'(a) - \eta \hl\right)
        | A
        \right]
    \\
    \Rightarrow
    \E\!\left[
        m_\ell A \tfrac{T_A}{P_A} \odot A_\star
        | A
        \right]
     & =
    \E\!\left[
        m_\ell A \tfrac{T_A}{P_A} \odot {F^\star}'\left(F'(a) - \eta m_\ell A \tfrac{T_A}{P_A}\right)
        | A
        \right]
    \\
    \abovelabel{$A,T_A \neq 0$}
    \Rightarrow
    \E\!\left[
        m_\ell \odot A_\star
        | A
        \right]
     & =
    \E\!\left[
        m_\ell \odot {F^\star}'\left(F'(a) - \eta m_\ell A \tfrac{T_A}{P_A}\right)
        | A
        \right]
    \\
    \abovelabel{$F$ negentropy}
    \Rightarrow
    \E\!\left[
        m_\ell \odot A_\star
        | A
        \right]
     & =
    \E\!\left[
        m_\ell \odot \exp\left(\ln(a) - \eta m_\ell A \tfrac{T_A}{P_A}\right)
        | A
        \right]
    \\
    \Rightarrow
    \E\!\left[
        m_\ell \odot A_\star
        | A
        \right]
     & =
    a \odot
    \E\!\left[
        m_\ell \odot \exp\left(- \eta m_\ell A \tfrac{T_A}{P_A}\right)
        | A
        \right]
    \\
    \abovelabel{For each coord $z$}
    \Rightarrow
    \E\!\left[

\paragraph{The function $f(x) = 1 - \exp(-x) - x \exp(-x)$ and $X \geq 0$ a.s.}
Can we show that
\begin{align*}
    \E[f(X)] \simeq \E[X^2]\,?
\end{align*}

First, we have
\begin{align*}
    \E[X^2] = 2 \int_{0}^{\infty} t \PP(X \geq t) dt\,.
\end{align*}
Secondly, note that $f'(x) = x \exp(-x)$, and $f(0) = 0$. Hence,
$f(x) = \int_{0}^x t \exp(-t) dt$, so
\begin{align*}
    \E[f(X)]
    = \E\left[
        \int_0^X t \exp(-t) dt
        \right]
    = \int_0^\infty t \exp(-t) \PP(X \geq t) dt\,.
\end{align*}
Now, since $\exp(-t) \leq 1$ for $t \geq 0$, we have
\begin{align*}
    \E[f(X)] \leq \frac{\E[X^2]}{2}\,,
\end{align*}

However, if $X \in [0, B]$ a.s., then
\begin{align*}
    \E[f(X)]
     & = \int_0^\infty t \exp(-t) \PP(X \geq t) dt                                  \\
     & \geq \int_0^\infty t \exp(-B) \PP(X \geq t) dt                               \\
     & = \exp(-B)  \int_0^\infty t \PP(X \geq t) dt = \exp(-B) \frac{\E[X^2]}{2}\,.
\end{align*}

Given that $\E[f(X)] \geq \Var[X]$, (dropping $\exp(-B)$ for now) we have
\begin{align*}

     & \simeq \E[(-\eta \hl - \E[-\eta \hl|A])^2|A]                                             \\

     & =( \tfrac{T_A}{ P_A})^2 \E[(Y - \E[Y|A])^2|A]                                            \\
     & =(\tfrac{T_A}{P_A})^2 \E[(\langle A, \ell\rangle - \E[\langle A, \ell\rangle|A])^2|A]\,,
\end{align*}

\begin{align*}
    \E_{t-1}[\Delta_t]

     & =\E_{t-1}[\langle \bar{A}^b_t - A_\star, \hl_t \rangle]          \\

\begin{align*}

\section{Bandit convex optimization(\bco)}
The losses sequence is a sequence of convex functions
$f_1, f_2, \cdots, f_n: \cA \to \R$.
In each round $t$, the learner chooses an action $A_t \in \cA$,
and observes $Y_t = f_t(A_t) \in \R$.
\subsection{\bco in one dimension}
Suppose that $\cA = [-1, 1]$.
We use the idea of covering + \mdalg on simplex as discussed in
section~\ref{sec:md-negentropy} to solve this setting.

Let $\cC_\epsilon = \{ \alpha\epsilon: \alpha \in \mathbb{Z}, \alpha\epsilon \in [-1, 1]\}$ be an $\epsilon$-covering of $\cA$.
Let
\begin{align*}
    \ell_t =
    \begin{bmatrix}
        f_t(c_1) \\
        f_t(c_2) \\
        \vdots   \\
        f_t(c_n)
    \end{bmatrix}\,, \quad \text{where } c_i \in \cC_\epsilon\,,
\end{align*}
and with that the 1D \bco problem is reduced (with $n\epsilon$ looseness
due to covering) to a \blo problem in $d=\ceil{2/\epsilon}$ dimensions,
with the action space $\cA_\blo = \Delta(\ceil{2/\epsilon})$.

\begin{remark}[\textcolor{red}{$n^{3/4}$} upper-bound for \textit{Lip}-Bandits]
    Naively applying the bound on \mdalg with $\epsilon=\frac{1}{\sqrt{n}}$
    from remark~\ref{rem:md-simplex-explicit} gives an upper-bound of
    $\sqrt{2 n \cdot d} = \sqrt{2 n \cdot \ceil{2/\epsilon}}= n^{3/4}$
    on the \blo reduction, which together with the covering
    error of $\sqrt{n}$ results in a bound of $\tilde{O}(n^{3/4})$.
    This applies to all Lipschitz bandits.
\end{remark}

The main idea to get $\widetilde{O}(\sqrt{n})$ bound is to exploit
the fact that not all $\ell \in [0,1]^d$ are possible in the
\blo-reduction: $\ell_t$ is a discretization of a
convex function.
The result should be a better bound on the sum in
theorem~\ref{thm:md-simplex}.

\begin{remark}[Actions are distributions when $\cA=\Delta(d)$]\label{rem:actions-are-dists}
    The very nice (and weird) thing about running any algorithm
    on the simplex with linear losses is that if it plays $A_t \in \Delta(\cA)$,
    which could be random itself, it can be further
    restricted to play $e_{\omega}$ with $\omega \sim A_t$, without
    changing the \textit{conditional expected per step loss}.
    With that in mind, \mdalg does come with a default randomization
    in the simplex: instead of playing $\fmd(\ell_{1:t-1})$
    play $e_\omega$ with $\omega \sim \fmd(\ell_{1:t-1})$.
\end{remark}

\paragraph{Randomness in the virtual game}
The algorithm knows the estimates $\hl_{1:t-1}$,
so it's technically in the \olo setting in the virtual game,
where it can simply run \mdalg \textbf{with no randomness}---just
take $\hat{a}_t = \fmd(\hl_{1:t-1})$.
However, everything is unchanged if the algorithm
decides to randomize in the \textit{virtual game}.
The algorithm can play a distribution $Q_t \in \Delta(\cC)$
with $\E_{A \sim Q_t}[A] = \hat{a}_t$.
The benefit of this will be clear in a moment.

To emphasize the fact that $\hat{a}_t$ is a distribution
we use $Q_t\triangleq \hat{a}_t$ to be the \textit{virtual policy}:
\begin{align*}
    Q_t(\omega) & = \hat{a}_t^\top e_\omega\,, \forall \omega \in \cC\,,
\end{align*}
and let $K \in \R^{|\cC| \times |\cC|}$ be a probability kernel,
meaning that $K Q_t \in \Delta(\cC)$.
Then we choose $P_t \in \Delta(\cC)$ to be
\begin{align*}
    P_t & = K Q_t\,,
\end{align*}
and note that $A_t \sim P_t$ can be written as
\begin{align*}
    A_t = e_{\alpha_t}\,, \qquad \text{where } \alpha_t \sim P_t\,.
\end{align*}


\paragraph{A chance to make Biased-\smdalg somewhat unbiased:}
So far $\E_t[A_t] \neq \fmd(\hl_{1:t-1})$
necessarily, but we can choose $P_t$
such that they are \textit{jointly} unbiased:
\begin{align*}
    \langle \ell_t, P_t \rangle
     & =
    \E_t\left[\langle \hl_t, e_{\beta_t}\rangle\right]
    =
    \langle \E[\hl_t], Q_t \rangle\,,
\end{align*}
where $\beta_t \sim Q_t$ is the virtual action.
This can be done by setting
\begin{align*}
    \hl_t & = E(e_{\alpha_t}, Y_t)
    = E(e_{\beta_t}, \langle \alpha_t, \ell_t\rangle)
    = \frac{K^\top e_{\alpha_t} \cdot \langle e_{\alpha_t}, \ell_t\rangle}{
        P_t^\top e_{\alpha_t}
    }\,,
\end{align*}
since then
\begin{align*}
    \E_t\left[\hl_t\right]
     & =
    \E_t\left[
        \frac{K^\top e_{\alpha_t} \cdot \langle e_{\alpha_t}, \ell_t\rangle}{
            P_t^\top e_{\alpha_t}}
    \right] \\
     & =
    \sum_{\alpha \in \cC}
    \frac{K^\top e_{\alpha}
        \cdot \langle e_{\alpha}, \ell_t\rangle}{
        P_t^\top e_{\alpha}}
    P_t^\top e_{\alpha}
    \\
     & =
    \sum_{\alpha \in \cC}
    K^\top e_{\alpha}
    \cdot \langle e_{\alpha}, \ell_t\rangle
    \\
     & =
    K^\top \ell_t\,,
\end{align*}
and finally
\begin{align*}
    \langle \ell_t, P_t \rangle
    = \langle \ell_t, K Q_t \rangle
    = \langle K^\top \ell_t, Q_t \rangle
    = \langle \E_t[\hl_t], Q_t \rangle\,.
\end{align*}

\paragraph{A specific choice of $K$---exploiting 1d geometry:}
Let $\bar{A}$ be
\begin{align*}
    \bar{A}_t = \langle P_t, \cC \rangle,
\end{align*}
which should be thought of as the expected action in round $t$.
Let $\Pi_\cC(x) = \argmin_{y \in \cC, |y| \leq |x|} |x - y|$,
Given $I(x,y) = \{z \in \cC: \min(x,y) \leq z \leq \max(x,y)\}$,
let
\begin{align*}
    K_{x,y} & =
    \begin{cases}
        \frac{1}{|I(x,y)|} & \text{if } x\in I(\bar{A}_t, y) \\
        0                  & \text{otherwise}
    \end{cases}\,,
\end{align*}

\begin{lemma}[\cite{lattimore2024bandit}[Lemma 8.3]]
    We have
    \begin{align*}
        \langle \ell_t, P_t - e_{b} \rangle
        \leq 2 \cdot \langle \E_t[\hl_t], Q_t - e_{b} \rangle\,,
    \end{align*}
    and
    \begin{align*}
        \E_t[(e_{\beta_t}^\top \hl_t)^2]
        \leq 2 + \ln(n)
    \end{align*}
\end{lemma}
