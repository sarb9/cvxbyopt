\section{Overview}
\begin{enumerate}
    \item \texttt{OLO} algs solve \texttt{OCO}: $f(x) - f(x_\star) \leq \langle \nabla f(x), x - x_\star \rangle$.
    \item \textbf{Random} algs are \textit{suboptimal} for \texttt{OCO} and \textit{pointless} for \texttt{OLO}.
    \item First order information $\nabla f(x)$ makes \texttt{BCO} as hard as \texttt{OLO}.
    \item \textbf{Random} algs are \textit{necessary} for \texttt{BCO} and \texttt{BLO}.
    \item \textbf{Random} algs are \textit{free} for \texttt{BLO} but appear to be \textit{expensive} ($d^{1.5}$) for \texttt{BCO}.
\end{enumerate}
We study algs for \texttt{BCO} and \texttt{BLO} which are based on Stochastic Mirror Decent(\textsf{SMD}) which means choosing\\
(1) randomization scheme, (2) estimation scheme, (3) potential + $\eta$.

$\star$ Duality with the Bayesian world is curious and helps with the analysis.
\subsection{Notation}
Let $[n] = \{1,2, \cdots, n\}$.
Let $D_F(a, b) = F(a) - F(b) - \nabla F(b)^\top (a - b)$ be the Bregman divergence.
Also let $\cP(A)$ be the set of all probability measures over the set $A$.

\section{Online linear optimization(\texttt{OLO})}
The adversary chooses a sequence of losses $\ell_1, \cdots, \ell_n \in \cL \subseteq \Rd$.
In each round $t \in [n]$, the player chooses an action $a_t \in \cA \subseteq \Rd$, and observes $y_t = \langle a_t, \ell_t \rangle$.
The regret w.r.t to an action $a_\star \in \cA$ is
\begin{align*}
    \Reg_n(a_\star) = \sum_{t=1}^{n} \langle \ell_t, a_t - a_\star \rangle\,.
\end{align*}

\begin{theorem}\label{thm:MD}
    Let $F$ be Legendre with domain $\cD$ such that $\cA \cap \interior(\cD)\neq \emptyset$.
    Then
    \begin{align*}
        \Reg_n(a_\star)
         & \leq
        \sum_{t=1}^{n} \langle \ell_t, a_t - a_{t+1} \rangle
        + \frac{1}{\eta}
        \left(
        D_F(a_\star, a_1) -
        \sum_{t=1}^{n} D_F(a_{t+1}, a_t)
        \right) \\
         & \leq
        \frac{1}{\eta}
        \left(
        D_F(a_\star, a_1) + \sum_{t=1}^{n} D_F(a_{t}, \tilde{a}_{t+1})
        \right) \\
         & =
        \frac{1}{\eta}
        \left(
        D_{F}(a_\star, a_1) + \sum_{t=1}^{n} D_{F^\star}\left(\nabla F(a_t) - \eta \ell_t, \nabla F(a_t)\right)
        \right)\,.
    \end{align*}
\end{theorem}
\begin{proof}
    We have
    \begin{align*}
        \Reg_n(a_\star)
         & = \sum_{t=1}^{n} \langle \ell_t, a_t - a_\star \rangle                                                                                                                                                                          \\
         & = \sum_{t=1}^{n} \underbrace{\langle \ell_t, a_t - a_{t+1} \rangle}_{\triangleq \texttt{st}_t \text{ or stableness}} + \underbrace{\langle \ell_t, a_{t+1} - a_\star \rangle}_{\triangleq \texttt{tr}_t \text{ or tracking}}\,.
    \end{align*}
    What do we know about $a_{t+1}$?
    \begin{align*}
         & a_{t+1} = \argmin_{a \in \cA} \eta \langle \ell_t, a \rangle + D_F(a, a_t)                                                                                                          \\
        \Rightarrow
         & \langle \eta \ell_t + \nabla F(a_{t+1}) - \nabla F(a_t), \tilde{a} - a_{t+1} \rangle \geq 0                                                           & (\forall \tilde{a} \in \cA) \\
        \Rightarrow
         & \texttt{tr}_t = \langle \ell_t, a_{t+1} - a_\star \rangle \leq \frac{1}{\eta} \langle \nabla F(a_{t+1}) - \nabla F(a_t), a_\star - a_{t+1} \rangle\,.
    \end{align*}
    Then there's the nice equality
    \begin{align*}
        \langle \nabla F(a_{t+1}) - \nabla F(a_t), a_\star - a_{t+1} \rangle
         & =
        D_F(a_\star, a_t) - D_F(a_{t+1}, a_t) - D_F(a_\star, a_{t+1})\,,
    \end{align*}
    which gives us a bound on the \textit{tracking} terms
    \begin{align*}
        \eta \sum_{t=1}^{n} \langle \ell_t, a_{t+1} - a_\star \rangle
         & \leq \sum_{t=1}^{n} D_F(a_\star, a_t)  - D_F(a_\star, a_{t+1}) - D_F(a_{t+1}, a_t)                   \\
         & = \sum_{t=1}^{n} D_F(a_\star, a_t)  - D_F(a_\star, a_{t+1}) - D_F(a_{t+1}, a_t)                      \\
         & = D_F(a_\star, a_1) - \underbrace{D_F(a_\star, a_{n+1})}_{\geq 0} - \sum_{t=1}^{n} D_F(a_{t+1}, a_t) \\
         & = D_F(a_\star, a_1) - \sum_{t=1}^{n} D_F(a_{t+1}, a_t)\,.
    \end{align*}
    Without touching the \textit{stableness} term, we have the first part as
    \begin{align*}
        \Reg_n(a_\star)
         & \leq
        \sum_{t=1}^{n} \langle \ell_t, a_t - a_{t+1} \rangle
        + \frac{1}{\eta}
        \left(
        D_F(a_\star, a_1) -
        \sum_{t=1}^{n} D_F(a_{t+1}, a_t)
        \right)\,.
    \end{align*}
    Moreover, $\textsf{MD}$ stores losses in the gradient of $\tilde{a}$,
    \begin{align*}
         & \tilde{a}_{t+1}  = \argmin_{a \in \R^d} \eta \langle \ell_t, a \rangle + D_F(a, a_t) \\
        \Rightarrow
         & \ell_t = \frac{1}{\eta}
        \left(
        \nabla F(a_t) - \nabla F(\tilde{a}_{t+1})
        \right)
    \end{align*}
    which mixes the \textit{stableness} term nicely with $D_F(a_{t+1}, a_t)$, as
    \begin{align*}
        \texttt{st}_t
         & = \langle \ell_t, a_{t} - a_{t+1} \rangle                                                 \\
         & = \frac{1}{\eta} \langle \nabla F(a_t) - \nabla F(\tilde{a}_{t+1}), a_t - a_{t+1} \rangle \\
         & = \frac{1}{\eta} \left(D_F(a_{t+1}, a_t) + D_F(a_t, \tilde{a}_{t+1})\right)
    \end{align*}
    and gives the second part,
    \begin{align*}
        \Reg_n(a_\star)
         & \leq
        \frac{1}{\eta}
        \left(
        D_F(a_\star, a_1) + \sum_{t=1}^{n} D_F(a_{t}, \tilde{a}_{t+1})
        \right)\,.
    \end{align*}
    To see the third part, note that
    \begin{align*}
        D_F(a_{t}, \tilde{a}_{t+1})
         & =D_{F^\star}\left(\nabla F(\tilde{a}_{t+1}), \nabla F(a_t)\right)    \\
         & = D_{F^\star}\left(\nabla F(a_t) - \eta \ell_t, \nabla F(a_t)\right)
    \end{align*}
\end{proof}
\begin{remark}
    \label{rem:ellt-adapted}
    The analysis works even if $\ell_t$ that is adapted to $\ell_1, a_1, \cdots, a_{t-1}$.
\end{remark}

\begin{theorem}[\cite{lattimore2024bandit}, Theorem 8.11]
    \label{thm:md-with-D_star}
    We have
    \begin{align*}
        \sum_{t=1}^{n} \langle a_t - a_\star, \ell_t \rangle
        \leq \frac{1}{\eta}
        \left(
        D_F(a_\star, a_1) + \sum_{t=1}^{n} \cS_t(\eta \ell_t)
        \right)\,,
    \end{align*}
    where $\cS_t(u) = D_{F^\star}(R'(q_t)- u , R'(q_t))$.
\end{theorem}
\begin{proof}
    Starting from the second part of
    \begin{align*}
        \Reg_n(a_\star)
         & \leq
        \sum_{t=1}^{n} \langle \ell_t, a_t - a_{t+1} \rangle
        + \frac{1}{\eta}
        \left(
        D_F(a_\star, a_1) -
        \sum_{t=1}^{n} D_F(a_{t+1}, a_t)
        \right) \\
         & \leq
        \frac{1}{\eta}
        \left(
        D_F(a_\star, a_1) + \sum_{t=1}^{n} D_F(a_{t}, \tilde{a}_{t+1})
        \right)\,.
    \end{align*}
\end{proof}

\subsection{Stochastic mirror decent(\smdalg)}
\begin{algorithm}[h!]
    \begin{minipage}{12cm}
        \begin{mdframed}
            \begin{lstlisting}
args: $\eta \in \R$, $F$
$a_1 = \argmin_{a \in \cA} F(a)$
$\ta_1 = \argmin_{a \in \Rd} F(a)$
for $t = 1$ to $n$:
    Choose $P_t \in \Delta(\cA)$, s.t. $\E_{A \sim P_t}[A] = a_t$
    Play $A_t \sim P_t$ and observe $Y_t$
    Compute $\hat{\ell}_t = E_t(A_t, Y_t)$
    $a_{t+1} = \argmin_{a \in \cA} \eta \langle a, \hat{\ell}_t\rangle + F(a)$
    $\ta_{t+1} = \argmin_{a \in \Rd} \eta \langle a, \hat{\ell}_t\rangle + F(a)$
\end{lstlisting}
            \caption{\smdalg}\label{alg:SMD}
        \end{mdframed}
    \end{minipage}
\end{algorithm}

Instead of choosing an action $a_t \in \cA$, \smdalg chooses an
\textit{exploration} distribution $P_t \in \cP(\cA)$
\textbf{with mean} $a_t$ in round $t \in [n]$.
Next, \smdalg pretends that $\hat{\ell}$ which is an \textit{estimate} of $\ell_t$ is the actual loss.
\textbf{In the most general form}, this estimate can be made through a function
$E_t: \cA \times \R \to \Rd$ in each round $t \in [n]$
such that $\hat{\ell}_t = E_t(A_t, Y_t)$.
Since losses are linear in \blo, there is
\begin{align*}
    \texttt{reg}_t : \E_t[\langle A_t, \ell \rangle| a_t] = \langle a_t, \ell\rangle\,,
\end{align*}
which means \smdalg and \mdalg have the \textbf{same expected loss} given
the \mdalg action $a_t$, and they only differ through the losses they use:
\mdalg uses $\ell_t$, but \smdalg uses $\hat{\ell}_t$.

Since $a_t$ depends on the random estimates $\hat{\ell}_{1:t-1}$,
we use $\bar{A}_t$ as in the following theorem.
\begin{theorem}[Unbiased-\smdalg]\label{thm:SMD-unbiased}
    Suppose $\E[\hat{\ell}_t|\bar{A}_t] = \ell_t$ (unbiased estimator).
    The following results continue to hold
    \begin{align*}
        \Reg_n(a_\star)
         & \leq
        \frac{D_F(a_\star, \bar{A}_1)}{\eta}
        +
        \E\left[
            \sum_{t=1}^{n}
            \left(
            \langle \hat{\ell}_t, \bar{A}_t - \bar{A}_{t+1} \rangle
            - \frac{1}{\eta}
            D_F(\bar{A}_{t+1}, \bar{A}_t)
            \right)
        \right] \\
         & \leq
        \frac{1}{\eta}\left(
        D_F(a_\star, \bar{A}_1)
        +
        \E\left[
            \sum_{t=1}^{n} D_F(\bar{A}_{t}, \tilde{A}_{t+1})
            \right]
        \right)\,.
    \end{align*}
\end{theorem}
\begin{proof}
    The proof follows simply from
    \begin{align*}
        \Reg_n(a_\star)
        =
        \E\left[
            \sum_{t=1}^{n} \langle A_t - a_\star, \ell_t\rangle
            \right]
        =
        \E\left[
            \sum_{t=1}^{n} \langle \bar{A}_t - a_\star, \hat{\ell}_t\rangle
            \right]\,,
    \end{align*}
    and doing the same steps for the random game in the expectation and Remark~\ref{rem:ellt-adapted}.
\end{proof}

\subsubsection{\smdalg in the shade}
Suppose there are two sampling rules $P_t$ and $P_t'$ where
$\E_{A \sim P_t}[A] = \E_{A \sim P_t'}[A]$, and \smdalg can choose
from them for sampling $A_t$ at round $t$.
In \olo, these two \textit{result in the exact same loss} in round $t$,
and the reason for prefering one over the other is \textit{better exploration}.
The situation is much different for \bco.

This observation hints us that $P_t$ can be chosen as a function of $\bar{A}_t$:
amongst all distributions $P$ that have the mean $\bar{A}_t$ choose the one that
explores the best.
Better exploration means having better $\hat{\ell}_{t}$, and $\hat{\ell}_t$ is a
random vector that depends on both $P_t$ and $E_t$;
in this sense the best sampling (exploration) rule depends on the current
estimation scheme $E_t$, and vice-versa.

The \smdalg can be tweaked in two ways.
\begin{enumerate}
    \item The estimates are biased $\E_t[\hat{\ell}_t] \neq \ell_t$,
    \item The actions are biased $\E_{A \sim P_t}[A_t] \neq a_t$,
\end{enumerate}


The algorithm can \textit{pretend} the actual sequence of losses is
$g_1, \cdots, g_n$, and account for the difference of this sequence and
the actual sequence $\ell_1, \cdots, \ell_n$.
The regret would be
\begin{align*}
    \Reg_n(a_\star)
     & = \E\left[
        \sum_{t=1}^{n} \langle A_t - a_\star \rangle
    \right]       \\
     & = \E\left[
        \sum_{t=1}^{n} \langle P_t - a_\star \rangle
    \right]       \\
     & = \E\left[
        \sum_{t=1}^{n} \langle \ell_t - g_t, A_t - a_\star \rangle
        + \langle g_t, A_t - a_\star \rangle
    \right]       \\
\end{align*}


\begin{theorem}[Biased-\smdalg]\label{thm:SMD-unbiased}
    The expected regret of \smdalg is bounded as
    \begin{align*}
        \Reg_n(a_\star)
         & \leq
        \frac{D_F(a_\star, \bar{A}_1)}{\eta}
        +
        \E\left[
            \sum_{t=1}^{n}
            \left(
            \langle \hat{\ell}_t, \bar{A}_t - \bar{A}_{t+1} \rangle
            - \frac{1}{\eta}
            D_F(\bar{A}_{t+1}, \bar{A}_t)
            \right)
            \right]
        \\
         & \leq
        \frac{1}{\eta}\left(
        D_F(a_\star, \bar{A}_1)
        +
        \E\left[
            \sum_{t=1}^{n} D_F(\bar{A}_{t}, \tilde{A}_{t+1})
            \right]
        \right)\,.
    \end{align*}
\end{theorem}
\begin{proof}
    The proof follows simply from
    \begin{align*}
        \Reg_n(a_\star)
        =
        \E\left[
            \sum_{t=1}^{n} \langle A_t - a_\star, \ell_t\rangle
            \right]
        =
        \E\left[
            \sum_{t=1}^{n} \langle \bar{A}_t - a_\star, \hat{\ell}_t\rangle
            \right]\,,
    \end{align*}
    and doing the same steps for the random game in the expectation and Remark~\ref{rem:ellt-adapted}.
\end{proof}

            \caption{OSMD}\label{alg:OSMD}

\citet{zimmert2019connections} suppose that (with simplifications)
\begin{enumerate}
    \item The exploration is done through a collection
          $P = \{P_a: a \in \cA\}$ such that $P_a \in \Delta(\cA)$ and
          $\E_{A \sim P_a}[A] = a$.
    \item The estimation is done through a function
          $E: \cA \times \cA \times \R \to \R^d$ such that
          \begin{align*}
              \E_{A \sim P_a}\left[
                  E(a, A, \underbrace{\langle \ell_t, A\rangle}_{Y_t})
                  \right] = \ell_t\,.
          \end{align*}
    \item Let $\texttt{MD}(\ell, b; \eta) = \argmin_{a \in \cA} \eta \langle \ell, a \rangle + D_F(a, b)$.
\end{enumerate}
The stability term is what inside the
expectation in Theorem~\ref{thm:SMD},
\begin{align*}
    \texttt{stab}_t(a) & \triangleq
    \E_{A_t \sim P_a}\left[
        \langle
        \hat{\ell}_t,
        a - \bar{A}_{t+1}
        \rangle
        - \frac{D_F(\bar{A}_{t+1}, a)}{\eta}
        \right]
    \\
                       & =
    \E_{A_t \sim P_a}\left[
        \langle E(a, A_t, Y_t), a - \bar{A}_{t+1} \rangle
        - \frac{D_F(\bar{A}_{t+1}, a)}{\eta}
        \right]\,,
\end{align*}
where $\bar{A}_{t+1} = \texttt{MD}(\hat{\ell}_t, \bar{A}_t)$,
which means the loss comes from $A_t$ but the `anchor' comes from $\bar{A}_t$.

\paragraph{Bayesian:}
Suppose that $(\ell_1, \ell_2, \cdots, \ell_n) \sim \nu$,
and $A_\star = \argmin_{a \in \cA} \sum_{t=1}^{n} \langle \ell_t, a\rangle$ is random.
The \textsf{MTS} \textit{Bayesian} algorithm just does one thing differently:
\begin{align*}
    \bar{A}^b_{t+1} = \E_{t}[A_\star]\,,
\end{align*}
so instead of \textsf{MD} update, it uses the expected optimal action.

The crucial connection is made through the fact that
\begin{align*}
    \E_{t-1}[A_t] = \E_{t-1}[A_\star]\,.
\end{align*}


\begin{algorithm}[h!]
    \begin{minipage}{8cm}
        \begin{mdframed}
            \begin{lstlisting}
args: Prior $\nu$, and $P$
$A_1^b = \E[A_\star]$
for $t = 1$ to $n$:
    Play $A_t \sim P_{A^b_t}$.
    Update $\bar{A}^b_{t+1} = \E_{t}[A_\star]$.
\end{lstlisting}
            \caption{MTS}\label{alg:MTS}
        \end{mdframed}
    \end{minipage}
\end{algorithm}

Beginning with the definition of the per-step Bayesian regret,
\begin{align*}
    \E_{t-1}[\Delta_t]
     & =\E_{t-1}[\langle A_t - A_\star, \ell_t \rangle]                        \\
     & =\E_{t-1}[\langle \bar{A}^b_t - A_\star, \hat{\ell}_t \rangle]          \\
     & = \E_{t-1}[\langle \bar{A}^b_t - \bar{A}^b_{t+1}, \hat{\ell}_t \rangle]
\end{align*}
Let $A_t \sim P_{\bar{A}^b_t}$, and recall the $\textsf{MD}$ update
\begin{align*}
    \texttt{MD}_t \triangleq \texttt{MD}(\hat{\ell}_t, \bar{A}^b_t) = \argmin_{a \in \cA} \eta \langle \underbrace{E(\bar{A}^b_t, A_t, Y_t)}_{\hat{\ell}_t}, a \rangle + D_F(a, \bar{A}^b_t)\,,
\end{align*}
so
\begin{align*}
     & \langle \hat{\ell}_t, \texttt{MD}_t \rangle + \frac{1}{\eta} D_F(\texttt{MD}_t, \bar{A}_t^b)
    \leq
    \langle \hat{\ell}_t, \bar{A}^b_{t+1} \rangle + \frac{1}{\eta}D_F(\bar{A}^b_{t+1}, \bar{A}_t^b)                 \\
    \Rightarrow
     & \langle \hat{\ell}_t, \texttt{MD}_t - \bar{A}_{t}^b \rangle + \frac{1}{\eta} D_F(\texttt{MD}_t, \bar{A}_t^b)
    \leq
    \langle \hat{\ell}_t, \bar{A}^b_{t+1} - \bar{A}_{t}^b \rangle + \frac{1}{\eta}D_F(\bar{A}^b_{t+1}, \bar{A}_t^b) \\
    \Rightarrow
     & \langle \hat{\ell}_t, \bar{A}_{t}^b - \bar{A}^b_{t+1} \rangle
    \leq
    \langle \hat{\ell}_t, \bar{A}_{t}^b - \texttt{MD}_t \rangle
    +
    \frac{1}{\eta}D_F(\bar{A}^b_{t+1}, \bar{A}_t^b)
    - \frac{1}{\eta} D_F(\texttt{MD}_t, \bar{A}_t^b)\,,
\end{align*}
which gives
\begin{align*}
    \E_{t-1}[\Delta_t]
     & = \E_{t-1}[\langle \bar{A}^b_t - \bar{A}^b_{t+1}, \hat{\ell}_t \rangle] \\
     & \leq \E_{t-1}\left[
    \underbrace{
        \langle \hat{\ell}_t, \bar{A}_{t}^b - \texttt{MD}_t \rangle
        - \frac{1}{\eta} D_F(\texttt{MD}_t, \bar{A}_t^b)}_{
        \texttt{stab}_t(\bar{A}_t^b; \eta)
    }
    +\frac{1}{\eta}D_F(\bar{A}^b_{t+1}, \bar{A}_t^b)
    \right]                                          \,,                       \\
\end{align*}
which means bounding the $\texttt{stab}_t(a, \eta)$ gives a regret bound for the Bayesian regret
since the sum of divergences is bounded by \citet{lattimore2019information}[Thm 3].

\subsubsection{\smdalg with negentropy}
Fix $F(a): \R^d \to \R$ to be the negentropy potential,
\begin{align*}
    F(a) = \langle a, \ln(a) - 1 \rangle\,.
\end{align*}
Then it's easy to check that
\begin{align}
    \ta_{t+1} = a_t \odot \exp(-\eta \hat{\ell}_t)
\end{align}

\begin{theorem}[\cite{lattimore2024bandit}, Theorem 8.14]
    \label{thm:exp-imp-regret}
    Let $x_\star = \argmin_{x \in \cC} \sum_{t=1}^{n} f_t(x)$ and
    $p_\star \in \Delta(\cC)$ be a Dirac on $x_\star$.
    The expected regret of \cref{alg:exp-imp} is bounded by
    \begin{align*}
        \E\left[\Reg_n(x_\star)\right]
        = \frac{\log(|\cC|)}{\eta} +
        \sum_{t=1}^{n} \E\left[
            \langle p_t - p_\star, f_t \rangle
            + \langle p_\star - q_t, \hat{s}_t \rangle
            + \frac{1}{\eta} \cS_t(\eta \hat{s}_t)
            \right]\,.
    \end{align*}
\end{theorem}
\begin{proof}
    We have
    \begin{align*}
        \E\left[\Reg_n(x_\star)\right]
         & =
        \E\left[
            \sum_{t=1}^{n} \langle p_t - p_\star, f_t \rangle
        \right] \\
         & =
        \E\left[
        \sum_{t=1}^{n}
        \underbrace{\langle p_t - p_\star, f_t \rangle}_{\text{actual}}
        - \underbrace{\langle q_t - p_\star, \hat{s}_t \rangle}_{\text{shadow}}
        + \langle q_t - p_\star, \hat{s}_t \rangle
        \right] \\
         & \leq
        \frac{\log(|\cC|)}{\eta}+
        \E\left[
            \sum_{t=1}^{n}
            \langle p_t - p_\star, f_t \rangle
            - \langle q_t - p_\star, \hat{s}_t \rangle
            + \frac{1}{\eta} \cS_t(\eta \hat{s}_t)
            \right]
    \end{align*}
    where the last inequality follows from \cref{thm:exp-imp}.
\end{proof}

\subsection{Exploration By Optimisation}
The idea is to bound the term inside the expectation in \cref{thm:exp-imp-regret} uniformly over all $t \in [n]$.
For the learner, the degree of freedom is in the choice of the estimator $\hat{s}_t$ and the exploration distribution $p_t$.
To this end, define
\begin{itemize}
    \item $\cG$ be the set of all functions $g:\cC \to \R$, i.e., $\cG = \R^{|\cC|}$ (all loss vectors),
    \item $\cE$ be the set of all functions $E: \cC \times \R \to \cG$ (choice of $\hat{s}$),
    \item
          $\Lambda_\eta(q, p, E, r, f) =
              \frac{1}{\eta}
              \E_{X\sim p}\!\left[
                  \langle p - r, f \rangle
                  + \langle r - q, E(X, Y) \rangle
                  + \frac{1}{\eta} \cS_q(\eta E(X, Y))
                  \right]$,
\end{itemize}
where $q \in \Delta(\cC)$ is the exponential weights distribution at time $t$, $p \in \Delta(\cC)$ is the exploration distribution, $E \in \cE$ is the estimator, $r \in \Delta(\cC)$ is the Dirac on the best action, and $f \in \cF_b$ is the loss function.
Note that the randomness is only through $X$ and $Y$, where $X \sim p$ and $Y = f(X)$.
% Therefore, fix $t \in [n]$ and define
% \begin{align*}
%     \Lambda_\eta(q) & =
%     \inf_{p \in \Delta(\cC), E \in \cE}
%     \sup_{r \in \Delta(\cC), f \in \cF_b} \frac{1}{\eta}
%     \Lambda_\eta(q, p, E, r, f)\,.
% \end{align*}
\begin{remark}
    There is a bit of waste later since we suppose that $r \in \Delta(\cC)$, but it could've been restricted to Dirac distributions.
\end{remark}

\begin{algorithm}[h!]
    \begin{minipage}{12cm}
        \begin{mdframed}
            \begin{lstlisting}
args: learning rate $\eta >0$, precision $\epsilon > 0$
let $\cC \in K$ be finite
for $t = 1$ to $n$:
    compute $q_t(x) = \frac{\exp(-\eta\sum_{u=1}^{t-1}\hat{s}_u(x))}{\sum_{y\in\cC}\exp(-\eta\sum_{u=1}^{t-1}\hat{s}_u(x))}$ for all $x \in \cC$
    find distribution $p_t \in \Delta(\cC)$ and $E_t \in \cE$ such that
        $\Lambda_\eta(q_t, p_t, E_t) \leq \inf_{p,E} \Lambda_\eta(q_t) + \epsilon$
    sample $X_t \sim p_t$, and observe $Y_t = f_t(X_t)$
    compute $\hat{s}_t(x) = E_t(X_t, Y_t)$
\end{lstlisting}
            \caption{Exploration by Optimisation}\label{alg:exp-by-opt}
        \end{mdframed}
    \end{minipage}
\end{algorithm}
It's worth noting that \cref{alg:exp-by-opt} is a instantiation of \cref{alg:exp-imp} where the estimator $\hat{s}_t$ and the exploration distribution $p_t$ are chosen by solving an optimisation problem.

\begin{theorem}[\cite{lattimore2024bandit}, Theorem 8.15] \label{thm:exp-by-opt-regret}
    The expected regret of \cref{alg:exp-by-opt} is bounded by
    \begin{align*}
        \E\left[
            \Reg_n(x_\star)
            \right]
         & \leq
        \frac{\log(|\cC|)}{\eta}
        +
        n\eta \sup_{q \in \Delta(\cC)} \Lambda_\eta(q) + n\eta\epsilon\,.
    \end{align*}
\end{theorem}

Therefore, from \cref{thm:exp-by-opt-regret}, to bound the regret of \cref{alg:exp-by-opt} it suffices to bound
\begin{align}
    \Lambda^\star_\cC =
    \sup_{\eta > 0, q \in \Delta(\cC)}
    \inf_{p \in \Delta(\cC), E \in \cE}
    \sup_{y \in \cC, f \in \cF_b}
    \Lambda_\eta(q, p, E, y, f)\,.
    \label{eq:exp-by-opt-saddle}
\end{align}

The following lemma shows that the order of the infimum and supremum can be interchanged.
\begin{lemma}
    \label{lem:exp-by-opt-minimax-wrong}
    For all $\eta > 0$ and $q \in \Delta(\cC)$,  we have
    \begin{align*}
        \inf_{p \in \Delta(\cC), E \in \cE}
        \sup_{r \in \Delta(\cC), f \in \cF_b}
        \Lambda_\eta(q, p, E, r, f)
        =
        \sup_{r \in \Delta(\cC), f \in \cF_b}
        \inf_{p \in \Delta(\cC), E \in \cE}
        \Lambda_\eta(q, p, E, r, f)\,.
    \end{align*}
\end{lemma}
This lemma is \textcolor{red}{wrong} though... The minimax theorem only applies if the sup player should can choose from a joint distribution over $r$ and $f$.

Let $\nu \in \Delta(\Delta(\cC) \times \cF_b)$ be a joint distribution over $p_\star \in \Delta(\cC)$ and $f \in \cF_b$.
\textit{$p_\star$ is the same thing as $r$...}
Define the bayesian version of $\Lambda_\eta(q, p, E, \nu)$ as
\begin{align*}
    \bar{\Lambda}_\eta(q, p, E, \nu) & :=
    \frac{1}{\eta}
    \E_{X \sim p, (p_\star, f) \sim \nu}\!\left[
        \langle p_\star - q, E(X, Y) \rangle
        + \langle p_\star - p, f \rangle
        + \frac{1}{\eta} \cS_q(\eta E(X, Y))
        \right]\,.
\end{align*}
It's easy to see that
\begin{align*}
    \inf_{p \in \Delta(\cC), E \in \cE}
    \sup_{r \in \Delta(\cC), f \in \cF_b}
    \Lambda_\eta(q, p, E, r, f)
     & \leq
    \inf_{p \in \Delta(\cC), E \in \cE}
    \sup_{\nu \in \Delta(\Delta(\cC) \times \cF_b)}
    \bar{\Lambda}_\eta(q, p, E, \nu)\,.
\end{align*}
\begin{lemma}[\cite{lattimore2024bandit}, Lemma 8.21]
    \label{lem:exp-by-opt-minimax}
    For all $\eta > 0$ and $q \in \Delta(\cC)$,  we have
    \begin{align*}
        \inf_{p \in \Delta(\cC), E \in \cE}
        \sup_{\nu \in \Delta(\Delta(\cC) \times \cF_b)}
        \bar{\Lambda}_\eta(q, p, E, \nu)
         & =
        \sup_{\nu \in \Delta(\Delta(\cC) \times \cF_b)}
        \inf_{p \in \Delta(\cC), E \in \cE}
        \bar{\Lambda}_\eta(q, p, E, \nu)\,.
    \end{align*}
\end{lemma}
In the following two subsections we will solve for the $\inf$ in the RHS of \cref{lem:exp-by-opt-minimax}. In the following $q\in \Delta(\cC)$ is fixed and we will drop it from the notation.

\subsection{Simplifying $\Lambda_\eta$}
\begin{remark}
    We specialize to $R$ being the unnormalised negentropy function, so that we can make the connection to the KL term in IDS.
\end{remark}
Note that for $a \in \R^{|\cC|}$ we have
\begin{align*}
    R^\star(a) = \sum_{x \in \cC} \exp(a(x)) = \langle \exp(a) , 1_\cC \rangle \,,
\end{align*}
and $R'(p) = \ln(p)$, which means that
\begin{align*}
    R^\star(R'(p)) = \sum_{x \in \cC} p(x) = 1\,.
\end{align*}
Suppose that $X \sim p$, $Y = f(X)$, and $(p_\star, f) \sim \nu$, and define
\begin{align*}
    \Delta(\nu, p) & = \E\!\left[
        \langle p_\star - p, f \rangle
        \right]\,,
\end{align*}
Then we can write
\begin{align*}
     & \eta \Lambda_\eta(q, p, E, \nu)
    \\
     & =
    \E\!\left[
        \langle p_\star - q, E(X, Y) \rangle
        + \langle p_\star - p, f \rangle
        + \frac{1}{\eta} \cS_q(\eta E(X, Y))
        \right]
    \\
     & =
    \Delta(\nu, p) +
    \E\!\left[
        \langle p_\star - q, E(X, Y) \rangle
        + \frac{1}{\eta}
        \left(
        R^\star\left(R'(q) - \eta E(X, Y)\right)
        - R^\star(R'(q))
        - \nabla R^\star(R'(q))^\top (- \eta E(X, Y))
        \right)
        \right]
    \\
     & =
    \Delta(\nu, p) +
    \E\!\left[
        \langle p_\star - q, E(X, Y) \rangle
        + \frac{1}{\eta}
        \left(
        R^\star\left(\ln(q) - \eta E(X, Y)\right)
        - 1
        + \eta \langle q, E(X, Y) \rangle
        \right)
        \right]
    \\
     & =
    \Delta(\nu, p) +
    \E\!\left[
        \langle p_\star, E(X, Y) \rangle
        + \frac{1}{\eta} R^\star\left(\ln(q) - \eta E(X, Y)\right)
        \right]
    - \frac{1}{\eta}
    \\
     & =
    \Delta(\nu, p) +
    \E\!\left[
        \langle p_\star, E(X, Y) \rangle
        + \frac{1}{\eta} \left\langle\exp(\ln(q) - \eta E(X, Y)), 1_\cC\right\rangle
        \right]
    - \frac{1}{\eta}
    \\
     & =
    \Delta(\nu, p) +
    \E\!\left[
        \langle p_\star, E(X, Y) \rangle
        + \frac{1}{\eta}
        \langle q, \exp(- \eta E(X, Y))\rangle
        \right]
    - \frac{1}{\eta}
\end{align*}

\subsection{Unrestricted estimator class $\cE$}
The minimizer $E$ in the general case has the following close form.
\begin{lemma}
    Given $\nu \in \Delta(\Delta(\cC), \cF_b)$, and $p \in \Delta(\cC)$
    define $G_{\nu, p} \in \cE$ as
    \begin{align*}
        G_{\nu, p}(x, y) & =
        \frac{1}{\eta}\left(
        R'(q) - R'(\E[p_\star|f(x) = y])
        \right)\,,
    \end{align*}
    then we have
    $
        \inf_{E \in \cE}
        \Lambda_\eta(p, E, \nu)
        =
        \Lambda_\eta(p, G_{\nu, p}, \nu)\,.
    $
\end{lemma}
\begin{proof}
    By taking the derivative of $\Lambda_\eta(p, E, \nu)$ with respect to $E(x,y)$ we have
    \begin{align*}
         & \nabla_{E(x,y)}
        \Lambda_\eta(p, E, \nu)
        \\
         & =
        \tfrac{\PP(f(x)=y,X=x)}{\eta}
        \nabla_{E(x,y)}
        \E\!\left[
            \langle p_\star, E(X, Y) \rangle
            + \tfrac{1}{\eta}
            \langle q, \exp(- \eta E(X, Y))\rangle
            |
            X = x, f(x) = y
            \right]
        \\
         & =
        \tfrac{\PP(f(x)=y,X=x)}{\eta}
        \E\!\left[
            p_\star
            -
            q \odot \exp(- \eta E(X, Y))
            |
            X = x, f(x) = y
            \right]
        \\
         & =
        \tfrac{\PP(f(x)=y,X=x)}{\eta}
        \E\!\left[
            p_\star
            |
            X = x, f(x) = y
            \right]
        -
        \tfrac{1}{\eta}
        q \odot \exp(- \eta E(x, y))\,.
    \end{align*}
    By setting the gradient to zero, for any $z \in \cC$ we have
    \begin{align*}
        \E\!\left[
            p_\star
            |
            X = x, f(x) = y
            \right]_z
         & =
        q_z \exp(- \eta E(x, y))_z\,,
    \end{align*}
    which gives
    \begin{align*}
        E(x, y)_z
         & =
        -\frac{1}{\eta} \ln\left(
        \frac{\E\!\left[p_\star|X = x, f(x) = y\right]_z}{q_z}
        \right)
        \\
         & =
        \frac{1}{\eta}\left(
        \ln(q_z)-
        \ln\left(
            \E\!\left[p_\star|X = x, f(x) = y\right]_z
            \right)
        \right)\,,
    \end{align*}
    which, since $R'(q_z) = \ln(q_z)$ gives the desired result.
\end{proof}

\subsection{IS estimator class--multiplicative in $y$}
Restrict $E(x,y)$ to be of the form
\begin{align*}
    E(x,y) & = \frac{y}{p(x)} T(x) \quad \text{or} \quad \left( = \frac{y}{p(x)}T(x|\cdot) \right)\,.
\end{align*}
with $T: \cC \to \R^{|\cC|}$, and $p\in\Delta(\cC)$.
Let $\nu \in \Delta(\Delta(\cC) \times \cF_b)$.
We are interested in the $T$ that minimizes
\begin{align*}
    \eta \bar \Lambda(T)
     & :=
    \Delta(\nu, p) +
    \E\!\left[
        \langle p_\star, E(X, Y) \rangle
        + \frac{1}{\eta}
        \langle q, \exp(- \eta E(X, Y))\rangle
        \right]
    - \frac{1}{\eta}
    \\
     & :=
    \Delta(\nu, p) +
    \E\!\left[
        \E\left[
            \langle p_\star, \tfrac{T(X) Y}{p(X)} \rangle
            + \frac{1}{\eta}
            \langle q, \exp(- \eta \tfrac{T(X) Y}{p(X)})\rangle
            \bigg| X
            \right]
        \right]
    - \frac{1}{\eta}\,.
\end{align*}
Note that we drop it from the notation, but it's important to remember that $f, p_\star, X$ and $Y$ are random.
Define the shorthand $T_x = T(x) \in \R^{|\cC|}$.
Then through differentiation we have
\begin{align*}
    \nabla_{T_x} \bar{\Lambda}(T)
     & = \tfrac{p(x)}{\eta}\nabla_{T_x} \E\!\left[
        \langle p_\star, \tfrac{T_x Y}{p(X)} \rangle
        + \frac{1}{\eta}
        \langle q, \exp(- \eta \tfrac{T_x Y}{p(X)})\rangle
        | X = x
        \right]
    \\
     & = \tfrac{p(x)}{\eta} \E\!\left[
        \tfrac{Y}{p(X)} p_\star
        - \tfrac{Y}{p(X)} q \odot \exp(- \tfrac{\eta Y}{p(X)} T_X)
        | X = x
        \right]
    \\
     & = \tfrac{1}{\eta} \E\!\left[
        Y \left(
        p_\star
        - q \odot \exp(- \tfrac{\eta Y}{p(X)} T_X)
        \right)
        | X = x
        \right]\,,
\end{align*}
where the expectation is over $(p_\star, f) \sim \nu$.
We want to solve for $T_x$ such that the gradient is zero, so for coordinate $z \in \cC$ we have
\begin{align}
    q(z) \E\!\left[
        f(X)\exp(- \tfrac{\eta f(X)}{p(X)} T(X)_z)
        | X = x
        \right]
     & =
    \E\!\left[
        f(X) p_\star(z)
        | X = x
        \right]
    \label{eq:zero-grad-element-wise}
    \\
    \Rightarrow
    q \odot \E\!\left[
        f(X)\exp(- \tfrac{\eta f(X)}{p(X)} T(X))
        | X = x
        \right]
     & =
    \E\!\left[
        f(X) p_\star
        | X = x
        \right]
    \label{eq:zero-grad-vector}
\end{align}
\begin{remark}[Negative $T(x)_z$]
    If $p_\star$ is a Dirac on $z$ almost surely, then the RHS in \cref{eq:zero-grad} is $\E[f(X)|X=x]$,
    while if $T(x)_z \geq 0$ then the LHS is bounded by
    \begin{align*}
        q(z)
        \E\!\left[
            f(X)\exp(- \tfrac{\eta f(X)}{p(X)} T(X)_z)
            | X = x
            \right]
        \leq
        q(z)
        \E\!\left[f(X)| X = x\right]\,.
    \end{align*}
    So, if $T(x) \geq 0$, the gradient can't be zero when $q(z)< 1$.
\end{remark}
\begin{remark}
    The LHS is the first derivative of the moment generating function of $f(X)|X=x$ evaluated at $-\frac{\eta}{p(X)} T(x)_z$.
\end{remark}
\paragraph{$\bullet$ Closed form for $T(x)_z$?}
It's unlikely that $T(x)_z$ has a closed form.
However, it might be possible to use \cref{eq:zero-grad-vector} to simplify $\bar{\Lambda}(T)$:
\begin{align}
    \eta\bar{\Lambda}(T)
     & =
    \Delta(\nu, p) +
    \E\!\left[
        \langle p_\star, E(X, Y) \rangle
        + \frac{1}{\eta}
        \langle q, \exp(- \eta E(X, Y))\rangle
        \right]
    - \frac{1}{\eta}
    \nonumber
    \\
     & =
    \Delta(\nu, p) +
    \E\!\left[
        \langle p_\star, \tfrac{T(X) Y}{p(X)} \rangle
        + \frac{1}{\eta}
        \langle q, \exp(- \eta \tfrac{T(X) Y}{p(X)})\rangle
        \right]
    - \frac{1}{\eta}
    \nonumber
    \\
     & =
    \Delta(\nu, p) +
    \E\!\left[
        \langle f(X) p_\star, \tfrac{T(X)}{p(X)} \rangle
        + \frac{1}{\eta}
        \langle q, \exp(- \eta \tfrac{T(X) Y}{p(X)})\rangle
        \right]
    - \frac{1}{\eta}\,,
    \label{eq:checkpiont1}
\end{align}
For the first term, we can use \cref{eq:zero-grad-vector} to get
\begin{align*}
    \E\!\left[
        \langle f(X) p_\star, \tfrac{T(X)}{p(X)} \rangle
        | X =x
        \right]
     & =
    \left\langle
    \E\!\left[
        f(X) p_\star
        | X =x
        \right]
    , \tfrac{T(x)}{p(x)}
    \right\rangle
    \\
     & =
    \left\langle
    q \odot
    \E\!\left[
        f(X)\exp(- \tfrac{\eta f(X)}{p(X)} T(X))
        | X =x
        \right]
    , \tfrac{T(x)}{p(x)}
    \right\rangle
    \\
     & =
    \E\!\left[
        \left\langle
        q \odot
        f(X)\exp(- \tfrac{\eta f(X)}{p(X)} T(X))
        , \tfrac{T(x)}{p(x)}
        \right\rangle
        | X =x\right]
    \\
     & =
    \E\!\left[
        \left\langle
        q \odot
        \exp(- \tfrac{\eta f(X)}{p(X)} T(X))
        , \tfrac{T(x)}{p(x)} f(X)
        \right\rangle
        | X =x\right]
\end{align*}
putting this back into the expectation in \cref{eq:checkpiont1} gives
\begin{align*}
      & \E\!\left[
        \langle f(X) p_\star, \tfrac{T(X)}{p(X)} \rangle
        + \frac{1}{\eta}
        \langle q, \exp(- \eta \tfrac{T(X) Y}{p(X)})\rangle
        \right]
    \\
    = &
    \E\!\left[
        \left\langle
        q \odot
        \exp(- \tfrac{\eta f(X)}{p(X)} T(X))
        , \tfrac{T(X)}{p(X)} f(X)
        \right\rangle
        +
        \langle q \odot \exp(- \tfrac{\eta T(X) Y}{p(X)}),  \tfrac{1}{\eta} 1_\cC\rangle
        \right]
    \\
    = &
    \E\!\left[
        \left\langle
        q
        \odot
        \exp(- \tfrac{\eta T(X) Y}{p(X)})
        , Y \tfrac{T(X)}{p(X)} + \tfrac{1}{\eta} 1_\cC
        \right\rangle
        \right]\,,
\end{align*}
which makes the whole thing equal to
\begin{align*}
    \bar{\Lambda}(T)
     & =
    \eta^{-1}
    \Delta(\nu, p) +
    \eta^{-2}
    \left(
    \E\!\left[
            \left\langle
            q
            \odot
            \exp(- \tfrac{\eta T(X) Y}{p(X)})
            , \tfrac{\eta T(X)Y}{p(X)} +  1_\cC
            \right\rangle
            \right]
    -1
    \right)\,.
\end{align*}
Recall that $q$ was fixed through the minimax step, so we still have
\begin{align*}

\subsection{IS estimator--linear in $y$}
Restrict $E(x,y)$ to be of the form
\begin{align*}
    E(x,y) & = \frac{y}{p(x)} T(x) + b(x)\,,
\end{align*}
which includes all importance sampling estimators that are \textbf{linear} in $y$.
We are interested in the $T$ that minimizes
\begin{align*}
    \eta \bar \Lambda(T, b) & :=
    \Delta(\nu, p) +
    \E\!\left[
        \E\left[
            \langle p_\star, \tfrac{T(X) Y}{p(X)} + b(X)\rangle
            + \frac{1}{\eta}
            \langle q, \exp(- \eta \tfrac{T(X) Y}{p(X)} + b(X))\rangle
            \bigg| X
            \right]
        \right]
    - \frac{1}{\eta}\,.
\end{align*}
Then through differentiation we have
\begin{align*}
    \nabla_{T_x} \eta \bar{\Lambda}(T)
     & = \nabla_{T_x} \E\!\left[
        \langle p_\star, \tfrac{T_x Y}{p(X)} + b_x \rangle
        + \frac{1}{\eta}
        \langle q, \exp(- \eta \tfrac{T_x Y}{p(X)} + b_x)\rangle
        \right]
    \\
     & = \E\!\left[
        \tfrac{Y}{p(x)} p_\star
        - \tfrac{Y}{p(x)} q \odot \exp(- \tfrac{\eta Y}{p(x)} T_x + b_x)
        \right]\,,
\end{align*}
and
\begin{align*}
    \nabla_{b_x} \eta \bar{\Lambda}(T)
     & = \nabla_{b_x} \E\!\left[
        \langle p_\star, \tfrac{T_x Y}{p(X)} + b_x \rangle
        + \frac{1}{\eta}
        \langle q, \exp(- \eta \tfrac{T_x Y}{p(X)} + b_x)\rangle
        \right]
    \\
     & = \E\!\left[
        p_\star
        - \tfrac{1}{\eta} q \odot \exp(- \tfrac{\eta Y}{p(x)} T_x + b_x)
        \right]\,,
\end{align*}
where the expectation is over $(p_\star, f) \sim \nu$.
We want to find $T_x$ such that the gradient is zero, so for coordinate $z \in \cC$ we have
\begin{align*}
    q(z) \E\!\left[
        f(x)\exp(- \tfrac{\eta f(x)}{p(x)} T_{x,z} + b_{x,z})
        \right]
     & =
    \E\!\left[
        f(x) p_{\star, z}
        \right]
    =
    \PP(x_\star = z)
    \E\!\left[
        f(x)
        | x_\star = z
        \right]\,.
\end{align*}
